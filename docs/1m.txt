# 1. Make sure PERSONAL main is current
git checkout main
git pull --ff-only origin main

# 2. Get latest LBG main
git fetch company main

# 3. Create a fresh LBG sync branch based on LBG main
SYNC_BRANCH="rove-sync-$(date +%Y%m%d-%H%M%S)"
git switch -c "$SYNC_BRANCH" company/main

# 4. Replace the application code with PERSONAL main
git read-tree --reset -u origin/main

# 5. IMPORTANT:
# Restore LBG-protected GitHub files so company rules are not violated
git checkout company/main -- \
  .github/checkov-config.yml \
  .github/workflows/core-actions-manager.yml \
  .github/workflows/core-checkov-action.yml \
  .github/workflows/core-linter.yml \
  .github/workflows/core-semantic-pr.yml

# 6. Stage everything
git add -A

# 7. Check exactly what will go to LBG
git status
git diff --cached --stat

# 8. Commit the sync
git commit -m "Sync ROVE application from personal main"

# 9. Push the new branch to LBG
git push -u company "$SYNC_BRANCH"

# 10. Return your local repo to PERSONAL main
git switch main

echo
echo "LBG branch created: $SYNC_BRANCH"
echo "Create PR: $SYNC_BRANCH -> main"

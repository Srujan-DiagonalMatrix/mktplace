#!/usr/bin/env bash
# Place in the repository root. Requires Node.js and npm.
# Usage: bash ui-rebuild.sh [--restart-web]
set -Eeuo pipefail

cd "$(dirname "$0")"
restart_web=0
case "${1:-}" in
  "") ;;
  --restart-web) restart_web=1 ;;
  *) echo "Usage: bash ui-rebuild.sh [--restart-web]"; exit 2 ;;
esac

test -f package.json && test -f package-lock.json || {
  echo "Place ui-rebuild.sh beside package.json and package-lock.json."
  exit 1
}
command -v node >/dev/null
command -v npm >/dev/null

compose=(docker compose)
if [ "$restart_web" -eq 1 ]; then
  docker compose version >/dev/null
  if [ -f .env.docker ]; then
    compose+=(--env-file .env.docker)
  fi
  "${compose[@]}" config --services | grep -qx web || {
    echo "No Docker Compose web service found."
    exit 1
  }
fi

stage=$(mktemp -d "$PWD/.ui-build.XXXXXX")
trap 'rm -rf "$stage"' EXIT
trap 'echo "UI rebuild failed at line $LINENO. Deployment was not started." >&2' ERR

echo "Source commit: $(git rev-parse --short HEAD 2>/dev/null || echo unknown)"
echo "Node: $(node --version); npm: $(npm --version)"
echo "Installing dependencies..."
npm ci --include=dev

echo "Building the UI..."
npm run build -- --outDir "$stage" --emptyOutDir
test -s "$stage/index.html"

# Preserve the dist directory itself because local Nginx bind-mounts it.
# Copy assets first and publish index.html last. Keep older hashed assets
# available for browser tabs opened before this rebuild.
mkdir -p dist
for item in "$stage"/* "$stage"/.[!.]* "$stage"/..?*; do
  [ -e "$item" ] || continue
  [ "$(basename "$item")" = index.html ] && continue
  cp -R "$item" dist/
done
cp "$stage/index.html" dist/index.html

if [ "$restart_web" -eq 1 ]; then
  "${compose[@]}" up -d --no-deps --force-recreate web
  echo "Local Docker web container refreshed. Hard-refresh your browser."
fi

echo "UI ready in dist/."
echo "Build-only mode does not deploy to GCP; run your deployment command next."

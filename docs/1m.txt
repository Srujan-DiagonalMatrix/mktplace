#!/usr/bin/env bash
set -euo pipefail

PROJECT_DIR="/Users/srujan.kumar/Documents/git/rove"

GCP_PROJECT="tactical-innovation-f2b97b"
REGION="europe-west1"
REPOSITORY="cloud-run-source-deploy"

WEB_SERVICE="rove-web"
API_SERVICE="rove-api"

WEB_IMAGE="rove-web"
API_IMAGE="rove-api"

cd "$PROJECT_DIR"

echo "=================================================="
echo " ROVE - GCP DEPLOYMENT"
echo " API + WEB"
echo "=================================================="

# --------------------------------------------------
# GCP AUTH
# --------------------------------------------------

gcloud config set project "$GCP_PROJECT" >/dev/null

if ! gcloud auth print-access-token >/dev/null 2>&1; then
    gcloud auth login
fi

gcloud auth configure-docker \
    "${REGION}-docker.pkg.dev" \
    --quiet >/dev/null

TIMESTAMP="$(date +%Y%m%d-%H%M%S)"

API_VERSION_IMAGE="${REGION}-docker.pkg.dev/${GCP_PROJECT}/${REPOSITORY}/${API_IMAGE}:${TIMESTAMP}"

WEB_VERSION_IMAGE="${REGION}-docker.pkg.dev/${GCP_PROJECT}/${REPOSITORY}/${WEB_IMAGE}:${TIMESTAMP}"


# ==================================================
# 1. CREATE CLOUD RUN API DOCKERFILE
# ==================================================

echo
echo "=================================================="
echo "[1/8] PREPARE + BUILD API"
echo "=================================================="

cat > server/Dockerfile.gcp <<'EOF'
FROM python:3.12-slim

WORKDIR /app

ENV PYTHONDONTWRITEBYTECODE=1
ENV PYTHONUNBUFFERED=1

COPY requirements.txt .

RUN pip install \
    --no-cache-dir \
    --trusted-host pypi.org \
    --trusted-host files.pythonhosted.org \
    --trusted-host pypi.python.org \
    -r requirements.txt

COPY app ./app
COPY scripts ./scripts
COPY alembic ./alembic
COPY alembic.ini ./alembic.ini

EXPOSE 8080

# IMPORTANT:
# Do NOT run Alembic here.
# Cloud Run must start Uvicorn immediately.
CMD ["sh", "-c", "exec uvicorn app.main:app --host 0.0.0.0 --port ${PORT:-8080}"]
EOF


echo "Building API image:"

echo "$API_VERSION_IMAGE"

docker buildx build \
    --platform linux/amd64 \
    -f server/Dockerfile.gcp \
    -t "$API_VERSION_IMAGE" \
    --push \
    server


# ==================================================
# 2. DEPLOY API
# ==================================================

echo
echo "=================================================="
echo "[2/8] DEPLOY API TO CLOUD RUN"
echo "=================================================="

gcloud run deploy "$API_SERVICE" \
    --image "$API_VERSION_IMAGE" \
    --region "$REGION" \
    --platform managed \
    --port 8080 \
    --no-invoker-iam-check \
    --quiet


# ==================================================
# 3. GET + TEST API
# ==================================================

echo
echo "=================================================="
echo "[3/8] TEST API"
echo "=================================================="

API_URL="$(gcloud run services describe "$API_SERVICE" \
    --region "$REGION" \
    --format='value(status.url)')"

echo
echo "API URL:"
echo "$API_URL"

echo
echo "Testing FastAPI startup..."

HTTP_CODE="$(curl -s \
    -o /tmp/rove-api-test.html \
    -w "%{http_code}" \
    "$API_URL/docs" || true)"

echo "API /docs HTTP status: $HTTP_CODE"

if [ "$HTTP_CODE" != "200" ]; then

    echo
    echo "ERROR: API did not respond successfully."
    echo
    echo "Recent API logs:"
    echo

    gcloud logging read \
      "resource.type=cloud_run_revision AND resource.labels.service_name=${API_SERVICE}" \
      --limit=40 \
      --freshness=10m \
      --format="value(timestamp,severity,textPayload)"

    exit 1
fi

echo "API started successfully."


# ==================================================
# 4. BUILD REACT
# ==================================================

echo
echo "=================================================="
echo "[4/8] BUILD REACT"
echo "=================================================="

rm -rf dist

NODE_USE_SYSTEM_CA=1 npm ci

NODE_USE_SYSTEM_CA=1 npm run build

if [ ! -d "dist" ]; then
    echo "ERROR: React dist directory not created."
    exit 1
fi

echo
echo "React build successful."


# ==================================================
# 5. FIX DOCKERIGNORE
# ==================================================

echo
echo "=================================================="
echo "[5/8] PREPARE WEB BUILD CONTEXT"
echo "=================================================="

if [ -f ".dockerignore" ]; then

python3 <<'PY'
from pathlib import Path

p = Path(".dockerignore")

lines = p.read_text().splitlines()

blocked = {
    "dist",
    "./dist",
    "**/dist",
    "*/dist"
}

clean = []

for line in lines:

    value = line.strip().rstrip("/")

    if value in blocked:
        print("Removing .dockerignore rule:", line)
        continue

    clean.append(line)

p.write_text("\n".join(clean) + "\n")

PY

fi


# ==================================================
# 6. CREATE NGINX PROXY
# ==================================================

echo
echo "=================================================="
echo "[6/8] CONFIGURE API PROXY"
echo "=================================================="

API_HOST="$(echo "$API_URL" | sed 's#https://##' | sed 's#/$##')"

cat > nginx.conf.gcp <<EOF
server {

    listen 8080;
    server_name _;

    root /usr/share/nginx/html;
    index index.html;


    # -----------------------------------------
    # BACKEND API
    # -----------------------------------------

    location /api/ {

        proxy_ssl_server_name on;
        proxy_ssl_name ${API_HOST};

        proxy_set_header Host ${API_HOST};

        proxy_set_header X-Real-IP \$remote_addr;
        proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto https;

        proxy_pass ${API_URL}/api/;

    }


    # -----------------------------------------
    # HEALTH CHECK
    # -----------------------------------------

    location = /health {

        proxy_ssl_server_name on;
        proxy_ssl_name ${API_HOST};

        proxy_set_header Host ${API_HOST};

        proxy_pass ${API_URL}/api/health;

    }


    # -----------------------------------------
    # REACT SPA
    # -----------------------------------------

    location / {

        try_files \$uri \$uri/ /index.html;

    }

}
EOF


cat > Dockerfile.gcp <<'EOF'
FROM nginx:alpine

COPY nginx.conf.gcp /etc/nginx/conf.d/default.conf

COPY dist /usr/share/nginx/html

EXPOSE 8080

CMD ["nginx", "-g", "daemon off;"]
EOF


# ==================================================
# 7. BUILD WEB IMAGE
# ==================================================

echo
echo "=================================================="
echo "[7/8] BUILD + PUSH WEB"
echo "=================================================="

docker buildx build \
    --platform linux/amd64 \
    -f Dockerfile.gcp \
    -t "$WEB_VERSION_IMAGE" \
    --push \
    .


# ==================================================
# 8. DEPLOY WEB
# ==================================================

echo
echo "=================================================="
echo "[8/8] DEPLOY WEB"
echo "=================================================="

gcloud run deploy "$WEB_SERVICE" \
    --image "$WEB_VERSION_IMAGE" \
    --region "$REGION" \
    --platform managed \
    --port 8080 \
    --no-invoker-iam-check \
    --quiet


WEB_URL="$(gcloud run services describe "$WEB_SERVICE" \
    --region "$REGION" \
    --format='value(status.url)')"


echo
echo "=================================================="
echo " ROVE DEPLOYMENT SUCCESSFUL"
echo "=================================================="

echo
echo "API:"
echo "$API_URL"

echo
echo "WEB:"
echo "$WEB_URL"

echo
echo "Images:"
echo "API : $API_VERSION_IMAGE"
echo "WEB : $WEB_VERSION_IMAGE"

echo
echo "=================================================="
echo " API TEST"
echo "=================================================="

curl -s -o /dev/null \
    -w "FastAPI docs: HTTP %{http_code}\n" \
    "$API_URL/docs" || true

echo
echo "=================================================="
echo " WEB TEST"
echo "=================================================="

curl -s -o /dev/null \
    -w "ROVE website: HTTP %{http_code}\n" \
    "$WEB_URL/" || true

echo
echo "=================================================="

open "$WEB_URL" 2>/dev/null || true

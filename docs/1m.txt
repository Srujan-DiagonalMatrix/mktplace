cd /Users/srujan.kumar/Documents/git/rove

PROJECT="tactical-innovation-f2b97b"
REGION="europe-west1"
REPOSITORY="cloud-run-source-deploy"
IMAGE_NAME="rove-web"

FULL_IMAGE="${REGION}-docker.pkg.dev/${PROJECT}/${REPOSITORY}/${IMAGE_NAME}"

echo
echo "============================================================"
echo "1. LIST ROVE-WEB IMAGE VERSIONS + TAGS"
echo "============================================================"

gcloud artifacts docker images list "$FULL_IMAGE" \
  --project="$PROJECT" \
  --include-tags \
  --sort-by="~UPDATE_TIME" \
  --format="table(
    IMAGE,
    DIGEST,
    TAGS,
    CREATE_TIME,
    UPDATE_TIME
  )"

echo
echo "============================================================"
echo "2. LIST ROVE-WEB TAGS"
echo "============================================================"

gcloud artifacts docker tags list "$FULL_IMAGE" \
  --project="$PROJECT" \
  --format="table(
    tag,
    version
  )" \
  2>&1 || true

echo
echo "============================================================"
echo "3. GET LATEST DIGEST"
echo "============================================================"

LATEST_DIGEST="$(gcloud artifacts docker images list "$FULL_IMAGE" \
  --project="$PROJECT" \
  --include-tags \
  --sort-by="~UPDATE_TIME" \
  --limit=1 \
  --format='value(DIGEST)')"

echo "Latest digest:"
echo "$LATEST_DIGEST"

echo
echo "============================================================"
echo "4. BUILD DEPLOYABLE IMAGE REFERENCE"
echo "============================================================"

if [ -n "$LATEST_DIGEST" ]; then

  DEPLOY_IMAGE="${FULL_IMAGE}@${LATEST_DIGEST}"

  echo "Deployable image:"
  echo "$DEPLOY_IMAGE"

else

  echo "ERROR: No rove-web image digest found."
  echo "This means the web image was not successfully pushed."
  exit 1

fi

echo
echo "============================================================"
echo "5. VERIFY IMAGE EXISTS"
echo "============================================================"

gcloud artifacts docker images describe "$DEPLOY_IMAGE" \
  --project="$PROJECT" \
  --format="yaml(
    image_summary.digest,
    image_summary.fully_qualified_digest
  )"

echo
echo "============================================================"
echo "6. CURRENT ROVE-WEB CLOUD RUN STATUS"
echo "============================================================"

gcloud run services describe rove-web \
  --project="$PROJECT" \
  --region="$REGION" \
  --format="yaml(
    metadata.name,
    status.conditions,
    status.latestCreatedRevisionName,
    status.latestReadyRevisionName
  )" \
  2>&1 || true

echo
echo "============================================================"
echo " ANALYSIS COMPLETE"
echo "============================================================"

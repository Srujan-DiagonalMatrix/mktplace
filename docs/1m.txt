#!/usr/bin/env bash
set -euo pipefail

# ============================================================
# ROVE - COMPLETE GCP CLEANUP
#
# Deletes:
#   - rove-api Cloud Run service
#   - rove-web Cloud Run service
#   - rove-postgres Cloud SQL instance + ALL DATABASE DATA
#   - cloud-run-source-deploy Artifact Registry repository
#   - locally generated deployment files
#
# DOES NOT:
#   - delete GCP project
#   - disable shared GCP APIs
#
# WARNING:
#   DATABASE DATA WILL BE PERMANENTLY DELETED.
# ============================================================


PROJECT_DIR="/Users/srujan.kumar/Documents/git/rove"

PROJECT="tactical-innovation-f2b97b"
REGION="europe-west1"

API_SERVICE="rove-api"
WEB_SERVICE="rove-web"

DB_INSTANCE="rove-postgres"

REPOSITORY="cloud-run-source-deploy"


cd "$PROJECT_DIR"


echo
echo "============================================================"
echo " ROVE COMPLETE GCP CLEANUP"
echo "============================================================"
echo
echo "Project : $PROJECT"
echo "Region  : $REGION"
echo
echo "WARNING:"
echo "ROVE PostgreSQL data will be permanently deleted."
echo "============================================================"


gcloud config set project "$PROJECT" >/dev/null


# ============================================================
# 1. DELETE CLOUD RUN API
# ============================================================

echo
echo "============================================================"
echo "[1/5] DELETE ROVE API"
echo "============================================================"


if gcloud run services describe "$API_SERVICE" \
    --project="$PROJECT" \
    --region="$REGION" \
    >/dev/null 2>&1
then

    echo "Deleting $API_SERVICE..."

    gcloud run services delete "$API_SERVICE" \
        --project="$PROJECT" \
        --region="$REGION" \
        --quiet

else

    echo "$API_SERVICE does not exist."

fi


# ============================================================
# 2. DELETE CLOUD RUN WEB
# ============================================================

echo
echo "============================================================"
echo "[2/5] DELETE ROVE WEB"
echo "============================================================"


if gcloud run services describe "$WEB_SERVICE" \
    --project="$PROJECT" \
    --region="$REGION" \
    >/dev/null 2>&1
then

    echo "Deleting $WEB_SERVICE..."

    gcloud run services delete "$WEB_SERVICE" \
        --project="$PROJECT" \
        --region="$REGION" \
        --quiet

else

    echo "$WEB_SERVICE does not exist."

fi


# ============================================================
# 3. DELETE CLOUD SQL
# ============================================================

echo
echo "============================================================"
echo "[3/5] DELETE POSTGRESQL"
echo "============================================================"


if gcloud sql instances describe "$DB_INSTANCE" \
    --project="$PROJECT" \
    >/dev/null 2>&1
then

    echo "Cloud SQL instance found."
    echo "Disabling deletion protection..."

    gcloud sql instances patch "$DB_INSTANCE" \
        --project="$PROJECT" \
        --no-deletion-protection \
        --quiet \
        >/dev/null 2>&1 || true


    echo "Deleting $DB_INSTANCE..."


    gcloud sql instances delete "$DB_INSTANCE" \
        --project="$PROJECT" \
        --no-enable-final-backup \
        --quiet


    echo "PostgreSQL deleted."

else

    echo "$DB_INSTANCE does not exist."

fi


# ============================================================
# 4. DELETE ARTIFACT REGISTRY
# ============================================================

echo
echo "============================================================"
echo "[4/5] DELETE ROVE CONTAINER REPOSITORY"
echo "============================================================"


if gcloud artifacts repositories describe "$REPOSITORY" \
    --project="$PROJECT" \
    --location="$REGION" \
    >/dev/null 2>&1
then

    echo "Deleting Artifact Registry repository:"
    echo "$REPOSITORY"

    gcloud artifacts repositories delete "$REPOSITORY" \
        --project="$PROJECT" \
        --location="$REGION" \
        --quiet

else

    echo "$REPOSITORY does not exist."

fi


# ============================================================
# 5. CLEAN LOCAL GENERATED FILES
# ============================================================

echo
echo "============================================================"
echo "[5/5] CLEAN LOCAL DEPLOYMENT FILES"
echo "============================================================"


rm -f \
    Dockerfile.gcp \
    Dockerfile.gcp.dockerignore \
    nginx.conf.gcp \
    server/Dockerfile.gcp


rm -rf \
    .rove-gcp-web-build \
    /tmp/rove-api-docs.html \
    /tmp/rove-cloudrun-debug.log


echo "Local deployment files removed."


# ============================================================
# FINAL VERIFICATION
# ============================================================

echo
echo "============================================================"
echo " VERIFY CLEAN STATE"
echo "============================================================"

echo
echo "Cloud Run:"
gcloud run services list \
    --project="$PROJECT" \
    --region="$REGION" \
    --filter="metadata.name:(rove-api OR rove-web)" \
    --format="table(metadata.name,status.url)" || true


echo
echo "Cloud SQL:"
gcloud sql instances list \
    --project="$PROJECT" \
    --filter="name=$DB_INSTANCE" \
    --format="table(name,state)" || true


echo
echo "Artifact Registry:"
gcloud artifacts repositories list \
    --project="$PROJECT" \
    --location="$REGION" \
    --filter="name~$REPOSITORY" \
    --format="table(name)" || true


echo
echo "============================================================"
echo " ROVE CLEANUP COMPLETE"
echo "============================================================"
echo
echo "ROVE GCP environment is ready for a fresh installation."
echo

#!/usr/bin/env bash
set -euo pipefail

# ==========================================================
# ROVE - GCP DEPLOYMENT
#
# Behaviour:
#   1. Keep existing PostgreSQL data
#   2. Create PostgreSQL only if it does not exist
#   3. Ensure ROVE database exists
#   4. Build + deploy API
#   5. Run Alembic migrations automatically
#   6. Build React
#   7. Build + deploy Web
#   8. Test API, DB and Web
#
# NOTE:
#   Existing database is NEVER deleted by this script.
# ==========================================================


# ==========================================================
# CONFIGURATION
# ==========================================================

PROJECT_DIR="/Users/srujan.kumar/Documents/git/rove"

GCP_PROJECT="tactical-innovation-f2b97b"
REGION="europe-west1"
REPOSITORY="cloud-run-source-deploy"

WEB_SERVICE="rove-web"
API_SERVICE="rove-api"

WEB_IMAGE="rove-web"
API_IMAGE="rove-api"

DB_INSTANCE="rove-postgres"
DB_NAME="rove"
DB_USER="postgres"

# PoV password - >12 chars, uppercase/lowercase/numeric/special
DB_PASSWORD="RovePov#2026Secure!"

DEPLOYMENT_ID="$(date +%Y%m%d-%H%M%S)"

cd "$PROJECT_DIR"


echo
echo "=========================================================="
echo " ROVE GCP DEPLOYMENT"
echo "=========================================================="
echo "Deployment ID : $DEPLOYMENT_ID"
echo "Project       : $GCP_PROJECT"
echo "Region        : $REGION"
echo "DB Instance   : $DB_INSTANCE"
echo "=========================================================="


# ==========================================================
# PRE-CHECKS
# ==========================================================

command -v gcloud >/dev/null 2>&1 || {
    echo "ERROR: gcloud CLI is not installed."
    exit 1
}

command -v docker >/dev/null 2>&1 || {
    echo "ERROR: Docker is not installed."
    exit 1
}

command -v npm >/dev/null 2>&1 || {
    echo "ERROR: npm is not installed."
    exit 1
}

docker info >/dev/null 2>&1 || {
    echo "ERROR: Docker Desktop is not running."
    exit 1
}


# ==========================================================
# 0. GCP AUTHENTICATION
# ==========================================================

echo
echo "=========================================================="
echo "[0/9] GCP AUTHENTICATION"
echo "=========================================================="

if ! gcloud auth print-access-token >/dev/null 2>&1; then
    gcloud auth login
fi

gcloud config set project "$GCP_PROJECT" >/dev/null

echo "Enabling required APIs..."

gcloud services enable \
    run.googleapis.com \
    sqladmin.googleapis.com \
    artifactregistry.googleapis.com \
    --project="$GCP_PROJECT" \
    --quiet

gcloud auth configure-docker \
    "${REGION}-docker.pkg.dev" \
    --quiet >/dev/null


# ==========================================================
# 1. CHECK / CREATE POSTGRESQL
# ==========================================================

echo
echo "=========================================================="
echo "[1/9] CHECK POSTGRESQL"
echo "=========================================================="

if gcloud sql instances describe "$DB_INSTANCE" \
    --project="$GCP_PROJECT" \
    >/dev/null 2>&1
then

    echo "Existing PostgreSQL instance found:"
    echo "$DB_INSTANCE"
    echo
    echo "Database will be preserved."

else

    echo "PostgreSQL instance not found."
    echo "Creating bank-compliant PostgreSQL..."

    gcloud sql instances create "$DB_INSTANCE" \
        --project="$GCP_PROJECT" \
        --database-version=POSTGRES_16 \
        --edition=ENTERPRISE \
        --tier=db-f1-micro \
        --region="$REGION" \
        --storage-size=10GB \
        --storage-type=SSD \
        --availability-type=zonal \
        --ssl-mode=ENCRYPTED_ONLY \
        --database-flags=cloudsql.iam_authentication=on,cloudsql.enable_pgaudit=on,pgaudit.log=all \
        --enable-password-policy \
        --password-policy-min-length=12 \
        --password-policy-complexity=COMPLEXITY_DEFAULT \
        --password-policy-disallow-username-substring \
        --root-password="$DB_PASSWORD" \
        --no-deletion-protection

    echo
    echo "PostgreSQL created successfully."

fi


# ==========================================================
# 2. CHECK / CREATE ROVE DATABASE
# ==========================================================

echo
echo "=========================================================="
echo "[2/9] CHECK ROVE DATABASE"
echo "=========================================================="

if gcloud sql databases describe "$DB_NAME" \
    --instance="$DB_INSTANCE" \
    --project="$GCP_PROJECT" \
    >/dev/null 2>&1
then

    echo "Database '$DB_NAME' already exists."
    echo "Existing data will be preserved."

else

    echo "Database '$DB_NAME' not found."
    echo "Creating database..."

    gcloud sql databases create "$DB_NAME" \
        --instance="$DB_INSTANCE" \
        --project="$GCP_PROJECT"

fi


CONNECTION_NAME="$(gcloud sql instances describe "$DB_INSTANCE" \
    --project="$GCP_PROJECT" \
    --format='value(connectionName)')"

echo
echo "Cloud SQL connection:"
echo "$CONNECTION_NAME"


# ==========================================================
# 3. CONFIGURE CLOUD RUN DATABASE ACCESS
# ==========================================================

echo
echo "=========================================================="
echo "[3/9] CONFIGURE CLOUD SQL ACCESS"
echo "=========================================================="

PROJECT_NUMBER="$(gcloud projects describe "$GCP_PROJECT" \
    --format='value(projectNumber)')"

CLOUD_RUN_SA="${PROJECT_NUMBER}-compute@developer.gserviceaccount.com"

echo "Cloud Run service account:"
echo "$CLOUD_RUN_SA"

# Safe to run repeatedly.
gcloud projects add-iam-policy-binding "$GCP_PROJECT" \
    --member="serviceAccount:${CLOUD_RUN_SA}" \
    --role="roles/cloudsql.client" \
    --condition=None \
    --quiet \
    >/dev/null 2>&1 || true


# ==========================================================
# 4. BUILD API IMAGE
# ==========================================================

echo
echo "=========================================================="
echo "[4/9] BUILD API"
echo "=========================================================="

cat > server/Dockerfile.gcp <<'EOF'
FROM python:3.12-slim

WORKDIR /app

ENV PYTHONDONTWRITEBYTECODE=1
ENV PYTHONUNBUFFERED=1

COPY requirements.txt .

RUN pip install \
    --no-cache-dir \
    --trusted-host pypi.org \
    --trusted-host files.pythonhosted.org \
    --trusted-host pypi.python.org \
    -r requirements.txt

COPY app ./app
COPY scripts ./scripts
COPY alembic ./alembic
COPY alembic.ini ./alembic.ini

EXPOSE 8080

# Every deployment:
#   1. Upgrade DB schema
#   2. Start API
#
# Existing data remains intact.
CMD ["sh", "-c", "alembic upgrade head && exec uvicorn app.main:app --host 0.0.0.0 --port ${PORT:-8080}"]
EOF


API_VERSION_IMAGE="${REGION}-docker.pkg.dev/${GCP_PROJECT}/${REPOSITORY}/${API_IMAGE}:${DEPLOYMENT_ID}"

echo "Building:"
echo "$API_VERSION_IMAGE"

docker buildx build \
    --platform linux/amd64 \
    -f server/Dockerfile.gcp \
    -t "$API_VERSION_IMAGE" \
    --push \
    server


# ==========================================================
# 5. DEPLOY / RESTART API
# ==========================================================

echo
echo "=========================================================="
echo "[5/9] DEPLOY API"
echo "=========================================================="

DATABASE_URL="postgresql+psycopg://${DB_USER}:${DB_PASSWORD}@/${DB_NAME}?host=/cloudsql/${CONNECTION_NAME}"

gcloud run deploy "$API_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --platform=managed \
    --image="$API_VERSION_IMAGE" \
    --port=8080 \
    --add-cloudsql-instances="$CONNECTION_NAME" \
    --service-account="$CLOUD_RUN_SA" \
    --set-env-vars="DATABASE_URL=${DATABASE_URL},APP_ENV=production,DEPLOYMENT_ID=${DEPLOYMENT_ID}" \
    --no-invoker-iam-check \
    --quiet


API_URL="$(gcloud run services describe "$API_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --format='value(status.url)')"


echo
echo "API deployed:"
echo "$API_URL"

echo
echo "Testing FastAPI..."


API_HTTP="$(curl -s \
    -o /tmp/rove-api-docs.txt \
    -w "%{http_code}" \
    "$API_URL/docs" || true)"


echo "FastAPI /docs HTTP status: $API_HTTP"


if [ "$API_HTTP" != "200" ]; then

    echo
    echo "ERROR: API deployment failed."
    echo
    echo "Recent logs:"
    echo

    gcloud logging read \
        "resource.type=cloud_run_revision AND resource.labels.service_name=${API_SERVICE}" \
        --project="$GCP_PROJECT" \
        --limit=50 \
        --freshness=15m \
        --format="value(timestamp,severity,textPayload)"

    exit 1

fi


echo
echo "Testing PostgreSQL..."

DB_HEALTH="$(curl -s "$API_URL/api/health" || true)"

echo "$DB_HEALTH"


# ==========================================================
# 6. BUILD REACT
# ==========================================================

echo
echo "=========================================================="
echo "[6/9] BUILD REACT"
echo "=========================================================="

rm -rf dist

NODE_USE_SYSTEM_CA=1 npm ci
NODE_USE_SYSTEM_CA=1 npm run build

if [ ! -d "dist" ]; then

    echo "ERROR: React dist directory was not created."
    exit 1

fi

echo "React build successful."


# ==========================================================
# 7. PREPARE WEB / NGINX
# ==========================================================

echo
echo "=========================================================="
echo "[7/9] PREPARE WEB"
echo "=========================================================="

# Make sure dist is available to Docker.
if [ -f ".dockerignore" ]; then

python3 <<'PY'
from pathlib import Path

p = Path(".dockerignore")

lines = p.read_text().splitlines()

blocked = {
    "dist",
    "./dist",
    "**/dist",
    "*/dist"
}

clean = []

for line in lines:

    value = line.strip().rstrip("/")

    if value in blocked:
        print("Removing .dockerignore entry:", line)
        continue

    clean.append(line)

p.write_text("\n".join(clean) + "\n")
PY

fi


API_HOST="$(echo "$API_URL" \
    | sed 's#https://##' \
    | sed 's#/$##')"


cat > nginx.conf.gcp <<EOF
server {

    listen 8080;
    server_name _;

    root /usr/share/nginx/html;
    index index.html;


    # ------------------------------------------------------
    # ROVE BACKEND API
    # ------------------------------------------------------

    location /api/ {

        proxy_ssl_server_name on;
        proxy_ssl_name ${API_HOST};

        proxy_set_header Host ${API_HOST};

        proxy_set_header X-Real-IP \$remote_addr;
        proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto https;

        proxy_pass ${API_URL}/api/;

    }


    # ------------------------------------------------------
    # REACT SPA
    # ------------------------------------------------------

    location / {

        try_files \$uri \$uri/ /index.html;

    }

}
EOF


cat > Dockerfile.gcp <<'EOF'
FROM nginx:alpine

COPY nginx.conf.gcp /etc/nginx/conf.d/default.conf
COPY dist /usr/share/nginx/html

EXPOSE 8080

CMD ["nginx", "-g", "daemon off;"]
EOF


# ==========================================================
# 8. BUILD + DEPLOY WEB
# ==========================================================

echo
echo "=========================================================="
echo "[8/9] BUILD + DEPLOY WEB"
echo "=========================================================="

WEB_VERSION_IMAGE="${REGION}-docker.pkg.dev/${GCP_PROJECT}/${REPOSITORY}/${WEB_IMAGE}:${DEPLOYMENT_ID}"

echo "Building:"
echo "$WEB_VERSION_IMAGE"

docker buildx build \
    --platform linux/amd64 \
    -f Dockerfile.gcp \
    -t "$WEB_VERSION_IMAGE" \
    --push \
    .


gcloud run deploy "$WEB_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --platform=managed \
    --image="$WEB_VERSION_IMAGE" \
    --port=8080 \
    --set-env-vars="DEPLOYMENT_ID=${DEPLOYMENT_ID}" \
    --no-invoker-iam-check \
    --quiet


WEB_URL="$(gcloud run services describe "$WEB_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --format='value(status.url)')"


# ==========================================================
# 9. FINAL TESTS
# ==========================================================

echo
echo "=========================================================="
echo "[9/9] FINAL TESTS"
echo "=========================================================="

echo
echo "API:"
curl -s -o /dev/null \
    -w "FastAPI: HTTP %{http_code}\n" \
    "$API_URL/docs" || true


echo
echo "Database:"
curl -s "$API_URL/api/health" || true


echo
echo
echo "Website:"
curl -s -o /dev/null \
    -w "Web: HTTP %{http_code}\n" \
    "$WEB_URL/" || true


echo
echo
echo "API through Web proxy:"
curl -s "$WEB_URL/api/health" || true


# ==========================================================
# COMPLETE
# ==========================================================

echo
echo
echo "=========================================================="
echo " ROVE DEPLOYMENT SUCCESSFUL"
echo "=========================================================="

echo
echo "Deployment ID:"
echo "$DEPLOYMENT_ID"

echo
echo "PostgreSQL:"
echo "$DB_INSTANCE"

echo
echo "Database:"
echo "$DB_NAME"

echo
echo "Cloud SQL connection:"
echo "$CONNECTION_NAME"

echo
echo "API:"
echo "$API_URL"

echo
echo "Website:"
echo "$WEB_URL"

echo
echo "API image:"
echo "$API_VERSION_IMAGE"

echo
echo "Web image:"
echo "$WEB_VERSION_IMAGE"

echo
echo "=========================================================="
echo " DATABASE DATA PRESERVED"
echo "=========================================================="

open "$WEB_URL" 2>/dev/null || true

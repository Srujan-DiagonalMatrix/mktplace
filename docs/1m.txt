#!/usr/bin/env bash
set -euo pipefail

# ============================================================
# ROVE - EXISTING GCP ENVIRONMENT UPDATE
#
# DOES:
#   - preserve PostgreSQL / Cloud SQL
#   - preserve all database data
#   - rebuild API
#   - redeploy API
#   - run Alembic migrations
#   - inject Admin/Auth/SMTP settings

#
# DOES NOT:

#   - delete PostgreSQL
#   - reset database
#   - recreate Artifact Registry
# ============================================================


# ============================================================
# CONFIG
# ============================================================

PROJECT_DIR="/Users/srujan.kumar/Documents/git/rove"

PROJECT="tactical-innovation-f2b97b"

echo "============================================================"
echo
echo "Deployment ID : $DEPLOYMENT_ID"

  echo "ERROR: gcloud not installed"
  exit 1

  echo "ERROR: npm not installed"

  exit 1

  exit 1
}

docker info >/dev/null 2>&1 || {
  echo "ERROR: Docker Desktop is not running"
  exit 1
}


# ============================================================
# LOAD APPLICATION SETTINGS
# ============================================================

if [ ! -f ".env.docker" ]; then
  echo "ERROR: .env.docker not found."
  exit 1
fi

set -a
source .env.docker
set +a


# Gmail app password is often displayed with spaces.
# Remove spaces before passing it to the application.

SMTP_PASSWORD_RUNTIME="${SMTP_PASSWORD// /}"


# ============================================================
# 1. GCP / EXISTING SERVICES
# ============================================================

echo
echo "============================================================"
echo "[1/8] CHECK EXISTING GCP SERVICES"
echo "============================================================"

gcloud config set project "$PROJECT" >/dev/null


if ! gcloud sql instances describe "$DB_INSTANCE" \
  --project="$PROJECT" >/dev/null 2>&1
then
  echo "ERROR: Existing Cloud SQL instance '$DB_INSTANCE' not found."
  exit 1
fi


DB_STATE="$(gcloud sql instances describe "$DB_INSTANCE" \
  --project="$PROJECT" \
  --format='value(state)')"

echo "PostgreSQL state: $DB_STATE"

if [ "$DB_STATE" != "RUNNABLE" ]; then
  echo "ERROR: PostgreSQL is not RUNNABLE."
  exit 1
fi


if ! gcloud run services describe "$API_SERVICE" \
  --project="$PROJECT" \
  --region="$REGION" >/dev/null 2>&1
then
  echo "ERROR: Existing rove-api service not found."
  exit 1
fi


if ! gcloud run services describe "$WEB_SERVICE" \
  --project="$PROJECT" \
  --region="$REGION" >/dev/null 2>&1
then
  echo "ERROR: Existing rove-web service not found."
  exit 1
fi


# ============================================================
# EXISTING URLS
# ============================================================

CURRENT_API_URL="$(gcloud run services describe "$API_SERVICE" \
  --project="$PROJECT" \
  --region="$REGION" \
  --format='value(status.url)')"

CURRENT_WEB_URL="$(gcloud run services describe "$WEB_SERVICE" \
  --project="$PROJECT" \
  --region="$REGION" \
  --format='value(status.url)')"


echo
echo "Current API:"
echo "$CURRENT_API_URL"

echo
echo "Current Web:"
echo "$CURRENT_WEB_URL"


# ============================================================
# PRESERVE EXISTING DATABASE_URL
# ============================================================

echo
echo "Reading existing Cloud SQL DATABASE_URL..."

DATABASE_URL="$(
  gcloud run services describe "$API_SERVICE" \
    --project="$PROJECT" \
    --region="$REGION" \
    --format=json |
  python3 <<'PY'
import json
import sys

data = json.load(sys.stdin)

containers = (
    data.get("spec", {})
        .get("template", {})
        .get("spec", {})
        .get("containers", [])
)

if not containers:
    raise SystemExit("ERROR: rove-api container configuration not found")

for item in containers[0].get("env", []):
    if item.get("name") == "DATABASE_URL":
        print(item.get("value", ""))
        break
else:
    raise SystemExit("ERROR: DATABASE_URL not found on existing rove-api service")
PY
)"


if [ -z "$DATABASE_URL" ]; then
  echo "ERROR: Existing DATABASE_URL is empty."
  exit 1
fi


CONNECTION_NAME="$(gcloud sql instances describe "$DB_INSTANCE" \
  --project="$PROJECT" \
  --format='value(connectionName)')"


# ============================================================
# 2. VERIFY ALEMBIC
# ============================================================

echo
echo "============================================================"
echo "[2/8] VERIFY ALEMBIC CONFIG"
echo "============================================================"

python3 <<'PY'
from pathlib import Path

path = Path("server/alembic/env.py")

if not path.exists():
    raise SystemExit("ERROR: server/alembic/env.py not found")

text = path.read_text()

old = 'config.set_main_option("sqlalchemy.url", database_url)'

new = '''config.set_main_option(
        "sqlalchemy.url",
        database_url.replace("%", "%%")
    )'''

if 'database_url.replace("%", "%%")' in text:
    print("Alembic interpolation fix already present.")

elif old in text:
    path.write_text(text.replace(old, new))
    print("Alembic interpolation fix applied.")

else:
    raise SystemExit(
        "ERROR: Unable to locate Alembic sqlalchemy.url configuration."
    )
PY

RUN pip install \
  --no-cache-dir \
  --trusted-host pypi.org \
  --trusted-host files.pythonhosted.org \
  --trusted-host pypi.python.org \
  -r requirements.txt

COPY app ./app
COPY scripts ./scripts
COPY alembic ./alembic
COPY alembic.ini ./alembic.ini

EXPOSE 8080

CMD ["sh", "-c", "alembic upgrade head && exec uvicorn app.main:app --host 0.0.0.0 --port ${PORT:-8080}"]
DOCKERFILE

    --project="$PROJECT" \

  --no-invoker-iam-check \
  --quiet


API_URL="$(gcloud run services describe "$API_SERVICE" \
  --project="$PROJECT" \
  --region="$REGION" \
  --format='value(status.url)')"


echo
echo "API URL:"

  | sed 's#^https://##' \
  | sed 's#/$##')"


cat > "$WEB_BUILD_DIR/nginx.conf" <<NGINX
server {

    listen 8080;
    server_name _;

    root /usr/share/nginx/html;
    index index.html;

    location /api/ {

        proxy_ssl_server_name on;
        proxy_ssl_name ${API_HOST};

        proxy_set_header Host ${API_HOST};

        proxy_set_header X-Real-IP \$remote_addr;
        proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto https;

        proxy_connect_timeout 30s;
        proxy_read_timeout 120s;

        proxy_pass ${API_URL}/api/;
    }

    location / {
        try_files \$uri \$uri/ /index.html;
    }
}
NGINX


cat > "$WEB_BUILD_DIR/Dockerfile" <<'DOCKERFILE'
FROM nginx:alpine

COPY nginx.conf /etc/nginx/conf.d/default.conf
COPY dist /usr/share/nginx/html

EXPOSE 8080

cd /Users/srujan.kumar/Documents/git/rove

PROJECT="tactical-innovation-f2b97b"
REGION="europe-west1"
WEB_SERVICE="rove-web"

WEB_IMAGE="europe-west1-docker.pkg.dev/tactical-innovation-f2b97b/cloud-run-source-deploy/rove-web:20261029-152916"

DEPLOYMENT_ID="$(date +%Y%m%d-%H%M%S)"

echo
echo "============================================================"
echo " DEPLOY CORRECT ROVE WEB IMAGE"
echo "============================================================"

echo "Image:"
echo "$WEB_IMAGE"

echo
echo "Removing failed rove-web service first..."

gcloud run services delete "$WEB_SERVICE" \
  --project="$PROJECT" \
  --region="$REGION" \
  --quiet 2>/dev/null || true

echo
echo "Deploying rove-web..."

gcloud run deploy "$WEB_SERVICE" \
  --project="$PROJECT" \
  --region="$REGION" \
  --platform=managed \
  --execution-environment=gen2 \
  --image="$WEB_IMAGE" \
  --port=8080 \
  --memory=512Mi \
  --cpu=1 \
  --min=0 \
  --max=2 \
  --set-env-vars="DEPLOYMENT_ID=${DEPLOYMENT_ID}" \
  --no-invoker-iam-check \
  --quiet

echo
echo "============================================================"
echo " GET WEB URL"
echo "============================================================"

WEB_URL="$(gcloud run services describe "$WEB_SERVICE" \
  --project="$PROJECT" \
  --region="$REGION" \
  --format='value(status.url)')"

echo
echo "ROVE Web:"
echo "$WEB_URL"

echo
echo "============================================================"
echo " TEST WEB"
echo "============================================================"

curl -s \
  -o /dev/null \
  -w "ROVE Web HTTP %{http_code}\n" \
  "$WEB_URL/"

echo
echo "============================================================"
echo " TEST API THROUGH WEB"
echo "============================================================"

curl -s "$WEB_URL/api/health"

echo
echo
echo "============================================================"
echo " DEPLOYMENT COMPLETE"
echo "============================================================"

bash <<'SH'
set -u

PROJECT="tactical-innovation-f2b97b"
REGION="europe-west1"
EXECUTION="rove-migrate-update-20261110150441-9e23e9e1f840-28ckd"
JOB="rove-migrate-update-20261110150441-9e23e9e1f840"

echo "=== 1. Failed execution status ==="
gcloud --project="$PROJECT" run jobs executions describe "$EXECUTION" \
  --region="$REGION" \
  --format='yaml(status.conditions,status.failedCount,status.succeededCount,status.logUri)' || true

echo
echo "=== 2. Migration container logs ==="
gcloud --project="$PROJECT" beta run jobs executions logs read "$EXECUTION" \
  --region="$REGION" --limit=200 || \
gcloud --project="$PROJECT" run jobs logs read "$JOB" \
  --region="$REGION" --limit=200

echo
echo "=== 3. Job configuration (image and command only) ==="
gcloud --project="$PROJECT" run jobs describe "$JOB" \
  --region="$REGION" \
  --format='yaml(spec.template.spec.template.spec.containers.image,spec.template.spec.template.spec.containers.command,spec.template.spec.template.spec.containers.args)' || true
SH

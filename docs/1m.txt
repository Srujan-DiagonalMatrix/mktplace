#!/usr/bin/env bash
set -euo pipefail

PROJECT_DIR="/Users/srujan.kumar/Documents/git/rove"

GCP_PROJECT="tactical-innovation-f2b97b"
REGION="europe-west1"
REPOSITORY="cloud-run-source-deploy"

WEB_SERVICE="rove-web"
API_SERVICE="rove-api"

WEB_IMAGE="rove-web"
API_IMAGE="rove-api"

DB_INSTANCE="rove-postgres"
DB_NAME="rove"
DB_USER="rove"

# PoV only - simple disposable password
DB_PASSWORD="RovePov123!"

DEPLOYMENT_ID="$(date +%Y%m%d-%H%M%S)"

cd "$PROJECT_DIR"

echo "=================================================="
echo " ROVE FULL RESET + DEPLOYMENT"
echo "=================================================="
echo "Deployment ID : $DEPLOYMENT_ID"
echo "Project       : $GCP_PROJECT"
echo "Region        : $REGION"
echo "=================================================="

# --------------------------------------------------
# AUTH
# --------------------------------------------------

gcloud config set project "$GCP_PROJECT" >/dev/null

if ! gcloud auth print-access-token >/dev/null 2>&1; then
    gcloud auth login
fi

gcloud services enable \
    run.googleapis.com \
    sqladmin.googleapis.com \
    artifactregistry.googleapis.com \
    --project="$GCP_PROJECT" \
    >/dev/null

gcloud auth configure-docker \
    "${REGION}-docker.pkg.dev" \
    --quiet >/dev/null


# ==================================================
# 1. DELETE EXISTING ROVE ENVIRONMENT
# ==================================================

echo
echo "=================================================="
echo "[1/10] REMOVE EXISTING ROVE ENVIRONMENT"
echo "=================================================="

if gcloud run services describe "$WEB_SERVICE" \
    --region="$REGION" \
    --project="$GCP_PROJECT" \
    >/dev/null 2>&1; then

    echo "Deleting existing Cloud Run service: $WEB_SERVICE"

    gcloud run services delete "$WEB_SERVICE" \
        --region="$REGION" \
        --project="$GCP_PROJECT" \
        --quiet
else
    echo "$WEB_SERVICE does not exist."
fi


if gcloud run services describe "$API_SERVICE" \
    --region="$REGION" \
    --project="$GCP_PROJECT" \
    >/dev/null 2>&1; then

    echo "Deleting existing Cloud Run service: $API_SERVICE"

    gcloud run services delete "$API_SERVICE" \
        --region="$REGION" \
        --project="$GCP_PROJECT" \
        --quiet
else
    echo "$API_SERVICE does not exist."
fi


if gcloud sql instances describe "$DB_INSTANCE" \
    --project="$GCP_PROJECT" \
    >/dev/null 2>&1; then

    echo "Deleting existing Cloud SQL instance: $DB_INSTANCE"

    # Disable deletion protection if enabled
    gcloud sql instances patch "$DB_INSTANCE" \
        --project="$GCP_PROJECT" \
        --no-deletion-protection \
        --quiet \
        >/dev/null 2>&1 || true

    gcloud sql instances delete "$DB_INSTANCE" \
        --project="$GCP_PROJECT" \
        --no-enable-final-backup \
        --quiet

else
    echo "$DB_INSTANCE does not exist."
fi


# ==================================================
# 2. CREATE POSTGRES
# ==================================================

echo
echo "=================================================="
echo "[2/10] CREATE POSTGRESQL"
echo "=================================================="

gcloud sql instances create "$DB_INSTANCE" \
    --project="$GCP_PROJECT" \
    --database-version=POSTGRES_16 \
    --region="$REGION" \
    --tier=db-f1-micro \
    --storage-size=10GB \
    --storage-type=SSD \
    --no-deletion-protection


# ==================================================
# 3. CREATE DATABASE + USER
# ==================================================

echo
echo "=================================================="
echo "[3/10] CREATE DATABASE + USER"
echo "=================================================="

gcloud sql databases create "$DB_NAME" \
    --instance="$DB_INSTANCE" \
    --project="$GCP_PROJECT"


gcloud sql users create "$DB_USER" \
    --instance="$DB_INSTANCE" \
    --password="$DB_PASSWORD" \
    --project="$GCP_PROJECT"


CONNECTION_NAME="$(gcloud sql instances describe "$DB_INSTANCE" \
    --project="$GCP_PROJECT" \
    --format='value(connectionName)')"

echo "Cloud SQL connection:"
echo "$CONNECTION_NAME"


# ==================================================
# 4. PREPARE API IMAGE
# ==================================================

echo
echo "=================================================="
echo "[4/10] PREPARE API"
echo "=================================================="

cat > server/Dockerfile.gcp <<'EOF'
FROM python:3.12-slim

WORKDIR /app

ENV PYTHONDONTWRITEBYTECODE=1
ENV PYTHONUNBUFFERED=1

COPY requirements.txt .

RUN pip install \
    --no-cache-dir \
    --trusted-host pypi.org \
    --trusted-host files.pythonhosted.org \
    --trusted-host pypi.python.org \
    -r requirements.txt

COPY app ./app
COPY scripts ./scripts
COPY alembic ./alembic
COPY alembic.ini ./alembic.ini

EXPOSE 8080

CMD ["sh", "-c", "exec uvicorn app.main:app --host 0.0.0.0 --port ${PORT:-8080}"]
EOF


API_VERSION_IMAGE="${REGION}-docker.pkg.dev/${GCP_PROJECT}/${REPOSITORY}/${API_IMAGE}:${DEPLOYMENT_ID}"

docker buildx build \
    --platform linux/amd64 \
    -f server/Dockerfile.gcp \
    -t "$API_VERSION_IMAGE" \
    --push \
    server


# ==================================================
# 5. DEPLOY API
# ==================================================

echo
echo "=================================================="
echo "[5/10] DEPLOY API"
echo "=================================================="

DATABASE_URL="postgresql+psycopg://${DB_USER}:${DB_PASSWORD}@/${DB_NAME}?host=/cloudsql/${CONNECTION_NAME}"

gcloud run deploy "$API_SERVICE" \
    --image="$API_VERSION_IMAGE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --platform=managed \
    --port=8080 \
    --add-cloudsql-instances="$CONNECTION_NAME" \
    --set-env-vars="DATABASE_URL=${DATABASE_URL},APP_ENV=production,DEPLOYMENT_ID=${DEPLOYMENT_ID}" \
    --no-invoker-iam-check \
    --quiet

# Cloud Run supports attaching Cloud SQL instances directly during deployment. :chatgpt-content-reference{index="1"}


API_URL="$(gcloud run services describe "$API_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --format='value(status.url)')"

echo
echo "API URL:"
echo "$API_URL"


# ==================================================
# 6. RUN ALEMBIC MIGRATIONS
# ==================================================

echo
echo "=================================================="
echo "[6/10] RUN DATABASE MIGRATIONS"
echo "=================================================="

# Use a Cloud Run Job-style one-off container invocation via local Docker
# with Cloud SQL Auth Proxy unavailable here, so use Cloud Run service command
# temporarily for migration.

gcloud run services update "$API_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --command="sh" \
    --args="-c,alembic upgrade head && exec uvicorn app.main:app --host 0.0.0.0 --port \$PORT" \
    --quiet

echo "Waiting for new API revision..."

API_URL="$(gcloud run services describe "$API_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --format='value(status.url)')"


# ==================================================
# 7. TEST API
# ==================================================

echo
echo "=================================================="
echo "[7/10] TEST API"
echo "=================================================="

curl -s -o /dev/null \
    -w "FastAPI docs: HTTP %{http_code}\n" \
    "$API_URL/docs" || true

curl -s \
    "$API_URL/api/health" || true

echo


# ==================================================
# 8. BUILD REACT
# ==================================================

echo
echo "=================================================="
echo "[8/10] BUILD REACT"
echo "=================================================="

rm -rf dist

NODE_USE_SYSTEM_CA=1 npm ci
NODE_USE_SYSTEM_CA=1 npm run build

if [ ! -d "dist" ]; then
    echo "ERROR: dist directory not created."
    exit 1
fi


if [ -f ".dockerignore" ]; then

python3 <<'PY'
from pathlib import Path

p = Path(".dockerignore")

lines = p.read_text().splitlines()

blocked = {
    "dist",
    "./dist",
    "**/dist",
    "*/dist"
}

clean = []

for line in lines:

    if line.strip().rstrip("/") in blocked:
        continue

    clean.append(line)

p.write_text("\n".join(clean) + "\n")
PY

fi


# ==================================================
# 9. BUILD WEB
# ==================================================

echo
echo "=================================================="
echo "[9/10] BUILD WEB"
echo "=================================================="

API_HOST="$(echo "$API_URL" | sed 's#https://##' | sed 's#/$##')"

cat > nginx.conf.gcp <<EOF
server {

    listen 8080;
    server_name _;

    root /usr/share/nginx/html;
    index index.html;


    location /api/ {

        proxy_ssl_server_name on;
        proxy_ssl_name ${API_HOST};

        proxy_set_header Host ${API_HOST};
        proxy_set_header X-Real-IP \$remote_addr;
        proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto https;

        proxy_pass ${API_URL}/api/;
    }


    location / {

        try_files \$uri \$uri/ /index.html;

    }
}
EOF


cat > Dockerfile.gcp <<'EOF'
FROM nginx:alpine

COPY nginx.conf.gcp /etc/nginx/conf.d/default.conf
COPY dist /usr/share/nginx/html

EXPOSE 8080

CMD ["nginx", "-g", "daemon off;"]
EOF


WEB_VERSION_IMAGE="${REGION}-docker.pkg.dev/${GCP_PROJECT}/${REPOSITORY}/${WEB_IMAGE}:${DEPLOYMENT_ID}"

docker buildx build \
    --platform linux/amd64 \
    -f Dockerfile.gcp \
    -t "$WEB_VERSION_IMAGE" \
    --push \
    .


# ==================================================
# 10. DEPLOY WEB
# ==================================================

echo
echo "=================================================="
echo "[10/10] DEPLOY WEB"
echo "=================================================="

gcloud run deploy "$WEB_SERVICE" \
    --image="$WEB_VERSION_IMAGE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --platform=managed \
    --port=8080 \
    --set-env-vars="DEPLOYMENT_ID=${DEPLOYMENT_ID}" \
    --no-invoker-iam-check \
    --quiet


WEB_URL="$(gcloud run services describe "$WEB_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --format='value(status.url)')"


echo
echo "=================================================="
echo " ROVE DEPLOYMENT SUCCESSFUL"
echo "=================================================="

echo "Deployment ID : $DEPLOYMENT_ID"
echo
echo "Database:"
echo "$DB_INSTANCE"

echo
echo "API:"
echo "$API_URL"

echo
echo "Website:"
echo "$WEB_URL"

echo
echo "API Image:"
echo "$API_VERSION_IMAGE"

echo
echo "Web Image:"
echo "$WEB_VERSION_IMAGE"

echo
echo "=================================================="

open "$WEB_URL" 2>/dev/null || true

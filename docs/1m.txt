AOS_TOKEN_URL='https://<configured-token-host>/<token-path>'

docker compose --env-file .env.docker exec -T api \
  python - "$AOS_TOKEN_URL" <<'PY'
import sys
import httpx

try:
    response = httpx.get(sys.argv[1], timeout=10, follow_redirects=False)
    print(f"Token endpoint responded with HTTP {response.status_code}")
except httpx.RequestError as exc:
    print(f"{type(exc).__name__}: {exc}")
    sys.exit(1)
PY

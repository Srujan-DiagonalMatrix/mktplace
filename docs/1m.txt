#!/usr/bin/env bash
set -Eeuo pipefail
set +x
umask 077

export GCP_PROJECT="${GCP_PROJECT:-tactical-innovation-f2b97b}"
export REGION="${REGION:-europe-west1}"

BRANCH="${BRANCH:-main}"
REMOTE="${REMOTE:-origin}"

export SQL_INSTANCE=rove-v1-postgres
export DB_NAME=rove_v1
export DB_USER=rove_v1
export SQL_CONNECTION="$GCP_PROJECT:$REGION:$SQL_INSTANCE"

REPOSITORY=rove-v1
API_SERVICE=rove-v1-api
WEB_SERVICE=rove-v1

export DB_SECRET=rove-v1-database-url
export ADMIN_USER_SECRET=rove-v1-admin-username
export ADMIN_PASS_SECRET=rove-v1-admin-password
export DELIVERY_SECRET=rove-v1-credential-delivery-key

export EXTRA_SECRETS="${EXTRA_SECRETS:-}"
REMOVE_OLD_SERVICES="${REMOVE_OLD_SERVICES:-1}"

# Optional:
# API_ENV_FILE=/absolute/path/non-secret-settings.json
# EXTRA_SECRETS='SMTP_PASSWORD=existing-secret:latest'
# CORPORATE_CA_FILE=/absolute/path/company-ca.pem
# REPO_DIR=/absolute/path/rove
# CREDENTIALS_DIR=/private/directory
#
# New environment:
#   Cloud Run: rove-v1, rove-v1-api
#   Cloud SQL: rove-v1-postgres
#   Database/user: rove_v1
#   Artifact Registry: rove-v1
#   Service accounts/secrets: rove-v1-*
#
# Old rove-postgres is never modified or deleted.

fail() {
  echo "ERROR: $*" >&2
  exit 1
}

gc() {
  gcloud --project="$GCP_PROJECT" "$@"
}

for tool in git node npm python3 docker gcloud curl tar; do
  command -v "$tool" >/dev/null || fail "Missing tool: $tool"
done

docker info >/dev/null
docker buildx version >/dev/null

script_dir=$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)
root=$(git -C "${REPO_DIR:-$script_dir}" rev-parse --show-toplevel)

work=$(mktemp -d)
chmod 700 "$work"

trap 'rm -rf "$work"' EXIT
trap 'echo "Stopped at line $LINENO. Created resources may remain. Existing databases were not deleted or reset." >&2' ERR

export CREDENTIALS_DIR="${CREDENTIALS_DIR:-$(mktemp -d "${TMPDIR:-/tmp}/rove-v1-credentials.XXXXXXXX")}"

mkdir -p "$CREDENTIALS_DIR"
chmod 700 "$CREDENTIALS_DIR"

echo "=== 1. Build fresh frontend from GitHub ==="

git -C "$root" check-ref-format --branch "$BRANCH" >/dev/null
git -C "$root" fetch "$REMOTE" "refs/heads/$BRANCH"

commit=$(git -C "$root" rev-parse FETCH_HEAD)

mkdir "$work/source"

git -C "$root" archive "$commit" |
  tar -x -C "$work/source"

cd "$work/source"

echo "Source: $BRANCH at $commit"

NODE_USE_SYSTEM_CA=1 npm ci --include=dev

mkdir -p "$work/web/dist"

NODE_USE_SYSTEM_CA=1 VITE_API_URL='' npm run build \
  -- --outDir "$work/web/dist" --emptyOutDir

test -s "$work/web/dist/index.html"

echo "=== 2. Create deployment infrastructure ==="

gc services enable \
  run.googleapis.com \
  sqladmin.googleapis.com \
  artifactregistry.googleapis.com \
  secretmanager.googleapis.com \
  iam.googleapis.com \
  --quiet

if ! gc artifacts repositories describe "$REPOSITORY" \
  --location="$REGION" >/dev/null 2>&1; then
  gc artifacts repositories create "$REPOSITORY" \
    --location="$REGION" \
    --repository-format=docker \
    --quiet
fi

[[ "$(gc artifacts repositories describe "$REPOSITORY" \
  --location="$REGION" --format='value(format)')" == DOCKER ]] ||
  fail "Expected a Docker repository."

for account in rove-v1-api rove-v1-web; do
  if ! gc iam service-accounts describe \
    "$account@$GCP_PROJECT.iam.gserviceaccount.com" >/dev/null 2>&1; then
    gc iam service-accounts create "$account" \
      --display-name="$account" \
      --quiet
  fi
done

api_account="rove-v1-api@$GCP_PROJECT.iam.gserviceaccount.com"
web_account="rove-v1-web@$GCP_PROJECT.iam.gserviceaccount.com"

gc projects add-iam-policy-binding "$GCP_PROJECT" \
  --member="serviceAccount:$api_account" \
  --role=roles/cloudsql.client \
  --condition=None \
  --quiet >/dev/null

# Provision a separate instance. Never target old rove-postgres.
if ! gc sql instances describe "$SQL_INSTANCE" >/dev/null 2>&1; then
  echo "Creating NEW rove-v1-postgres."
  echo "Provisioning may take 10–20 minutes or longer."

  gc sql instances create "$SQL_INSTANCE" \
    --database-version=POSTGRES_16 \
    --edition=enterprise \
    --tier=db-f1-micro \
    --region="$REGION" \
    --availability-type=zonal \
    --assign-ip \
    --storage-type=SSD \
    --storage-size=10 \
    --storage-auto-increase \
    --backup-start-time=02:00 \
    --enable-point-in-time-recovery \
    --deletion-protection \
    --async \
    --quiet
else
  echo "Reusing rove-v1-postgres without deleting or resetting its data."
fi

echo "=== 3. Build fresh API while SQL provisions ==="

gc auth configure-docker "$REGION-docker.pkg.dev" --quiet

registry="$REGION-docker.pkg.dev/$GCP_PROJECT/$REPOSITORY"
tag="v1-$(date -u +%Y%m%d%H%M%S)-${commit:0:12}"
api_image="$registry/rove-v1-api:$tag"

if [[ -n "${CORPORATE_CA_FILE:-}" ]]; then
  test -s "$CORPORATE_CA_FILE"

  docker buildx build \
    --platform linux/amd64 \
    --pull \
    --load \
    --secret "id=corporate_ca,src=$CORPORATE_CA_FILE" \
    -t "$api_image" \
    -f server/Dockerfile server
else
  docker buildx build \
    --platform linux/amd64 \
    --pull \
    --load \
    -t "$api_image" \
    -f server/Dockerfile server
fi

docker run --rm --entrypoint python "$api_image" -c \
  'from alembic.config import Config; from alembic.script import ScriptDirectory; s=ScriptDirectory.from_config(Config("alembic.ini")); assert len(s.get_heads())==1, "Resolve migration heads before deploying"'

docker push "$api_image"

echo "=== 4. Wait for new SQL instance ==="

python3 - <<'PY'
import json
import os
import subprocess
import time

deadline = time.monotonic() + 1800

while True:
    result = subprocess.run(
        [
            "gcloud",
            "--project=" + os.environ["GCP_PROJECT"],
            "sql", "instances", "describe",
            os.environ["SQL_INSTANCE"],
            "--format=json",
        ],
        capture_output=True,
        text=True,
    )

    if result.returncode:
        raise SystemExit(
            "Cannot inspect the new SQL instance. "
            "Check permissions and Cloud SQL operations."
        )

    instance = json.loads(result.stdout)

    if instance.get("state") == "RUNNABLE":
        if (
            instance.get("connectionName") != os.environ["SQL_CONNECTION"]
            or not instance.get("databaseVersion", "").startswith("POSTGRES")
        ):
            raise SystemExit(
                "Existing rove-v1 instance has an unexpected region/version."
            )

        if not any(
            address.get("type") == "PRIMARY"
            for address in instance.get("ipAddresses", [])
        ):
            raise SystemExit(
                "Existing rove-v1 instance needs private-network configuration."
            )
        break

    if instance.get("state") in ("FAILED", "SUSPENDED"):
        raise SystemExit(
            "SQL instance state: " + instance["state"]
            + ". Check Cloud SQL operations."
        )

    if time.monotonic() > deadline:
        raise SystemExit(
            "SQL provisioning is still pending after 30 minutes. "
            "Re-run when the instance is ready."
        )

    print("Waiting for rove-v1-postgres...", flush=True)
    time.sleep(15)
PY

gc sql databases list \
  --instance="$SQL_INSTANCE" \
  --format=json > "$work/databases.json"

if ! python3 - "$work/databases.json" "$DB_NAME" <<'PY'
import json
import sys

exists = any(
    database["name"] == sys.argv[2]
    for database in json.load(open(sys.argv[1]))
)

sys.exit(0 if exists else 1)
PY
then
  gc sql databases create "$DB_NAME" \
    --instance="$SQL_INSTANCE" \
    --quiet
fi

echo "=== 5. Generate credentials and create new database user ==="

python3 - "$work" "${API_ENV_FILE:-}" <<'PY'
import json
import os
import re
import secrets
import subprocess
import sys
import time

from pathlib import Path
from urllib.parse import quote, urlencode, urlsplit, parse_qs, unquote
from urllib.request import Request, urlopen
from urllib.error import HTTPError, URLError

project = os.environ["GCP_PROJECT"]
instance = os.environ["SQL_INSTANCE"]
connection = os.environ["SQL_CONNECTION"]

# Hard boundary: never change a user on the old instance.
if instance != "rove-v1-postgres":
    raise SystemExit("Refusing an unexpected SQL instance.")

work = Path(sys.argv[1])
saved = Path(os.environ["CREDENTIALS_DIR"])

def stop(message):
    raise SystemExit(message)

def cli(*arguments):
    result = subprocess.run(
        ["gcloud", "--project=" + project, *arguments],
        capture_output=True,
        text=True,
    )

    if result.returncode:
        stop(
            "Google Cloud command failed: "
            + " ".join(arguments[:3])
            + ". Check permissions. No secret values were printed."
        )

    return result.stdout

def cloud_json(*arguments):
    return json.loads(cli(*arguments, "--format=json"))

existing = {
    str(secret["name"]).split("/")[-1]
    for secret in cloud_json("secrets", "list")
}

def get_or_create_secret(name, generate):
    if not name.startswith("rove-v1-"):
        stop("Refusing a secret outside the rove-v1 namespace.")

    if name in existing:
        # Capture in memory only; never print payloads.
        return cli(
            "secrets", "versions", "access", "latest",
            "--secret=" + name,
        )

    value = generate()
    path = saved / (name + ".txt")

    descriptor = os.open(
        path,
        os.O_WRONLY | os.O_CREAT | os.O_EXCL,
        0o600,
    )

    with os.fdopen(descriptor, "w") as file:
        file.write(value)

    cli(
        "secrets", "create", name,
        "--replication-policy=automatic",
        "--data-file=" + str(path),
        "--quiet",
    )

    existing.add(name)
    return value

references = {
    "DATABASE_URL": os.environ["DB_SECRET"] + ":latest",
    "ADMIN_USERNAME": os.environ["ADMIN_USER_SECRET"] + ":latest",
    "ADMIN_PASSWORD": os.environ["ADMIN_PASS_SECRET"] + ":latest",
    "CREDENTIAL_DELIVERY_KEY": os.environ["DELIVERY_SECRET"] + ":latest",
}

for item in filter(None, os.environ["EXTRA_SECRETS"].split(",")):
    key, separator, value = item.partition("=")

    if not separator or key in references:
        stop("Invalid or duplicate EXTRA_SECRETS entry.")

    references[key] = value

for key, value in references.items():
    if (
        not re.fullmatch(r"[A-Z][A-Z0-9_]*", key)
        or not re.fullmatch(r"[A-Za-z0-9_-]+:(latest|[0-9]+)", value)
    ):
        stop("Invalid Secret Manager reference.")

settings = json.load(open(sys.argv[2])) if sys.argv[2] else {}

if (
    not isinstance(settings, dict)
    or any(not isinstance(value, str) for value in settings.values())
    or set(settings).intersection(references)
):
    stop("API_ENV_FILE must contain JSON string settings without secret keys.")

settings.update(
    APP_ENV="production",
    AUTH_COOKIE_SECURE="true",
    SESSION_COOKIE_SECURE="true",
)

(work / "env.json").write_text(json.dumps(settings))

def generate_database_url():
    return (
        "postgresql+psycopg://"
        + quote(os.environ["DB_USER"], safe="")
        + ":"
        + secrets.token_urlsafe(36)
        + "@/"
        + quote(os.environ["DB_NAME"], safe="")
        + "?host=/cloudsql/"
        + connection
    )

database_url = get_or_create_secret(
    os.environ["DB_SECRET"],
    generate_database_url,
)

parsed = urlsplit(database_url)
username = unquote(parsed.username or "")
password = unquote(parsed.password or "")

if (
    username != os.environ["DB_USER"]
    or unquote(parsed.path.lstrip("/")) != os.environ["DB_NAME"]
    or not password
    or parsed.hostname
    or parse_qs(parsed.query).get("host") != ["/cloudsql/" + connection]
):
    stop(
        "Existing rove-v1 database secret does not match the new environment. "
        "Stopped without altering the SQL user."
    )

admin_username = get_or_create_secret(
    os.environ["ADMIN_USER_SECRET"],
    lambda: "rove-v1-admin",
)

admin_password = get_or_create_secret(
    os.environ["ADMIN_PASS_SECRET"],
    lambda: secrets.token_urlsafe(32),
)

get_or_create_secret(
    os.environ["DELIVERY_SECRET"],
    lambda: secrets.token_urlsafe(32),
)

descriptor = os.open(
    saved / "rove-v1-credentials.json",
    os.O_WRONLY | os.O_CREAT | os.O_EXCL,
    0o600,
)

with os.fdopen(descriptor, "w") as file:
    json.dump(
        {
            "database": os.environ["DB_NAME"],
            "database_user": username,
            "database_password": password,
            "ADMIN_USERNAME": admin_username,
            "ADMIN_PASSWORD": admin_password,
        },
        file,
    )

users = cloud_json(
    "sql", "users", "list",
    "--instance=" + instance,
)

user_exists = any(
    user.get("name") == username
    and user.get("type", "BUILT_IN") == "BUILT_IN"
    for user in users
)

token = cli("auth", "print-access-token").strip()

base = (
    "https://sqladmin.googleapis.com/sql/v1beta4/projects/"
    + quote(project, safe="")
)

def sql_api(method, path, data=None):
    request = Request(
        base + path,
        data=json.dumps(data).encode() if data is not None else None,
        method=method,
        headers={
            "Authorization": "Bearer " + token,
            "Content-Type": "application/json",
        },
    )

    try:
        with urlopen(request, timeout=60) as response:
            return json.load(response)
    except HTTPError as error:
        stop(
            "Cloud SQL API HTTP "
            + str(error.code)
            + ". Credentials remain in the protected local directory."
        )
    except URLError:
        stop(
            "Cloud SQL connection failed. "
            "Credentials remain in the protected local directory."
        )

path = "/instances/" + quote(instance, safe="") + "/users"

if user_exists:
    path += "?" + urlencode({"name": username})

operation = sql_api(
    "PUT" if user_exists else "POST",
    path,
    {
        "name": username,
        "password": password,
        "type": "BUILT_IN",
    },
)

deadline = time.monotonic() + 300

while operation.get("status") != "DONE":
    if time.monotonic() > deadline:
        stop(
            "New SQL user operation is still pending. "
            "Check Cloud SQL operations before retrying."
        )

    time.sleep(2)

    operation = sql_api(
        "GET",
        "/operations/" + quote(operation["name"], safe=""),
    )

if operation.get("error"):
    stop("New SQL user operation failed. Check Cloud SQL operations.")

(work / "secret-args").write_text(
    ",".join(key + "=" + value for key, value in references.items())
)

(work / "secret-refs").write_text(
    "".join(value + "\n" for value in sorted(set(references.values())))
)

print("Credentials saved privately in: " + str(saved))
PY

while IFS=: read -r secret version; do
  [[ "$(gc secrets versions describe "$version" \
    --secret="$secret" --format='value(state)')" == ENABLED ]] ||
    fail "Secret is disabled: $secret:$version"

  gc secrets add-iam-policy-binding "$secret" \
    --member="serviceAccount:$api_account" \
    --role=roles/secretmanager.secretAccessor \
    --condition=None \
    --quiet >/dev/null
done < "$work/secret-refs"

echo "=== 6. Deploy new API and apply normal migrations ==="

gc run deploy "$API_SERVICE" \
  --region="$REGION" \
  --image="$api_image" \
  --service-account="$api_account" \
  --port=8000 \
  --ingress=all \
  --allow-unauthenticated \
  --add-cloudsql-instances="$SQL_CONNECTION" \
  --env-vars-file="$work/env.json" \
  --update-secrets="$(cat "$work/secret-args")" \
  --max-instances=1 \
  --command=sh \
  --args='-c,alembic upgrade head && exec uvicorn app.main:app --host 0.0.0.0 --port 8000' \
  --quiet

gc run services update-traffic "$API_SERVICE" \
  --region="$REGION" --to-latest --quiet

api_url=$(gc run services describe "$API_SERVICE" \
  --region="$REGION" --format='value(status.url)')

curl --fail --silent --show-error \
  --retry 5 --retry-all-errors --max-time 60 \
  "$api_url/api/health" >/dev/null

echo "=== 7. Build and deploy new frontend ==="

cat > "$work/web/nginx.conf" <<EOF
server {
  listen 8080;
  server_name _;

  root /usr/share/nginx/html;
  index index.html;

  location /api/ {
    proxy_pass ${api_url};
    proxy_ssl_server_name on;
    proxy_set_header Host ${api_url#https://};
    proxy_set_header X-Forwarded-Proto https;
    proxy_http_version 1.1;
    proxy_read_timeout 120s;
  }

  location = /index.html {
    add_header Cache-Control "no-cache, no-store, must-revalidate" always;
  }

  location = /version.json {
    add_header Cache-Control "no-store" always;
  }

  location /assets/ {
    try_files \$uri =404;
    add_header Cache-Control "public, max-age=31536000, immutable";
  }

  location / {
    add_header Cache-Control "no-cache, no-store, must-revalidate" always;
    try_files \$uri \$uri/ /index.html;
  }
}
EOF

python3 - "$work/web/dist/version.json" "$commit" <<'PY'
import json
import sys

json.dump(
    {"commit": sys.argv[2], "environment": "rove-v1"},
    open(sys.argv[1], "w"),
)
PY

cat > "$work/web/Dockerfile" <<'EOF'
FROM nginx:alpine
COPY nginx.conf /etc/nginx/conf.d/default.conf
COPY dist /usr/share/nginx/html
EXPOSE 8080
CMD ["nginx", "-g", "daemon off;"]
EOF

web_image="$registry/rove-v1:$tag"

docker buildx build \
  --platform linux/amd64 \
  --pull --no-cache --push \
  -t "$web_image" "$work/web"

gc run deploy "$WEB_SERVICE" \
  --region="$REGION" \
  --image="$web_image" \
  --service-account="$web_account" \
  --port=8080 \
  --ingress=all \
  --allow-unauthenticated \
  --quiet

web_url=$(gc run services describe "$WEB_SERVICE" \
  --region="$REGION" --format='value(status.url)')

gc run services update "$API_SERVICE" \
  --region="$REGION" \
  --update-env-vars="CORS_ORIGIN=$web_url" \
  --quiet

gc run services update-traffic "$API_SERVICE" \
  --region="$REGION" --to-latest --quiet

gc run services update-traffic "$WEB_SERVICE" \
  --region="$REGION" --to-latest --quiet

echo "=== 8. Verify new environment ==="

curl --fail --silent --show-error \
  --retry 5 --retry-all-errors --max-time 60 \
  "$web_url/api/health" >/dev/null

curl --fail --silent --show-error --max-time 60 \
  "$web_url/version.json?deploy=$tag" > "$work/version.json"

python3 - "$work/version.json" "$commit" <<'PY'
import json
import sys

version = json.load(open(sys.argv[1]))

if (
    version.get("commit") != sys.argv[2]
    or version.get("environment") != "rove-v1"
):
    raise SystemExit("Live frontend does not match the new build.")

print("PASS: live UI matches the fresh build.")
PY

curl --fail --silent --show-error --max-time 60 \
  "$web_url/api/catalogue/vehicles" > "$work/catalogue.json"

python3 - "$work/catalogue.json" <<'PY'
import json
import sys

vehicles = json.load(open(sys.argv[1]))

if not isinstance(vehicles, list):
    raise SystemExit("Unexpected catalogue response.")

print("Catalogue vehicle count:", len(vehicles))

if not vehicles:
    raise SystemExit(
        "New catalogue is empty. Import inventory before removing old services. "
        "No old services were deleted."
    )
PY

# Delete only these exact OLD Cloud Run services, after successful checks.
# Never delete SQL, secrets, repositories, volumes or new services.
if [[ "$REMOVE_OLD_SERVICES" == 1 ]]; then
  echo "=== 9. Remove old Cloud Run services only ==="

  gc run services list --region="$REGION" \
    --format=json > "$work/services.json"

  python3 - "$work/services.json" > "$work/old-services" <<'PY'
import json
import sys

allowed = {"rove", "rove-api", "rove-web"}

for service in json.load(open(sys.argv[1])):
    name = service.get("metadata", {}).get("name", "").split("/")[-1]

    if name in allowed:
        print(name)
PY

  while IFS= read -r old; do
    [[ "$old" == rove || "$old" == rove-api || "$old" == rove-web ]] ||
      fail "Unexpected cleanup target."

    gc run services delete "$old" \
      --region="$REGION" \
      --quiet
  done < "$work/old-services"
fi

echo
echo "NEW DEMO URL: $web_url/?deploy=$tag"
echo "API: $api_url"
echo "New SQL instance: $SQL_CONNECTION"
echo "New database/user: $DB_NAME / $DB_USER"
echo "Frontend source: $BRANCH at $commit"
echo "Protected credentials: $CREDENTIALS_DIR/rove-v1-credentials.json"
echo
echo "Old rove-postgres database retained."
echo "SMTP/AOS integrations require your provider configuration."
echo "Open the demo URL in a private browser window."

cd ~/Documents/repo/rove || exit 1
bash -n deploy/new-deploy.sh
BRANCH=main bash deploy/new-deploy.sh

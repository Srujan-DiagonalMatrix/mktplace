#!/usr/bin/env bash
# Shared implementation: merge both histories, never replace a destination snapshot.
set -euo pipefail
DIRECTION="${1:?Expected pull or push}"; shift
PERSONAL_URL="${PERSONAL_URL:-git@github.com:Srujan-DiagonalMatrix/rove.git}"
COMPANY_URL="${COMPANY_URL:-git@github-company:lbg-cloud-platform/tactical-innovation-f2b97b.git}"
REPO_ROOT=$(git rev-parse --show-toplevel)
case "$DIRECTION" in
  pull)
    SOURCE_BRANCH="${1:-}"
    if [ -z "$SOURCE_BRANCH" ]; then read -r -p 'LBG branch: ' SOURCE_BRANCH; fi
    TARGET_BRANCH="${2:-feature/lbg-import-$(date +%Y%m%d-%H%M%S)}"
    SOURCE_URL="$COMPANY_URL"; TARGET_URL="$PERSONAL_URL"; BASE_BRANCH=main
    ;;
  push)
    SOURCE_BRANCH=main
    TARGET_BRANCH="${1:-feature/rove-sync-$(date +%Y%m%d-%H%M%S)}"
    SOURCE_URL="$PERSONAL_URL"; TARGET_URL="$COMPANY_URL"; BASE_BRANCH=main_stage
    ;;
  *) echo 'Expected pull or push.' >&2; exit 2 ;;
esac
git check-ref-format --branch "$SOURCE_BRANCH" >/dev/null
git check-ref-format --branch "$TARGET_BRANCH" >/dev/null
case "$TARGET_BRANCH" in feature/*) ;; *) echo 'Destination must begin with feature/.' >&2; exit 2 ;; esac
cd "$REPO_ROOT"
# Fetch URLs directly; leave the user's configured remotes and working tree alone.
git fetch --no-tags "$TARGET_URL" "refs/heads/$BASE_BRANCH"
TARGET_COMMIT=$(git rev-parse 'FETCH_HEAD^{commit}')
git fetch --no-tags "$SOURCE_URL" "refs/heads/$SOURCE_BRANCH"
SOURCE_COMMIT=$(git rev-parse 'FETCH_HEAD^{commit}')
if git show-ref --verify --quiet "refs/heads/$TARGET_BRANCH" ||
   [ -n "$(git ls-remote --heads "$TARGET_URL" "refs/heads/$TARGET_BRANCH")" ]; then
  echo "Branch already exists: $TARGET_BRANCH" >&2; exit 1
fi
WORKTREE_PARENT=$(mktemp -d "${TMPDIR:-/tmp}/rove-sync.XXXXXXXX")
WORKTREE="$WORKTREE_PARENT/worktree"
KEEP_WORKTREE=1
cleanup() {
  if [ "$KEEP_WORKTREE" -eq 0 ]; then
    git -C "$REPO_ROOT" worktree remove "$WORKTREE" >/dev/null 2>&1 || true
    rmdir "$WORKTREE_PARENT" 2>/dev/null || true
  else
    printf '\nSync worktree retained: %s\nBranch: %s\n' "$WORKTREE" "$TARGET_BRANCH" >&2
  fi
}
trap cleanup EXIT
git worktree add -b "$TARGET_BRANCH" "$WORKTREE" "$TARGET_COMMIT"
cd "$WORKTREE"
# Legacy snapshot imports did not record the other repository as a parent.
# On the first true merge, unrelated histories may need manual reconciliation.
# Never use -X ours/theirs or discard the other side's application changes.
merge_status=0
git merge --no-ff --no-commit --allow-unrelated-histories "$SOURCE_COMMIT" || merge_status=$?

# Keep destination-owned automation and the original local-only exclusions.
# Resolve only these paths by policy, never application code.
owned=(.github Dockerfile.backup Dockerfile.backup.bak Dockerfile.bak
  Dockerfile.before-ca-fix Dockerfile.web.bak 'Dockerfile.web.backup.*.bak'
  'server/Dockerfile.backup.*.bak' server/Dockerfile.bak
  server/Dockerfile.bank-backup company-ca.pem docker-compose.corporate.yml)
for path in "${owned[@]}"; do
  # Literal/glob pathspecs are intentionally interpreted by Git, not the shell.
  if [ -n "$(git ls-files -- "$path")" ] || [ -n "$(git ls-tree -r --name-only "$TARGET_COMMIT" -- "$path")" ]; then
    git restore --source="$TARGET_COMMIT" --staged --worktree -- "$path"
  fi
done
if [ -n "$(git diff --name-only --diff-filter=U)" ]; then
  echo 'Conflicting changes require review. Nothing has been pushed.' >&2
  git diff --name-only --diff-filter=U >&2
  printf 'Resolve the files in %s, then run:\n' "$WORKTREE" >&2
  printf '  git -C %q add <resolved-files>\n' "$WORKTREE" >&2
  printf '  git -C %q commit\n' "$WORKTREE" >&2
  if [ "$DIRECTION" = push ]; then
    printf 'Before pushing, export a clean LBG-only commit (after committing the resolution):\n' >&2
    printf '  cd %q\n' "$WORKTREE" >&2
    printf '  git diff --exit-code %q HEAD -- .github &&\n' "$TARGET_COMMIT" >&2
    printf '  git update-ref %q HEAD &&\n' "refs/rove-sync-local/$TARGET_BRANCH" >&2
    printf '  CLEAN_COMMIT=$(git commit-tree "HEAD^{tree}" -p %q -m "Sync reconciled ROVE application") &&\n' "$TARGET_COMMIT" >&2
    printf '  git reset --soft "$CLEAN_COMMIT" &&\n' >&2
  fi
  printf '  git -C %q push --force-with-lease=%q %q HEAD:refs/heads/%q\n' \
    "$WORKTREE" "refs/heads/$TARGET_BRANCH:" "$TARGET_URL" "$TARGET_BRANCH" >&2
  printf 'Create PR: %s -> %s\n' "$TARGET_BRANCH" "$BASE_BRANCH" >&2
  exit 1
fi
# A failed merge without MERGE_HEAD is an operational error, not a successful sync.
if [ "$merge_status" -ne 0 ] && ! git rev-parse --verify -q MERGE_HEAD >/dev/null; then
  echo 'Merge failed; nothing has been pushed.' >&2; exit 1
fi
if git rev-parse --verify -q MERGE_HEAD >/dev/null; then
  git commit -m "Merge $DIRECTION $SOURCE_BRANCH ($SOURCE_COMMIT); preserve both histories"
fi
# LBG accepts code changes but restricts imported commits touching .github.
# Preserve the reconciled tree while giving the exported commit only LBG ancestry.
# The local merge remains reachable for inspection; it is never pushed to LBG.
if [ "$DIRECTION" = push ]; then

  MERGED_COMMIT=$(git rev-parse HEAD)
  EXPORT_COMMIT=$(git commit-tree "HEAD^{tree}" -p "$TARGET_COMMIT" \
    -m "Sync reconciled ROVE application from personal $SOURCE_BRANCH ($SOURCE_COMMIT)")
  git update-ref "refs/rove-sync-local/$TARGET_BRANCH" "$MERGED_COMMIT"
  git reset --soft "$EXPORT_COMMIT"
fi

git push --force-with-lease="refs/heads/$TARGET_BRANCH:" "$TARGET_URL" "HEAD:refs/heads/$TARGET_BRANCH"
KEEP_WORKTREE=0
printf '\nCreated branch: %s\nCreate PR: %s -> %s\n' "$TARGET_BRANCH" "$TARGET_BRANCH" "$BASE_BRANCH"

#!/usr/bin/env bash
set -Eeuo pipefail
set +x

GCP_PROJECT="${GCP_PROJECT:-tactical-innovation-f2b97b}"
REGION="${REGION:-europe-west1}"
REPOSITORY="${REPOSITORY:-cloud-run-source-deploy}"

WEB_SERVICE="${WEB_SERVICE:-rove}"
API_SERVICE="${API_SERVICE:-rove-api}"

BRANCH="${BRANCH:-fix/frontend-build-source-cleanup}"
REMOTE="${REMOTE:-origin}"

CLOUD_SQL_CONNECTION="${CLOUD_SQL_CONNECTION:-tactical-innovation-f2b97b:europe-west1:rove-postgres}"

DATABASE_URL_SECRET="${DATABASE_URL_SECRET:-rove-database-url}"
ADMIN_USERNAME_SECRET="${ADMIN_USERNAME_SECRET:-rove-admin-username}"
ADMIN_PASSWORD_SECRET="${ADMIN_PASSWORD_SECRET:-rove-admin-password}"

# Accept names left over from the previous script, including :latest.
DATABASE_URL_SECRET="${DATABASE_URL_SECRET%%:*}"
ADMIN_USERNAME_SECRET="${ADMIN_USERNAME_SECRET%%:*}"
ADMIN_PASSWORD_SECRET="${ADMIN_PASSWORD_SECRET%%:*}"

# Optional configuration:
#
# FORCE_API_BUILD=1
#   Also rebuild the API. Frontend is always rebuilt.
#
# API_ENV_FILE=/absolute/path/api-env.json
#   Additional non-secret API environment settings.
#
# EXTRA_SECRETS='SMTP_PASSWORD=secret-name:latest'
#   Additional existing Secret Manager references.
#
# CORPORATE_CA_FILE=/absolute/path/company-ca.pem
#   Optional corporate certificate for an API Docker build.
#
# REPO_DIR=/absolute/path/rove
#   Repository location if running this script elsewhere.

fail() {
  echo "ERROR: $*" >&2
  exit 1
}

gc() {
  gcloud --project="$GCP_PROJECT" "$@"
}

for tool in git npm node python3 docker gcloud curl tar; do
  command -v "$tool" >/dev/null || fail "Missing tool: $tool"
done

docker info >/dev/null
docker buildx version >/dev/null

root=$(git -C "${REPO_DIR:-$PWD}" rev-parse --show-toplevel)

work=$(mktemp -d)
chmod 700 "$work"

trap 'rm -rf "$work"' EXIT
trap 'echo "Stopped at line $LINENO. Earlier cloud changes may remain. No database reset or automatic rollback was attempted." >&2' ERR

echo "=== Verify the existing database ==="

gc sql instances describe "${CLOUD_SQL_CONNECTION##*:}" \
  --format=json > "$work/sql.json"

python3 - "$work/sql.json" "$CLOUD_SQL_CONNECTION" "$GCP_PROJECT" <<'PY'
import json
import sys

instance = json.load(open(sys.argv[1]))

if (
    instance.get("connectionName") != sys.argv[2]
    or sys.argv[2].split(":")[0] != sys.argv[3]
    or instance.get("state") != "RUNNABLE"
    or not instance.get("databaseVersion", "").startswith("POSTGRES")
):
    raise SystemExit(
        "Expected the existing runnable PostgreSQL instance in GCP_PROJECT."
    )

if not any(
    address.get("type") == "PRIMARY"
    for address in instance.get("ipAddresses", [])
):
    raise SystemExit(
        "Private-IP-only SQL needs the original VPC configuration."
    )
PY

echo "=== Fetch latest selected branch ==="
echo "Local working files, branch and index will be preserved."

git -C "$root" check-ref-format --branch "$BRANCH" >/dev/null
git -C "$root" fetch "$REMOTE" "refs/heads/$BRANCH"

commit=$(git -C "$root" rev-parse FETCH_HEAD)

mkdir "$work/source"

git -C "$root" archive "$commit" |
  tar -x -C "$work/source"

cd "$work/source"

[[ -f package-lock.json && -f server/Dockerfile ]] ||
  fail "Expected ROVE source files are missing."

echo "=== Force fresh frontend build ==="
echo "Branch: $BRANCH"
echo "Commit: $commit"

NODE_USE_SYSTEM_CA=1 npm ci --include=dev

mkdir -p "$work/web/dist"

NODE_USE_SYSTEM_CA=1 VITE_API_URL='' npm run build \
  -- --outDir "$work/web/dist" --emptyOutDir

test -s "$work/web/dist/index.html"

echo "=== Ensure deployment resources exist ==="

gc services enable \
  run.googleapis.com \
  artifactregistry.googleapis.com \
  secretmanager.googleapis.com \
  sqladmin.googleapis.com \
  iam.googleapis.com \
  --quiet

if ! gc artifacts repositories describe "$REPOSITORY" \
  --location="$REGION" >/dev/null 2>&1; then
  gc artifacts repositories create "$REPOSITORY" \
    --location="$REGION" \
    --repository-format=docker \
    --quiet
fi

[[ "$(gc artifacts repositories describe "$REPOSITORY" \
  --location="$REGION" --format='value(format)')" == DOCKER ]] ||
  fail "Artifact repository must use Docker format."

for name in rove-api-runtime rove-web-runtime; do
  if ! gc iam service-accounts describe \
    "$name@$GCP_PROJECT.iam.gserviceaccount.com" >/dev/null 2>&1; then
    gc iam service-accounts create "$name" \
      --display-name="$name" \
      --quiet
  fi
done

api_account="rove-api-runtime@$GCP_PROJECT.iam.gserviceaccount.com"
web_account="rove-web-runtime@$GCP_PROJECT.iam.gserviceaccount.com"

gc projects add-iam-policy-binding "$GCP_PROJECT" \
  --member="serviceAccount:$api_account" \
  --role=roles/cloudsql.client \
  --condition=None \
  --quiet >/dev/null

# Existing secrets are preserved.
# Missing credentials are entered privately, without shell history.
# Local .env files are never sourced or uploaded.
ensure_secret() (
  set +x

  name="$1"
  kind="$2"

  if gc secrets describe "$name" >/dev/null 2>&1; then
    echo "Using existing secret: $name"
    return
  fi

  [[ -r /dev/tty ]] ||
    fail "Run interactively to create missing secret $name"

  file=$(mktemp)
  chmod 600 "$file"
  trap 'rm -f "$file"' EXIT

  if [[ "$kind" == database ]]; then
    python3 - "$file" "$CLOUD_SQL_CONNECTION" <<'PY'
import getpass
import sys
from urllib.parse import quote

with open("/dev/tty", "r+") as tty:
    tty.write("Existing Cloud SQL database name [rove]: ")
    tty.flush()
    database = tty.readline().strip() or "rove"

    tty.write("Existing Cloud SQL username [rove]: ")
    tty.flush()
    username = tty.readline().strip() or "rove"

    password = getpass.getpass(
        "Existing Cloud SQL password (hidden): ",
        stream=tty,
    )

if not password:
    raise SystemExit("Database password cannot be empty.")

url = (
    "postgresql+psycopg://"
    + quote(username, safe="")
    + ":"
    + quote(password, safe="")
    + "@/"
    + quote(database, safe="")
    + "?host=/cloudsql/"
    + sys.argv[2]
)

open(sys.argv[1], "w").write(url)
PY
  else
    printf 'Enter %s from your local configuration (hidden): ' \
      "$kind" > /dev/tty

    IFS= read -r -s value < /dev/tty
    printf '\n' > /dev/tty

    [[ -n "$value" ]] || fail "$kind cannot be empty"

    printf '%s' "$value" > "$file"
    unset value
  fi

  gc secrets create "$name" \
    --replication-policy=automatic \
    --data-file="$file" \
    --quiet
)

echo "=== Ensure required credentials exist ==="
echo "Use your existing Cloud SQL password, which may differ from Docker."

ensure_secret "$DATABASE_URL_SECRET" database
ensure_secret "$ADMIN_USERNAME_SECRET" ADMIN_USERNAME
ensure_secret "$ADMIN_PASSWORD_SECRET" ADMIN_PASSWORD

export DATABASE_URL_SECRET ADMIN_USERNAME_SECRET ADMIN_PASSWORD_SECRET
export EXTRA_SECRETS="${EXTRA_SECRETS:-}"

python3 - "$work" "${API_ENV_FILE:-}" <<'PY'
import json
import os
import re
import sys
from pathlib import Path

work = Path(sys.argv[1])

references = {
    key: os.environ[key + "_SECRET"] + ":latest"
    for key in ("DATABASE_URL", "ADMIN_USERNAME", "ADMIN_PASSWORD")
}

for item in filter(None, os.environ["EXTRA_SECRETS"].split(",")):
    key, separator, value = item.partition("=")

    if not separator or key in references:
        raise SystemExit("Invalid or duplicate EXTRA_SECRETS entry.")

    references[key] = value

for key, value in references.items():
    if (
        not re.fullmatch(r"[A-Z][A-Z0-9_]*", key)
        or not re.fullmatch(r"[A-Za-z0-9_-]+:(latest|[0-9]+)", value)
    ):
        raise SystemExit("Invalid Secret Manager reference.")

(work / "secret-args").write_text(
    ",".join(key + "=" + value for key, value in references.items())
)
(work / "secret-refs").write_text(
    "".join(value + "\n" for value in sorted(set(references.values())))
)

settings = json.load(open(sys.argv[2])) if sys.argv[2] else {}

if (
    not isinstance(settings, dict)
    or any(not isinstance(value, str) for value in settings.values())
    or set(settings).intersection(references)
):
    raise SystemExit(
        "API_ENV_FILE must contain string settings and no duplicate secret keys."
    )

settings.update(
    APP_ENV="production",
    AUTH_COOKIE_SECURE="true",
    SESSION_COOKIE_SECURE="true",
)

(work / "env.json").write_text(json.dumps(settings))
PY

while IFS=: read -r secret version; do
  [[ "$(gc secrets versions describe "$version" \
    --secret="$secret" --format='value(state)')" == ENABLED ]] ||
    fail "Secret is not enabled: $secret:$version"

  gc secrets add-iam-policy-binding "$secret" \
    --member="serviceAccount:$api_account" \
    --role=roles/secretmanager.secretAccessor \
    --condition=None \
    --quiet >/dev/null
done < "$work/secret-refs"

gc auth configure-docker "$REGION-docker.pkg.dev" --quiet

registry="$REGION-docker.pkg.dev/$GCP_PROJECT/$REPOSITORY"
tag="demo-$(date -u +%Y%m%d%H%M%S)-${commit:0:12}"

api_image="${API_IMAGE:-}"

echo "=== Select existing API image where available ==="

if [[ -z "$api_image" && "${FORCE_API_BUILD:-0}" != 1 ]]; then
  gc artifacts docker images list "$registry" \
    --format=json > "$work/images.json"

  api_image=$(python3 - "$work/images.json" "$registry/rove-api" <<'PY'
import json
import re
import sys

rows = json.load(open(sys.argv[1]))
rows.sort(
    key=lambda row: row.get("updateTime", row.get("createTime", "")),
    reverse=True,
)

for row in rows:
    image = (
        row.get("package")
        or row.get("image")
        or row.get("uri")
        or ""
    )

    if str(image).split("@")[0] not in (
        sys.argv[2],
        "https://" + sys.argv[2],
    ):
        continue

    digest = re.search(r"sha256:[0-9a-f]{64}", json.dumps(row))

    if digest:
        print(sys.argv[2] + "@" + digest.group())
        break
PY
)
fi

# Build API only when requested or no reusable API image was found.
if [[ -z "$api_image" || "${FORCE_API_BUILD:-0}" == 1 ]]; then
  echo "=== Build API image ==="

  api_image="$registry/rove-api:$tag"

  if [[ -n "${CORPORATE_CA_FILE:-}" ]]; then
    test -s "$CORPORATE_CA_FILE"

    docker buildx build \
      --platform linux/amd64 \
      --pull \
      --push \
      --secret "id=corporate_ca,src=$CORPORATE_CA_FILE" \
      -t "$api_image" \
      -f server/Dockerfile server
  else
    docker buildx build \
      --platform linux/amd64 \
      --pull \
      --push \
      -t "$api_image" \
      -f server/Dockerfile server
  fi
fi

echo "API image: $api_image"
echo "=== Create or update API service ==="

# Public API access supports the existing Nginx proxy architecture.
# Application authentication remains enabled.
# Startup runs normal Alembic migrations against the existing database.
gc run deploy "$API_SERVICE" \
  --region="$REGION" \
  --image="$api_image" \
  --service-account="$api_account" \
  --port=8000 \
  --ingress=all \
  --allow-unauthenticated \
  --add-cloudsql-instances="$CLOUD_SQL_CONNECTION" \
  --env-vars-file="$work/env.json" \
  --update-secrets="$(cat "$work/secret-args")" \
  --max-instances=1 \
  --command=sh \
  --args='-c,alembic upgrade head && exec uvicorn app.main:app --host 0.0.0.0 --port 8000' \
  --quiet

gc run services update-traffic "$API_SERVICE" \
  --region="$REGION" \
  --to-latest \
  --quiet

api_url=$(gc run services describe "$API_SERVICE" \
  --region="$REGION" \
  --format='value(status.url)')

curl --fail --silent --show-error \
  --retry 5 --retry-all-errors --max-time 60 \
  "$api_url/api/health" >/dev/null

echo "=== Build fresh frontend image ==="

cat > "$work/web/nginx.conf" <<EOF
server {
  listen 8080;
  server_name _;

  root /usr/share/nginx/html;
  index index.html;

  location /api/ {
    proxy_pass ${api_url};
    proxy_ssl_server_name on;
    proxy_set_header Host ${api_url#https://};
    proxy_set_header X-Forwarded-Proto https;
    proxy_http_version 1.1;
    proxy_read_timeout 120s;
  }

  location = /index.html {
    add_header Cache-Control "no-cache, no-store, must-revalidate" always;
  }

  location = /version.json {
    add_header Cache-Control "no-store" always;
  }

  location /assets/ {
    try_files \$uri =404;
    add_header Cache-Control "public, max-age=31536000, immutable";
  }

  location / {
    add_header Cache-Control "no-cache, no-store, must-revalidate" always;
    try_files \$uri \$uri/ /index.html;
  }
}
EOF

python3 - "$work/web/dist/version.json" "$commit" "$BRANCH" <<'PY'
import json
import sys

json.dump(
    {"commit": sys.argv[2], "branch": sys.argv[3]},
    open(sys.argv[1], "w"),
)
PY

cat > "$work/web/Dockerfile" <<'EOF'
FROM nginx:alpine
COPY nginx.conf /etc/nginx/conf.d/default.conf
COPY dist /usr/share/nginx/html
EXPOSE 8080
CMD ["nginx", "-g", "daemon off;"]
EOF

web_image="$registry/rove-web:$tag"

# Frontend is ALWAYS built fresh with a unique image tag.
docker buildx build \
  --platform linux/amd64 \
  --pull \
  --no-cache \
  --push \
  -t "$web_image" \
  "$work/web"

echo "=== Deploy fresh frontend revision ==="

gc run deploy "$WEB_SERVICE" \
  --region="$REGION" \
  --image="$web_image" \
  --service-account="$web_account" \
  --port=8080 \
  --ingress=all \
  --allow-unauthenticated \
  --quiet

web_url=$(gc run services describe "$WEB_SERVICE" \
  --region="$REGION" \
  --format='value(status.url)')

gc run services update "$API_SERVICE" \
  --region="$REGION" \
  --update-env-vars="CORS_ORIGIN=$web_url" \
  --quiet

gc run services update-traffic "$API_SERVICE" \
  --region="$REGION" \
  --to-latest \
  --quiet

gc run services update-traffic "$WEB_SERVICE" \
  --region="$REGION" \
  --to-latest \
  --quiet

echo "=== Verify live frontend and API ==="

curl --fail --silent --show-error \
  --retry 5 --retry-all-errors --max-time 60 \
  "$web_url/api/health" >/dev/null

curl --fail --silent --show-error --max-time 60 \
  "$web_url/version.json?deploy=$tag" > "$work/version.json"

python3 - "$work/version.json" "$commit" <<'PY'
import json
import sys

served = json.load(open(sys.argv[1]))

if served.get("commit") != sys.argv[2]:
    raise SystemExit("Served frontend does not match the newly built commit.")

print("PASS: live frontend matches freshly built source.")
PY

curl --fail --silent --show-error --max-time 60 \
  "$web_url/api/catalogue/vehicles" > "$work/catalogue.json"

python3 - "$work/catalogue.json" <<'PY'
import json
import sys

vehicles = json.load(open(sys.argv[1]))

if not isinstance(vehicles, list):
    raise SystemExit("Expected a catalogue JSON array.")

print("Catalogue vehicle count:", len(vehicles))

if not vehicles:
    print("WARNING: No public inventory; dropdowns will be empty.")
    print("No inventory was imported or database reset performed.")
PY

echo
echo "DEMO URL: $web_url/?deploy=$tag"
echo "Frontend branch: $BRANCH"
echo "Frontend commit: $commit"
echo "Frontend image: $web_image"
echo "API image: $api_image"
echo
echo "Open the DEMO URL in a private browser window."

cd ~/Documents/repo/rove || exit 1
chmod +x new-deploy.sh
bash -n new-deploy.sh

BRANCH=fix/frontend-build-source-cleanup ./new-deploy.sh

#!/usr/bin/env bash
set -euo pipefail

# ============================================================
# ROVE - CONSOLIDATED GCP DEPLOYMENT
# ============================================================
#
# Behaviour:
#
#   PostgreSQL:
#     - NEVER deletes the database
#     - Creates Cloud SQL only when missing
#     - Preserves all existing data
#     - Creates ROVE DB only when missing
#
#   Deployment:
#     - Builds new API image
#     - Deploys new Cloud Run API revision
#     - Runs Alembic migrations
#     - Builds React
#     - Builds new Web image
#     - Deploys new Cloud Run Web revision
#
#   Bank Cloud SQL requirements:
#     - SSL enabled
#     - IAM DB authentication enabled
#     - Password complexity enabled
#     - PostgreSQL pgAudit enabled
#
# ============================================================


# ============================================================
# CONFIGURATION
# ============================================================

PROJECT_DIR="/Users/srujan.kumar/Documents/git/rove"

GCP_PROJECT="tactical-innovation-f2b97b"
REGION="europe-west1"

REPOSITORY="cloud-run-source-deploy"

API_SERVICE="rove-api"
WEB_SERVICE="rove-web"

API_IMAGE="rove-api"
WEB_IMAGE="rove-web"

DB_INSTANCE="rove-postgres"
DB_NAME="rove"
DB_USER="postgres"

# Must satisfy bank password complexity policy.
# 19 characters, uppercase/lowercase/numbers/special characters.
DB_PASSWORD='RovePov#2026Secure!'

DEPLOYMENT_ID="$(date +%Y%m%d-%H%M%S)"


# ============================================================
# MOVE TO PROJECT
# ============================================================

cd "$PROJECT_DIR"


# ============================================================
# TEMPORARY DEPLOYMENT FILES
# ============================================================

API_DOCKERFILE="server/Dockerfile.gcp"
WEB_DOCKERFILE="Dockerfile.gcp"
WEB_DOCKERIGNORE="Dockerfile.gcp.dockerignore"
NGINX_CONFIG="nginx.conf.gcp"


cleanup() {

    rm -f \
        "$API_DOCKERFILE" \
        "$WEB_DOCKERFILE" \
        "$WEB_DOCKERIGNORE" \
        "$NGINX_CONFIG"

}

trap cleanup EXIT


# ============================================================
# HEADER
# ============================================================

echo
echo "============================================================"
echo " ROVE GCP DEPLOYMENT"
echo "============================================================"
echo
echo "Deployment ID : $DEPLOYMENT_ID"
echo "Project       : $GCP_PROJECT"
echo "Region        : $REGION"
echo "Database      : $DB_INSTANCE"
echo
echo "IMPORTANT:"
echo "Existing PostgreSQL data will be PRESERVED."
echo
echo "============================================================"


# ============================================================
# PRE-CHECKS
# ============================================================

command -v gcloud >/dev/null 2>&1 || {
    echo "ERROR: gcloud CLI is not installed."
    exit 1
}

command -v docker >/dev/null 2>&1 || {
    echo "ERROR: Docker is not installed."
    exit 1
}

command -v npm >/dev/null 2>&1 || {
    echo "ERROR: npm is not installed."
    exit 1
}

command -v python3 >/dev/null 2>&1 || {
    echo "ERROR: python3 is not installed."
    exit 1
}

command -v curl >/dev/null 2>&1 || {
    echo "ERROR: curl is not installed."
    exit 1
}


docker info >/dev/null 2>&1 || {
    echo "ERROR: Docker Desktop is not running."
    exit 1
}


# ============================================================
# FIX STALE NODE CERT ENVIRONMENT
# ============================================================

# Previous bank-certificate work may have left NODE_EXTRA_CA_CERTS
# pointing to a file that no longer exists.

if [ -n "${NODE_EXTRA_CA_CERTS:-}" ]; then

    if [ ! -f "$NODE_EXTRA_CA_CERTS" ]; then

        echo
        echo "Removing stale NODE_EXTRA_CA_CERTS:"
        echo "$NODE_EXTRA_CA_CERTS"

        unset NODE_EXTRA_CA_CERTS

    fi

fi


# ============================================================
# 0. GCP AUTHENTICATION
# ============================================================

echo
echo "============================================================"
echo "[0/10] GCP AUTHENTICATION"
echo "============================================================"


if ! gcloud auth print-access-token >/dev/null 2>&1; then

    echo "GCP authentication required..."

    gcloud auth login

fi


gcloud config set project "$GCP_PROJECT" >/dev/null


echo "Active GCP account:"

gcloud auth list \
    --filter=status:ACTIVE \
    --format="value(account)"


echo
echo "Enabling required APIs..."


gcloud services enable \
    run.googleapis.com \
    sqladmin.googleapis.com \
    artifactregistry.googleapis.com \
    --project="$GCP_PROJECT" \
    --quiet


echo
echo "Configuring Docker authentication..."


gcloud auth configure-docker \
    "${REGION}-docker.pkg.dev" \
    --quiet >/dev/null


# ============================================================
# CHECK ARTIFACT REGISTRY
# ============================================================

if ! gcloud artifacts repositories describe "$REPOSITORY" \
    --project="$GCP_PROJECT" \
    --location="$REGION" \
    >/dev/null 2>&1
then

    echo
    echo "Artifact Registry repository not found."
    echo "Creating: $REPOSITORY"

    gcloud artifacts repositories create "$REPOSITORY" \
        --project="$GCP_PROJECT" \
        --location="$REGION" \
        --repository-format=docker \
        --description="ROVE container images" \
        --quiet

else

    echo
    echo "Artifact Registry repository exists:"
    echo "$REPOSITORY"

fi


# ============================================================
# 1. CHECK / CREATE POSTGRESQL
# ============================================================

echo
echo "============================================================"
echo "[1/10] CHECK POSTGRESQL"
echo "============================================================"


DB_CREATED="false"


if gcloud sql instances describe "$DB_INSTANCE" \
    --project="$GCP_PROJECT" \
    >/dev/null 2>&1
then

    echo
    echo "Existing PostgreSQL instance found:"
    echo "$DB_INSTANCE"

    echo
    echo "Existing database WILL NOT be deleted."

else

    echo
    echo "PostgreSQL instance not found."
    echo
    echo "Creating bank-compliant Cloud SQL PostgreSQL instance..."


    gcloud sql instances create "$DB_INSTANCE" \
        --project="$GCP_PROJECT" \
        --database-version=POSTGRES_16 \
        --edition=ENTERPRISE \
        --tier=db-f1-micro \
        --region="$REGION" \
        --storage-size=10GB \
        --storage-type=SSD \
        --availability-type=zonal \
        --ssl-mode=ENCRYPTED_ONLY \
        --database-flags=cloudsql.iam_authentication=on,cloudsql.enable_pgaudit=on,pgaudit.log=all \
        --enable-password-policy \
        --password-policy-min-length=12 \
        --password-policy-complexity=COMPLEXITY_DEFAULT \
        --password-policy-disallow-username-substring \
        --root-password="$DB_PASSWORD" \
        --no-deletion-protection \
        --quiet


    DB_CREATED="true"


    echo
    echo "PostgreSQL created successfully."

fi


# ============================================================
# CHECK DATABASE STATE
# ============================================================

echo
echo "Checking PostgreSQL state..."


DB_STATE="$(gcloud sql instances describe "$DB_INSTANCE" \
    --project="$GCP_PROJECT" \
    --format='value(state)')"


echo "State: $DB_STATE"


if [ "$DB_STATE" != "RUNNABLE" ]; then

    echo
    echo "Database is not currently RUNNABLE."
    echo "Attempting to activate it..."

    gcloud sql instances patch "$DB_INSTANCE" \
        --project="$GCP_PROJECT" \
        --activation-policy=ALWAYS \
        --quiet || true


    echo
    echo "Waiting for PostgreSQL..."


    for i in {1..30}
    do

        DB_STATE="$(gcloud sql instances describe "$DB_INSTANCE" \
            --project="$GCP_PROJECT" \
            --format='value(state)' 2>/dev/null || true)"


        echo "Database state: $DB_STATE"


        if [ "$DB_STATE" = "RUNNABLE" ]; then
            break
        fi


        sleep 5

    done

fi


if [ "$DB_STATE" != "RUNNABLE" ]; then

    echo
    echo "ERROR: PostgreSQL is not RUNNABLE."
    echo "Current state: $DB_STATE"

    exit 1

fi


# ============================================================
# 2. SYNCHRONISE POSTGRES PASSWORD
# ============================================================

echo
echo "============================================================"
echo "[2/10] CONFIGURE DATABASE LOGIN"
echo "============================================================"


# This does NOT delete or modify application data.
# It guarantees DATABASE_URL always uses the current password.

echo "Synchronising PostgreSQL administrator password..."


gcloud sql users set-password "$DB_USER" \
    --instance="$DB_INSTANCE" \
    --project="$GCP_PROJECT" \
    --password="$DB_PASSWORD" \
    --quiet


# ============================================================
# 3. CHECK / CREATE ROVE DATABASE
# ============================================================

echo
echo "============================================================"
echo "[3/10] CHECK ROVE DATABASE"
echo "============================================================"


if gcloud sql databases describe "$DB_NAME" \
    --instance="$DB_INSTANCE" \
    --project="$GCP_PROJECT" \
    >/dev/null 2>&1
then

    echo
    echo "Database exists:"
    echo "$DB_NAME"

    echo
    echo "Existing application data preserved."

else

    echo
    echo "ROVE database not found."
    echo "Creating database: $DB_NAME"


    gcloud sql databases create "$DB_NAME" \
        --instance="$DB_INSTANCE" \
        --project="$GCP_PROJECT" \
        --quiet


    echo
    echo "ROVE database created."

fi


# ============================================================
# CLOUD SQL CONNECTION NAME
# ============================================================

CONNECTION_NAME="$(gcloud sql instances describe "$DB_INSTANCE" \
    --project="$GCP_PROJECT" \
    --format='value(connectionName)')"


if [ -z "$CONNECTION_NAME" ]; then

    echo "ERROR: Unable to obtain Cloud SQL connection name."

    exit 1

fi


echo
echo "Cloud SQL connection:"
echo "$CONNECTION_NAME"


# ============================================================
# URL-ENCODE DATABASE PASSWORD
# ============================================================

# Important because password contains characters such as # and !
# that have special meaning inside URLs.

DB_PASSWORD_URL="$(
    DB_PASSWORD="$DB_PASSWORD" python3 <<'PY'
import os
from urllib.parse import quote

print(
    quote(
        os.environ["DB_PASSWORD"],
        safe=""
    )
)
PY
)"


DATABASE_URL="postgresql+psycopg://${DB_USER}:${DB_PASSWORD_URL}@/${DB_NAME}?host=/cloudsql/${CONNECTION_NAME}"


# ============================================================
# 4. DETERMINE CLOUD RUN SERVICE ACCOUNT
# ============================================================

echo
echo "============================================================"
echo "[4/10] CONFIGURE CLOUD RUN DATABASE ACCESS"
echo "============================================================"


PROJECT_NUMBER="$(gcloud projects describe "$GCP_PROJECT" \
    --format='value(projectNumber)')"


# Preserve existing rove-api service account when possible.

EXISTING_API_SA="$(gcloud run services describe "$API_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --format='value(spec.template.spec.serviceAccountName)' \
    2>/dev/null || true)"


if [ -n "$EXISTING_API_SA" ]; then

    CLOUD_RUN_SA="$EXISTING_API_SA"

    echo "Reusing existing API service account:"

else

    CLOUD_RUN_SA="${PROJECT_NUMBER}-compute@developer.gserviceaccount.com"

    echo "Using default Cloud Run service account:"

fi


echo "$CLOUD_RUN_SA"


echo
echo "Ensuring Cloud SQL Client access..."


# Bank environments may prevent users from modifying IAM.
# Try the binding but do not destroy the deployment if IAM
# is centrally managed and the binding already exists.

gcloud projects add-iam-policy-binding "$GCP_PROJECT" \
    --member="serviceAccount:${CLOUD_RUN_SA}" \
    --role="roles/cloudsql.client" \
    --condition=None \
    --quiet \
    >/dev/null 2>&1 || {

        echo
        echo "NOTE:"
        echo "Unable to modify project IAM."
        echo "Continuing because bank IAM may already provide Cloud SQL access."

    }


# ============================================================
# 5. CREATE API DOCKERFILE
# ============================================================

echo
echo "============================================================"
echo "[5/10] BUILD API IMAGE"
echo "============================================================"


cat > "$API_DOCKERFILE" <<'EOF'
FROM python:3.12-slim

WORKDIR /app

ENV PYTHONDONTWRITEBYTECODE=1
ENV PYTHONUNBUFFERED=1


COPY requirements.txt .


RUN pip install \
    --no-cache-dir \
    --trusted-host pypi.org \
    --trusted-host files.pythonhosted.org \
    --trusted-host pypi.python.org \
    -r requirements.txt


COPY app ./app
COPY scripts ./scripts
COPY alembic ./alembic
COPY alembic.ini ./alembic.ini


EXPOSE 8080


# Existing DB is preserved.
#
# Each new Cloud Run revision:
#   1. applies pending Alembic migrations
#   2. starts FastAPI
#
# alembic upgrade head is safe when schema is already current.

CMD ["sh", "-c", "alembic upgrade head && exec uvicorn app.main:app --host 0.0.0.0 --port ${PORT:-8080}"]
EOF


API_VERSION_IMAGE="${REGION}-docker.pkg.dev/${GCP_PROJECT}/${REPOSITORY}/${API_IMAGE}:${DEPLOYMENT_ID}"


echo
echo "API image:"
echo "$API_VERSION_IMAGE"


docker buildx build \
    --platform linux/amd64 \
    -f "$API_DOCKERFILE" \
    -t "$API_VERSION_IMAGE" \
    --push \
    server


# ============================================================
# 6. DEPLOY / RESTART API
# ============================================================

echo
echo "============================================================"
echo "[6/10] DEPLOY API"
echo "============================================================"


echo "Deploying new rove-api revision..."


set +e

gcloud run deploy "$API_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --platform=managed \
    --image="$API_VERSION_IMAGE" \
    --port=8080 \
    --add-cloudsql-instances="$CONNECTION_NAME" \
    --service-account="$CLOUD_RUN_SA" \
    --set-env-vars="DATABASE_URL=${DATABASE_URL},APP_ENV=production,DEPLOYMENT_ID=${DEPLOYMENT_ID}" \
    --no-invoker-iam-check \
    --quiet

API_DEPLOY_STATUS=$?

set -e


if [ "$API_DEPLOY_STATUS" -ne 0 ]; then

    echo
    echo "============================================================"
    echo " API DEPLOYMENT FAILED"
    echo "============================================================"

    echo
    echo "Recent rove-api logs:"
    echo


    gcloud logging read \
        "resource.type=cloud_run_revision AND resource.labels.service_name=${API_SERVICE}" \
        --project="$GCP_PROJECT" \
        --limit=60 \
        --freshness=20m \
        --format="value(timestamp,severity,textPayload)" \
        2>/dev/null || true


    echo
    echo "Database was NOT deleted."
    echo "API image remains available:"
    echo "$API_VERSION_IMAGE"

    exit "$API_DEPLOY_STATUS"

fi


# ============================================================
# GET API URL
# ============================================================

API_URL="$(gcloud run services describe "$API_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --format='value(status.url)')"


if [ -z "$API_URL" ]; then

    echo "ERROR: Unable to determine rove-api URL."

    exit 1

fi


echo
echo "API deployed:"
echo "$API_URL"


# ============================================================
# TEST API
# ============================================================

echo
echo "Waiting for API..."

sleep 3


API_HTTP="$(curl \
    --silent \
    --show-error \
    --output /tmp/rove-api-docs.html \
    --write-out "%{http_code}" \
    "$API_URL/docs" \
    2>/dev/null || true)"


echo "FastAPI /docs: HTTP $API_HTTP"


if [ "$API_HTTP" != "200" ]; then

    echo
    echo "ERROR: FastAPI did not start successfully."
    echo
    echo "Recent logs:"


    gcloud logging read \
        "resource.type=cloud_run_revision AND resource.labels.service_name=${API_SERVICE}" \
        --project="$GCP_PROJECT" \
        --limit=60 \
        --freshness=20m \
        --format="value(timestamp,severity,textPayload)" \
        2>/dev/null || true


    exit 1

fi


echo
echo "Testing PostgreSQL connectivity..."


DB_HEALTH="$(curl \
    --silent \
    --show-error \
    "$API_URL/api/health" \
    2>/dev/null || true)"


echo "$DB_HEALTH"


if ! echo "$DB_HEALTH" | grep -q '"database":"connected"'; then

    # Allow output containing whitespace around JSON separators as well.

    if ! echo "$DB_HEALTH" | grep -q 'connected'; then

        echo
        echo "ERROR: API started but PostgreSQL health check failed."

        echo
        echo "Recent API logs:"


        gcloud logging read \
            "resource.type=cloud_run_revision AND resource.labels.service_name=${API_SERVICE}" \
            --project="$GCP_PROJECT" \
            --limit=60 \
            --freshness=20m \
            --format="value(timestamp,severity,textPayload)" \
            2>/dev/null || true


        exit 1

    fi

fi


echo
echo "API + PostgreSQL OK."


# ============================================================
# 7. BUILD REACT
# ============================================================

echo
echo "============================================================"
echo "[7/10] BUILD REACT"
echo "============================================================"


rm -rf dist


NODE_USE_SYSTEM_CA=1 npm ci

NODE_USE_SYSTEM_CA=1 npm run build


if [ ! -d "dist" ]; then

    echo "ERROR: React dist directory was not created."

    exit 1

fi


if [ ! -f "dist/index.html" ]; then

    echo "ERROR: dist/index.html was not created."

    exit 1

fi


echo
echo "React build successful."


# ============================================================
# 8. PREPARE WEB IMAGE
# ============================================================

echo
echo "============================================================"
echo "[8/10] PREPARE WEB"
echo "============================================================"


API_HOST="$(echo "$API_URL" \
    | sed 's#^https://##' \
    | sed 's#/$##')"


cat > "$NGINX_CONFIG" <<EOF
server {

    listen 8080;
    server_name _;

    root /usr/share/nginx/html;
    index index.html;


    # ========================================================
    # ROVE API
    # ========================================================

    location /api/ {

        proxy_ssl_server_name on;
        proxy_ssl_name ${API_HOST};

        proxy_set_header Host ${API_HOST};

        proxy_set_header X-Real-IP \$remote_addr;
        proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto https;

        proxy_connect_timeout 30s;
        proxy_read_timeout 120s;

        proxy_pass ${API_URL}/api/;

    }


    # ========================================================
    # REACT SPA
    # ========================================================

    location / {

        try_files \$uri \$uri/ /index.html;

    }

}
EOF


cat > "$WEB_DOCKERFILE" <<'EOF'
FROM nginx:alpine

COPY nginx.conf.gcp /etc/nginx/conf.d/default.conf

COPY dist /usr/share/nginx/html

EXPOSE 8080

CMD ["nginx", "-g", "daemon off;"]
EOF


# ============================================================
# DOCKERFILE-SPECIFIC IGNORE FILE
# ============================================================

# This prevents us from modifying the repository's real
# .dockerignore, which previously caused git pull/rebase issues.

if [ -f ".dockerignore" ]; then

    python3 <<'PY'
from pathlib import Path

source = Path(".dockerignore")
target = Path("Dockerfile.gcp.dockerignore")

blocked = {
    "dist",
    "./dist",
    "**/dist",
    "*/dist",
}

clean = []

for line in source.read_text().splitlines():

    value = line.strip().rstrip("/")

    if value in blocked:
        continue

    clean.append(line)

target.write_text("\n".join(clean) + "\n")
PY

else

    cat > "$WEB_DOCKERIGNORE" <<'EOF'
.git
node_modules
EOF

fi


WEB_VERSION_IMAGE="${REGION}-docker.pkg.dev/${GCP_PROJECT}/${REPOSITORY}/${WEB_IMAGE}:${DEPLOYMENT_ID}"


# ============================================================
# 9. BUILD WEB IMAGE
# ============================================================

echo
echo "============================================================"
echo "[9/10] BUILD WEB IMAGE"
echo "============================================================"


echo
echo "Web image:"
echo "$WEB_VERSION_IMAGE"


docker buildx build \
    --platform linux/amd64 \
    -f "$WEB_DOCKERFILE" \
    -t "$WEB_VERSION_IMAGE" \
    --push \
    .


# ============================================================
# 10. DEPLOY / RESTART WEB
# ============================================================

echo
echo "============================================================"
echo "[10/10] DEPLOY WEB"
echo "============================================================"


set +e

gcloud run deploy "$WEB_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --platform=managed \
    --image="$WEB_VERSION_IMAGE" \
    --port=8080 \
    --set-env-vars="DEPLOYMENT_ID=${DEPLOYMENT_ID}" \
    --no-invoker-iam-check \
    --quiet

WEB_DEPLOY_STATUS=$?

set -e


if [ "$WEB_DEPLOY_STATUS" -ne 0 ]; then

    echo
    echo "============================================================"
    echo " WEB DEPLOYMENT FAILED"
    echo "============================================================"

    echo
    echo "Recent rove-web logs:"


    gcloud logging read \
        "resource.type=cloud_run_revision AND resource.labels.service_name=${WEB_SERVICE}" \
        --project="$GCP_PROJECT" \
        --limit=60 \
        --freshness=20m \
        --format="value(timestamp,severity,textPayload)" \
        2>/dev/null || true


    exit "$WEB_DEPLOY_STATUS"

fi


# ============================================================
# GET WEB URL
# ============================================================

WEB_URL="$(gcloud run services describe "$WEB_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --format='value(status.url)')"


if [ -z "$WEB_URL" ]; then

    echo "ERROR: Unable to determine rove-web URL."

    exit 1

fi


# ============================================================
# FINAL TESTS
# ============================================================

echo
echo "============================================================"
echo " FINAL TESTS"
echo "============================================================"


echo
echo "1. API"

curl \
    --silent \
    --output /dev/null \
    --write-out "FastAPI docs: HTTP %{http_code}\n" \
    "$API_URL/docs" || true


echo
echo "2. DATABASE"

curl \
    --silent \
    "$API_URL/api/health" || true


echo
echo
echo "3. WEBSITE"

curl \
    --silent \
    --output /dev/null \
    --write-out "ROVE Web: HTTP %{http_code}\n" \
    "$WEB_URL/" || true


echo
echo "4. API THROUGH WEB PROXY"

curl \
    --silent \
    "$WEB_URL/api/health" || true


# ============================================================
# GIT CLEANLINESS CHECK
# ============================================================

echo
echo
echo "============================================================"
echo " GIT STATUS"
echo "============================================================"

cleanup

git status --short || true


# ============================================================
# COMPLETE
# ============================================================

echo
echo
echo "============================================================"
echo " ROVE DEPLOYMENT SUCCESSFUL"
echo "============================================================"

echo
echo "Deployment ID:"
echo "$DEPLOYMENT_ID"

echo
echo "Database instance:"
echo "$DB_INSTANCE"

echo
echo "Database:"
echo "$DB_NAME"

echo
echo "Database state:"
echo "$DB_STATE"

echo
echo "Cloud SQL connection:"
echo "$CONNECTION_NAME"

echo
echo "API:"
echo "$API_URL"

echo
echo "Website:"
echo "$WEB_URL"

echo
echo "API image:"
echo "$API_VERSION_IMAGE"

echo
echo "Web image:"
echo "$WEB_VERSION_IMAGE"

echo
echo "============================================================"
echo " DATABASE DATA PRESERVED"
echo "============================================================"

echo
echo "Future deployments:"
echo "./deploy/deploy-rove-updates-gcp.sh"

echo
echo "============================================================"


open "$WEB_URL" 2>/dev/null || true

#!/usr/bin/env bash
set -euo pipefail

# ==========================================================
# ROVE - FULL RESET + GCP DEPLOYMENT
#
# PoV behaviour:
#   1. Delete existing ROVE Cloud Run services
#   2. Delete existing ROVE PostgreSQL instance
#   3. Create fresh PostgreSQL
#   4. Create database
#   5. Deploy API
#   6. Run Alembic migrations automatically
#   7. Build React
#   8. Deploy Web
#
# WARNING:
#   THIS SCRIPT DELETES ALL EXISTING ROVE DATABASE DATA.
# ==========================================================


# ==========================================================
# CONFIGURATION
# ==========================================================

PROJECT_DIR="/Users/srujan.kumar/Documents/git/rove"

GCP_PROJECT="tactical-innovation-f2b97b"
REGION="europe-west1"
REPOSITORY="cloud-run-source-deploy"

WEB_SERVICE="rove-web"
API_SERVICE="rove-api"

WEB_IMAGE="rove-web"
API_IMAGE="rove-api"

DB_INSTANCE="rove-postgres"
DB_NAME="rove"
DB_USER="postgres"

# 19 characters, upper/lower/numeric/special
DB_PASSWORD="RovePov#2026Secure!"

DEPLOYMENT_ID="$(date +%Y%m%d-%H%M%S)"

cd "$PROJECT_DIR"


echo
echo "=========================================================="
echo " ROVE FULL RESET + DEPLOYMENT"
echo "=========================================================="
echo "Deployment ID : $DEPLOYMENT_ID"
echo "Project       : $GCP_PROJECT"
echo "Region        : $REGION"
echo "Database      : $DB_INSTANCE"
echo "=========================================================="


# ==========================================================
# PRE-CHECKS
# ==========================================================

command -v gcloud >/dev/null 2>&1 || {
    echo "ERROR: gcloud CLI not installed."
    exit 1
}

command -v docker >/dev/null 2>&1 || {
    echo "ERROR: Docker not installed."
    exit 1
}

command -v npm >/dev/null 2>&1 || {
    echo "ERROR: npm not installed."
    exit 1
}

docker info >/dev/null 2>&1 || {
    echo "ERROR: Docker Desktop is not running."
    exit 1
}


# ==========================================================
# 0. GCP AUTHENTICATION
# ==========================================================

echo
echo "=========================================================="
echo "[0/10] GCP AUTHENTICATION"
echo "=========================================================="

if ! gcloud auth print-access-token >/dev/null 2>&1; then
    gcloud auth login
fi

gcloud config set project "$GCP_PROJECT" >/dev/null

echo "Enabling required GCP APIs..."

gcloud services enable \
    run.googleapis.com \
    sqladmin.googleapis.com \
    artifactregistry.googleapis.com \
    --project="$GCP_PROJECT" \
    --quiet

gcloud auth configure-docker \
    "${REGION}-docker.pkg.dev" \
    --quiet >/dev/null


# ==========================================================
# 1. DELETE EXISTING ROVE SERVICES
# ==========================================================

echo
echo "=========================================================="
echo "[1/10] DELETE EXISTING ROVE SERVICES"
echo "=========================================================="

if gcloud run services describe "$WEB_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    >/dev/null 2>&1
then
    echo "Deleting existing web service: $WEB_SERVICE"

    gcloud run services delete "$WEB_SERVICE" \
        --project="$GCP_PROJECT" \
        --region="$REGION" \
        --quiet
else
    echo "No existing $WEB_SERVICE service found."
fi


if gcloud run services describe "$API_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    >/dev/null 2>&1
then
    echo "Deleting existing API service: $API_SERVICE"

    gcloud run services delete "$API_SERVICE" \
        --project="$GCP_PROJECT" \
        --region="$REGION" \
        --quiet
else
    echo "No existing $API_SERVICE service found."
fi


# ==========================================================
# 2. DELETE EXISTING POSTGRESQL
# ==========================================================

echo
echo "=========================================================="
echo "[2/10] DELETE EXISTING POSTGRESQL"
echo "=========================================================="

if gcloud sql instances describe "$DB_INSTANCE" \
    --project="$GCP_PROJECT" \
    >/dev/null 2>&1
then

    echo "Existing database found: $DB_INSTANCE"

    echo "Disabling deletion protection..."

    gcloud sql instances patch "$DB_INSTANCE" \
        --project="$GCP_PROJECT" \
        --no-deletion-protection \
        --quiet \
        >/dev/null 2>&1 || true

    echo "Deleting Cloud SQL instance..."

    gcloud sql instances delete "$DB_INSTANCE" \
        --project="$GCP_PROJECT" \
        --no-enable-final-backup \
        --quiet

    echo "Existing database deleted."

else
    echo "No existing PostgreSQL instance found."
fi


# ==========================================================
# 3. CREATE BANK-COMPLIANT POSTGRESQL
# ==========================================================

echo
echo "=========================================================="
echo "[3/10] CREATE POSTGRESQL"
echo "=========================================================="

echo "Creating bank-compliant Cloud SQL PostgreSQL instance..."

gcloud sql instances create "$DB_INSTANCE" \
    --project="$GCP_PROJECT" \
    --database-version=POSTGRES_16 \
    --edition=ENTERPRISE \
    --tier=db-f1-micro \
    --region="$REGION" \
    --storage-size=10GB \
    --storage-type=SSD \
    --availability-type=zonal \
    --ssl-mode=ENCRYPTED_ONLY \
    --database-flags=cloudsql.iam_authentication=on,cloudsql.enable_pgaudit=on,pgaudit.log=all \
    --enable-password-policy \
    --password-policy-min-length=12 \
    --password-policy-complexity=COMPLEXITY_DEFAULT \
    --password-policy-disallow-username-substring \
    --root-password="$DB_PASSWORD" \
    --no-deletion-protection

echo
echo "PostgreSQL instance created successfully."


# ==========================================================
# 4. CREATE ROVE DATABASE
# ==========================================================

echo
echo "=========================================================="
echo "[4/10] CREATE ROVE DATABASE"
echo "=========================================================="

echo "Creating database: $DB_NAME"

gcloud sql databases create "$DB_NAME" \
    --instance="$DB_INSTANCE" \
    --project="$GCP_PROJECT"

CONNECTION_NAME="$(gcloud sql instances describe "$DB_INSTANCE" \
    --project="$GCP_PROJECT" \
    --format='value(connectionName)')"

echo
echo "Cloud SQL connection:"
echo "$CONNECTION_NAME"


# ==========================================================
# 5. CONFIGURE CLOUD RUN ACCESS
# ==========================================================

echo
echo "=========================================================="
echo "[5/10] CONFIGURE CLOUD SQL ACCESS"
echo "=========================================================="

PROJECT_NUMBER="$(gcloud projects describe "$GCP_PROJECT" \
    --format='value(projectNumber)')"

CLOUD_RUN_SA="${PROJECT_NUMBER}-compute@developer.gserviceaccount.com"

echo "Cloud Run service account:"
echo "$CLOUD_RUN_SA"

echo "Granting Cloud SQL Client role..."

gcloud projects add-iam-policy-binding "$GCP_PROJECT" \
    --member="serviceAccount:${CLOUD_RUN_SA}" \
    --role="roles/cloudsql.client" \
    --condition=None \
    --quiet \
    >/dev/null || true


# ==========================================================
# 6. BUILD API
# ==========================================================

echo
echo "=========================================================="
echo "[6/10] BUILD API"
echo "=========================================================="

cat > server/Dockerfile.gcp <<'EOF'
FROM python:3.12-slim

WORKDIR /app

ENV PYTHONDONTWRITEBYTECODE=1
ENV PYTHONUNBUFFERED=1

COPY requirements.txt .

RUN pip install \
    --no-cache-dir \
    --trusted-host pypi.org \
    --trusted-host files.pythonhosted.org \
    --trusted-host pypi.python.org \
    -r requirements.txt

COPY app ./app
COPY scripts ./scripts
COPY alembic ./alembic
COPY alembic.ini ./alembic.ini

EXPOSE 8080

CMD ["sh", "-c", "alembic upgrade head && exec uvicorn app.main:app --host 0.0.0.0 --port ${PORT:-8080}"]
EOF


API_VERSION_IMAGE="${REGION}-docker.pkg.dev/${GCP_PROJECT}/${REPOSITORY}/${API_IMAGE}:${DEPLOYMENT_ID}"

echo
echo "Building API image:"
echo "$API_VERSION_IMAGE"

docker buildx build \
    --platform linux/amd64 \
    -f server/Dockerfile.gcp \
    -t "$API_VERSION_IMAGE" \
    --push \
    server


# ==========================================================
# 7. DEPLOY API
# ==========================================================

echo
echo "=========================================================="
echo "[7/10] DEPLOY API"
echo "=========================================================="

DATABASE_URL="postgresql+psycopg://${DB_USER}:${DB_PASSWORD}@/${DB_NAME}?host=/cloudsql/${CONNECTION_NAME}"

gcloud run deploy "$API_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --platform=managed \
    --image="$API_VERSION_IMAGE" \
    --port=8080 \
    --add-cloudsql-instances="$CONNECTION_NAME" \
    --service-account="$CLOUD_RUN_SA" \
    --set-env-vars="DATABASE_URL=${DATABASE_URL},APP_ENV=production,DEPLOYMENT_ID=${DEPLOYMENT_ID}" \
    --no-invoker-iam-check \
    --quiet


API_URL="$(gcloud run services describe "$API_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --format='value(status.url)')"

echo
echo "API URL:"
echo "$API_URL"

echo
echo "Testing FastAPI..."

API_HTTP="$(curl -s \
    -o /tmp/rove-api-docs.txt \
    -w "%{http_code}" \
    "$API_URL/docs" || true)"

echo "FastAPI /docs HTTP status: $API_HTTP"


if [ "$API_HTTP" != "200" ]; then

    echo
    echo "ERROR: API did not start successfully."
    echo
    echo "Recent Cloud Run logs:"
    echo

    gcloud logging read \
        "resource.type=cloud_run_revision AND resource.labels.service_name=${API_SERVICE}" \
        --project="$GCP_PROJECT" \
        --limit=50 \
        --freshness=15m \
        --format="value(timestamp,severity,textPayload)"

    exit 1
fi


echo
echo "API running successfully."

echo
echo "Testing database connectivity..."

curl -s "$API_URL/api/health" || true

echo


# ==========================================================
# 8. BUILD REACT
# ==========================================================

echo
echo "=========================================================="
echo "[8/10] BUILD REACT"
echo "=========================================================="

rm -rf dist

NODE_USE_SYSTEM_CA=1 npm ci
NODE_USE_SYSTEM_CA=1 npm run build

if [ ! -d "dist" ]; then
    echo "ERROR: React dist directory was not created."
    exit 1
fi

echo
echo "React build successful."


# Remove dist exclusions from .dockerignore
if [ -f ".dockerignore" ]; then

python3 <<'PY'
from pathlib import Path

p = Path(".dockerignore")

lines = p.read_text().splitlines()

blocked = {
    "dist",
    "./dist",
    "**/dist",
    "*/dist"
}

clean = []

for line in lines:
    value = line.strip().rstrip("/")

    if value in blocked:
        print("Removing .dockerignore entry:", line)
        continue

    clean.append(line)

p.write_text("\n".join(clean) + "\n")
PY

fi


# ==========================================================
# 9. BUILD WEB
# ==========================================================

echo
echo "=========================================================="
echo "[9/10] BUILD WEB"
echo "=========================================================="

API_HOST="$(echo "$API_URL" | sed 's#https://##' | sed 's#/$##')"

cat > nginx.conf.gcp <<EOF
server {

    listen 8080;
    server_name _;

    root /usr/share/nginx/html;
    index index.html;


    # ------------------------------------------------------
    # ROVE API
    # ------------------------------------------------------

    location /api/ {

        proxy_ssl_server_name on;
        proxy_ssl_name ${API_HOST};

        proxy_set_header Host ${API_HOST};

        proxy_set_header X-Real-IP \$remote_addr;
        proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto https;

        proxy_pass ${API_URL}/api/;
    }


    # ------------------------------------------------------
    # REACT SPA
    # ------------------------------------------------------

    location / {

        try_files \$uri \$uri/ /index.html;

    }

}
EOF


cat > Dockerfile.gcp <<'EOF'
FROM nginx:alpine

COPY nginx.conf.gcp /etc/nginx/conf.d/default.conf
COPY dist /usr/share/nginx/html

EXPOSE 8080

CMD ["nginx", "-g", "daemon off;"]
EOF


WEB_VERSION_IMAGE="${REGION}-docker.pkg.dev/${GCP_PROJECT}/${REPOSITORY}/${WEB_IMAGE}:${DEPLOYMENT_ID}"

echo
echo "Building web image:"
echo "$WEB_VERSION_IMAGE"

docker buildx build \
    --platform linux/amd64 \
    -f Dockerfile.gcp \
    -t "$WEB_VERSION_IMAGE" \
    --push \
    .


# ==========================================================
# 10. DEPLOY WEB
# ==========================================================

echo
echo "=========================================================="
echo "[10/10] DEPLOY WEB"
echo "=========================================================="

gcloud run deploy "$WEB_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --platform=managed \
    --image="$WEB_VERSION_IMAGE" \
    --port=8080 \
    --set-env-vars="DEPLOYMENT_ID=${DEPLOYMENT_ID}" \
    --no-invoker-iam-check \
    --quiet


WEB_URL="$(gcloud run services describe "$WEB_SERVICE" \
    --project="$GCP_PROJECT" \
    --region="$REGION" \
    --format='value(status.url)')"


# ==========================================================
# FINAL TESTS
# ==========================================================

echo
echo "=========================================================="
echo " FINAL TESTS"
echo "=========================================================="

echo
echo "API:"
curl -s -o /dev/null \
    -w "HTTP %{http_code}\n" \
    "$API_URL/docs" || true

echo
echo "Database:"
curl -s "$API_URL/api/health" || true

echo
echo
echo "Web:"
curl -s -o /dev/null \
    -w "HTTP %{http_code}\n" \
    "$WEB_URL/" || true


# ==========================================================
# COMPLETE
# ==========================================================

echo
echo
echo "=========================================================="
echo " ROVE DEPLOYMENT SUCCESSFUL"
echo "=========================================================="

echo
echo "Deployment ID:"
echo "$DEPLOYMENT_ID"

echo
echo "Cloud SQL:"
echo "$DB_INSTANCE"

echo
echo "Cloud SQL connection:"
echo "$CONNECTION_NAME"

echo
echo "API:"
echo "$API_URL"

echo
echo "Website:"
echo "$WEB_URL"

echo
echo "API image:"
echo "$API_VERSION_IMAGE"

echo
echo "Web image:"
echo "$WEB_VERSION_IMAGE"

echo
echo "=========================================================="
echo " WARNING: PREVIOUS ROVE DATABASE DATA WAS DELETED"
echo "=========================================================="

open "$WEB_URL" 2>/dev/null || true

cd /Users/srujan.kumar/Documents/git/rove

BRANCH="feature/rove-sync-20261028-123526"
GCP_PROJECT="tactical-innovation-f2b97b"
REGION="europe-west1"
REPOSITORY="cloud-run-source-deploy"
SERVICE="rove-web"

echo "=== 1. GET BANK FEATURE BRANCH ==="

git fetch company "$BRANCH" && \
git switch "$BRANCH" 2>/dev/null || git switch -c "$BRANCH" --track "company/$BRANCH"

git reset --hard "company/$BRANCH"

echo ""
echo "Deploying:"
git log -1 --oneline

echo ""
echo "=== 2. BUILD ROVE ==="

NODE_USE_SYSTEM_CA=1 npm ci && \
NODE_USE_SYSTEM_CA=1 npm run build

echo ""
echo "=== 3. PREPARE CLOUD RUN IMAGE ==="

cat > Dockerfile.gcp <<'EOF'
FROM nginx:alpine

COPY nginx.conf.gcp /etc/nginx/conf.d/default.conf
COPY dist /usr/share/nginx/html

EXPOSE 8080

CMD ["nginx", "-g", "daemon off;"]
EOF

cat > nginx.conf.gcp <<'EOF'
server {
    listen 8080;
    server_name _;

    root /usr/share/nginx/html;
    index index.html;

    location / {
        try_files $uri $uri/ /index.html;
    }
}
EOF

echo ""
echo "=== 4. LOGIN TO GCP ==="

gcloud config set project "$GCP_PROJECT"

gcloud auth configure-docker "${REGION}-docker.pkg.dev" --quiet

TIMESTAMP=$(date +%Y%m%d-%H%M%S)

IMAGE="${REGION}-docker.pkg.dev/${GCP_PROJECT}/${REPOSITORY}/rove-web:${TIMESTAMP}"

echo ""
echo "=== 5. BUILD + PUSH IMAGE ==="

docker buildx build \
  --platform linux/amd64 \
  -f Dockerfile.gcp \
  -t "$IMAGE" \
  --push \
  .

echo ""
echo "=== 6. DEPLOY TO CLOUD RUN ==="

gcloud run deploy "$SERVICE" \
  --image "$IMAGE" \
  --region "$REGION" \
  --platform managed \
  --port 8080 \
  --allow-unauthenticated \
  --quiet

echo ""
echo "=============================================="
echo "ROVE DEPLOYED SUCCESSFULLY"
echo "==============================================="

URL=$(gcloud run services describe "$SERVICE" \
  --region "$REGION" \
  --format='value(status.url)')

echo "Branch : $BRANCH"
echo "Image  : $IMAGE"
echo "URL    : $URL"

echo ""
open "$URL" 2>/dev/null || true

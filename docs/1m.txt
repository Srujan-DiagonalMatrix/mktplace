PROJECT="tactical-innovation-f2b97b"
REGION="europe-west1"

DB_INSTANCE="rove-postgres"

# Complex password for bank policy
ROOT_PASSWORD='Rove#Pov2026!Db9'

echo "=================================================="
echo " DELETE PREVIOUS TEST INSTANCE IF PRESENT"
echo "=================================================="

if gcloud sql instances describe "$DB_INSTANCE" \
  --project="$PROJECT" >/dev/null 2>&1
then

  gcloud sql instances patch "$DB_INSTANCE" \
    --project="$PROJECT" \
    --no-deletion-protection \
    --quiet || true

  gcloud sql instances delete "$DB_INSTANCE" \
    --project="$PROJECT" \
    --no-enable-final-backup \
    --quiet
fi


echo
echo "=================================================="
echo " CREATE BANK-COMPLIANT POSTGRESQL"
echo "=================================================="

gcloud sql instances create "$DB_INSTANCE" \
  --project="$PROJECT" \
  --database-version=POSTGRES_16 \
  --edition=ENTERPRISE \
  --tier=db-f1-micro \
  --region="$REGION" \
  --storage-size=10GB \
  --storage-type=SSD \
  --availability-type=zonal \
  --ssl-mode=ENCRYPTED_ONLY \
  --database-flags=cloudsql.iam_authentication=on,pgaudit.log=all \
  --enable-password-policy \
  --password-policy-min-length=12 \
  --password-policy-complexity=COMPLEXITY_DEFAULT \
  --password-policy-disallow-username-substring \
  --root-password="$ROOT_PASSWORD" \
  --no-deletion-protection

echo
echo "=================================================="
echo " INSTANCE CREATED"
echo "=================================================="

gcloud sql instances describe "$DB_INSTANCE" \
  --project="$PROJECT" \
  --format="yaml(
    name,
    databaseVersion,
    region,
    settings.edition,
    settings.tier,
    settings.ipConfiguration.sslMode,
    settings.databaseFlags,
    settings.passwordValidationPolicy
  )"

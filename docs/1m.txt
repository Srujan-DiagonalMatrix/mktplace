
# ============================================================

# ============================================================

#!/usr/bin/env bash
set -euo pipefail

# ============================================================
# ROVE - BRAND NEW GCP INSTALLATION
# FIRST INSTALLATION ONLY
# ============================================================

PROJECT_DIR="/Users/srujan.kumar/Documents/git/rove"

PROJECT="tactical-innovation-f2b97b"
PROJECT_NUMBER="446065888450"

REGION="europe-west1"

REPOSITORY="cloud-run-source-deploy"

API_SERVICE="rove-api"
WEB_SERVICE="rove-web"

API_IMAGE="rove-api"
WEB_IMAGE="rove-web"

DB_INSTANCE="rove-postgres"
DB_NAME="rove"
DB_USER="postgres"

DB_PASSWORD='RovePov#2026Secure!'

SERVICE_ACCOUNT="${PROJECT_NUMBER}-compute@developer.gserviceaccount.com"

DEPLOYMENT_ID="$(date +%Y%m%d-%H%M%S)"

WEB_BUILD_DIR="${PROJECT_DIR}/.rove-gcp-web-build"

cd "$PROJECT_DIR"


echo
echo "============================================================"
echo " ROVE BRAND NEW GCP INSTALLATION"
echo "============================================================"
echo
echo "Project       : $PROJECT"
echo "Region        : $REGION"
echo "Deployment ID : $DEPLOYMENT_ID"
echo


# ============================================================
# PRECHECKS
# ============================================================

command -v gcloud >/dev/null || {
  echo "ERROR: gcloud not installed"
  exit 1
}

command -v docker >/dev/null || {
  echo "ERROR: docker not installed"
  exit 1
}

command -v npm >/dev/null || {
  echo "ERROR: npm not installed"
  exit 1
}

command -v python3 >/dev/null || {
  echo "ERROR: python3 not installed"
  exit 1
}

docker info >/dev/null 2>&1 || {
  echo "ERROR: Docker Desktop not running"
  exit 1
}


# ============================================================
# FIX ALEMBIC % INTERPOLATION
# ============================================================

echo
echo "============================================================"
echo "[0/10] VERIFY ALEMBIC CONFIG"
echo "============================================================"

python3 <<'PY'
from pathlib import Path

path = Path("server/alembic/env.py")

if not path.exists():
    raise SystemExit("ERROR: server/alembic/env.py not found")

text = path.read_text()

old = 'config.set_main_option("sqlalchemy.url", database_url)'

new = '''config.set_main_option(
        "sqlalchemy.url",
        database_url.replace("%", "%%")
    )'''

if 'database_url.replace("%", "%%")' in text:
    print("Alembic interpolation fix already present.")

elif old in text:
    path.write_text(text.replace(old, new))
    print("Alembic interpolation fix applied.")

else:
    raise SystemExit(
        "ERROR: Unable to locate Alembic sqlalchemy.url configuration."
    )
PY


# ============================================================
# 1. AUTH + APIs
# ============================================================

echo
echo "============================================================"
echo "[1/10] ENABLE REQUIRED GOOGLE APIs"
echo "============================================================"

if ! gcloud auth print-access-token >/dev/null 2>&1
then
  gcloud auth login
fi

gcloud config set project "$PROJECT" >/dev/null

gcloud services enable \
  run.googleapis.com \
  sqladmin.googleapis.com \
  sql-component.googleapis.com \
  artifactregistry.googleapis.com \
  iam.googleapis.com \
  serviceusage.googleapis.com \
  --project="$PROJECT" \
  --quiet

echo "Required APIs enabled."


# ============================================================
# 2. ARTIFACT REGISTRY
# ============================================================

echo
echo "============================================================"
echo "[2/10] CREATE ARTIFACT REGISTRY"
echo "============================================================"

gcloud artifacts repositories create "$REPOSITORY" \
  --project="$PROJECT" \
  --location="$REGION" \
  --repository-format=docker \
  --description="ROVE container images" \
  --quiet

gcloud auth configure-docker \
  "${REGION}-docker.pkg.dev" \
  --quiet >/dev/null

echo "Artifact Registry ready."


# ============================================================
# 3. POSTGRESQL
# ============================================================

echo
echo "============================================================"
echo "[3/10] CREATE POSTGRESQL"
echo "============================================================"

gcloud sql instances create "$DB_INSTANCE" \
  --project="$PROJECT" \
  --database-version=POSTGRES_16 \
  --edition=ENTERPRISE \
  --tier=db-f1-micro \
  --region="$REGION" \
  --storage-size=10GB \
  --storage-type=SSD \
  --availability-type=zonal \
  --ssl-mode=ENCRYPTED_ONLY \
  --database-flags=cloudsql.iam_authentication=on,cloudsql.enable_pgaudit=on,pgaudit.log=all \
  --enable-password-policy \
  --password-policy-min-length=12 \
  --password-policy-complexity=COMPLEXITY_DEFAULT \
  --password-policy-disallow-username-substring \
  --root-password="$DB_PASSWORD" \
  --no-deletion-protection \
  --quiet


echo
echo "Waiting for PostgreSQL..."

for i in {1..40}
do
  DB_STATE="$(gcloud sql instances describe "$DB_INSTANCE" \
    --project="$PROJECT" \
    --format='value(state)' 2>/dev/null || true)"

  echo "State: $DB_STATE"

  if [ "$DB_STATE" = "RUNNABLE" ]; then
    break
  fi

  sleep 5
done

if [ "$DB_STATE" != "RUNNABLE" ]; then
  echo "ERROR: PostgreSQL did not become RUNNABLE."
  exit 1
fi


# ============================================================
# 4. DATABASE
# ============================================================

echo
echo "============================================================"
echo "[4/10] CREATE ROVE DATABASE"
echo "============================================================"

gcloud sql databases create "$DB_NAME" \
  --instance="$DB_INSTANCE" \
  --project="$PROJECT" \
  --quiet

gcloud sql users set-password "$DB_USER" \
  --instance="$DB_INSTANCE" \
  --project="$PROJECT" \
  --password="$DB_PASSWORD" \
  --quiet


CONNECTION_NAME="$(gcloud sql instances describe "$DB_INSTANCE" \
  --project="$PROJECT" \
  --format='value(connectionName)')"

echo
echo "Connection:"
echo "$CONNECTION_NAME"


# ============================================================
# URL ENCODE PASSWORD
# ============================================================

DB_PASSWORD_URL="$(
  DB_PASSWORD="$DB_PASSWORD" python3 <<'PY'
import os
from urllib.parse import quote
print(quote(os.environ["DB_PASSWORD"], safe=""))
PY
)"

DATABASE_URL="postgresql+psycopg://${DB_USER}:${DB_PASSWORD_URL}@/${DB_NAME}?host=/cloudsql/${CONNECTION_NAME}"


# ============================================================
# 5. SERVICE ACCOUNT
# ============================================================

echo
echo "============================================================"
echo "[5/10] CONFIGURE CLOUD SQL ACCESS"
echo "============================================================"

echo "Service account:"
echo "$SERVICE_ACCOUNT"

gcloud projects add-iam-policy-binding "$PROJECT" \
  --member="serviceAccount:${SERVICE_ACCOUNT}" \
  --role="roles/cloudsql.client" \
  --condition=None \
  --quiet

echo "Cloud SQL Client permission configured."


# ============================================================
# 6. BUILD API
# ============================================================

echo
echo "============================================================"
echo "[6/10] BUILD API"
echo "============================================================"

cat > server/Dockerfile.gcp <<'DOCKERFILE'
FROM python:3.12-slim

WORKDIR /app

ENV PYTHONDONTWRITEBYTECODE=1
ENV PYTHONUNBUFFERED=1

COPY requirements.txt .

RUN pip install \
  --no-cache-dir \
  --trusted-host pypi.org \
  --trusted-host files.pythonhosted.org \
  --trusted-host pypi.python.org \
  -r requirements.txt

COPY app ./app
COPY scripts ./scripts
COPY alembic ./alembic
COPY alembic.ini ./alembic.ini

EXPOSE 8080

CMD ["sh", "-c", "alembic upgrade head && exec uvicorn app.main:app --host 0.0.0.0 --port ${PORT:-8080}"]
DOCKERFILE


API_VERSION_IMAGE="${REGION}-docker.pkg.dev/${PROJECT}/${REPOSITORY}/${API_IMAGE}:${DEPLOYMENT_ID}"

echo
echo "Building API:"
echo "$API_VERSION_IMAGE"

docker buildx build \
  --platform linux/amd64 \
  -f server/Dockerfile.gcp \
  -t "$API_VERSION_IMAGE" \
  --push \
  server


# ============================================================
# 7. DEPLOY API
# ============================================================

echo
echo "============================================================"
echo "[7/10] DEPLOY API"
echo "============================================================"

gcloud run deploy "$API_SERVICE" \
  --project="$PROJECT" \
  --region="$REGION" \
  --platform=managed \
  --execution-environment=gen2 \
  --image="$API_VERSION_IMAGE" \
  --port=8080 \
  --timeout=300 \
  --memory=512Mi \
  --cpu=1 \
  --min=0 \
  --max=2 \
  --add-cloudsql-instances="$CONNECTION_NAME" \
  --service-account="$SERVICE_ACCOUNT" \
  --set-env-vars="DATABASE_URL=${DATABASE_URL},APP_ENV=production,DEPLOYMENT_ID=${DEPLOYMENT_ID}" \
  --no-invoker-iam-check \
  --quiet


API_URL="$(gcloud run services describe "$API_SERVICE" \
  --project="$PROJECT" \
  --region="$REGION" \
  --format='value(status.url)')"


echo
echo "API URL:"
echo "$API_URL"


echo
echo "Testing API..."

sleep 5

API_STATUS="$(curl \
  --silent \
  --output /tmp/rove-api-response.html \
  --write-out "%{http_code}" \
  "$API_URL/docs" || true)"

echo "FastAPI HTTP: $API_STATUS"


if [ "$API_STATUS" != "200" ]; then

  echo
  echo "ERROR: API startup failed."
  echo
  echo "Recent logs:"
  echo

  gcloud logging read \
    "resource.type=cloud_run_revision AND resource.labels.service_name=${API_SERVICE}" \
    --project="$PROJECT" \
    --freshness=30m \
    --limit=100 \
    --order=asc \
    --format="value(timestamp,severity,textPayload)"

  exit 1
fi


echo
echo "Database health:"
curl --silent "$API_URL/api/health"
echo


# ============================================================
# 8. BUILD REACT
# ============================================================

echo
echo "============================================================"
echo "[8/10] BUILD REACT"
echo "============================================================"

rm -rf dist

NODE_USE_SYSTEM_CA=1 npm ci
NODE_USE_SYSTEM_CA=1 npm run build

if [ ! -f dist/index.html ]; then
  echo "ERROR: React build failed."
  exit 1
fi

echo "React build successful."


# ============================================================
# 9. BUILD WEB IMAGE
# ============================================================

echo
echo "============================================================"
echo "[9/10] BUILD WEB IMAGE"
echo "============================================================"

rm -rf "$WEB_BUILD_DIR"

mkdir -p "$WEB_BUILD_DIR"

cp -R dist "$WEB_BUILD_DIR/dist"


API_HOST="$(echo "$API_URL" \
  | sed 's#^https://##' \
  | sed 's#/$##')"


cat > "$WEB_BUILD_DIR/nginx.conf" <<NGINX
server {

    listen 8080;
    server_name _;

    root /usr/share/nginx/html;
    index index.html;

    location /api/ {

        proxy_ssl_server_name on;
        proxy_ssl_name ${API_HOST};

        proxy_set_header Host ${API_HOST};
        proxy_set_header X-Real-IP \$remote_addr;
        proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto https;

        proxy_connect_timeout 30s;
        proxy_read_timeout 120s;

        proxy_pass ${API_URL}/api/;
    }

    location / {
        try_files \$uri \$uri/ /index.html;
    }
}
NGINX


cat > "$WEB_BUILD_DIR/Dockerfile" <<'DOCKERFILE'
FROM nginx:alpine

COPY nginx.conf /etc/nginx/conf.d/default.conf
COPY dist /usr/share/nginx/html

EXPOSE 8080

CMD ["nginx", "-g", "daemon off;"]
DOCKERFILE


WEB_VERSION_IMAGE="${REGION}-docker.pkg.dev/${PROJECT}/${REPOSITORY}/${WEB_IMAGE}:${DEPLOYMENT_ID}"

echo
echo "Building Web:"
echo "$WEB_VERSION_IMAGE"

docker buildx build \
  --platform linux/amd64 \
  -t "$WEB_VERSION_IMAGE" \
  --push \
  "$WEB_BUILD_DIR"


# ============================================================
# 10. DEPLOY WEB
# ============================================================

echo
echo "============================================================"
echo "[10/10] DEPLOY WEB"
echo "============================================================"

gcloud run deploy "$WEB_SERVICE" \
  --project="$PROJECT" \
  --region="$REGION" \
  --platform=managed \
  --execution-environment=gen2 \
  --image="$WEB_VERSION_IMAGE" \
  --port=8080 \
  --memory=256Mi \
  --cpu=1 \
  --min=0 \
  --max=2 \
  --set-env-vars="DEPLOYMENT_ID=${DEPLOYMENT_ID}" \
  --no-invoker-iam-check \
  --quiet


WEB_URL="$(gcloud run services describe "$WEB_SERVICE" \
  --project="$PROJECT" \
  --region="$REGION" \
  --format='value(status.url)')"


echo
echo "============================================================"
echo " FINAL TESTS"
echo "============================================================"

echo
echo "API:"
curl --silent \
  --output /dev/null \
  --write-out "HTTP %{http_code}\n" \
  "$API_URL/docs"

echo
echo "Database:"
curl --silent "$API_URL/api/health"
echo

echo
echo "Website:"
curl --silent \
  --output /dev/null \
  --write-out "HTTP %{http_code}\n" \
  "$WEB_URL/"

echo
echo "API through Web:"
curl --silent "$WEB_URL/api/health"
echo


echo
echo "============================================================"
echo " ROVE INSTALLATION SUCCESSFUL"
echo "============================================================"
echo
echo "Database:"
echo "$DB_INSTANCE"
echo
echo "API:"
echo "$API_URL"
echo
echo "Website:"
echo "$WEB_URL"
echo
echo "============================================================"

rm -f server/Dockerfile.gcp
rm -rf "$WEB_BUILD_DIR"

open "$WEB_URL" 2>/dev/null || true

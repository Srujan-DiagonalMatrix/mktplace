install-rove-gcp.sh

#!/usr/bin/env bash
set -euo pipefail

# ============================================================
# ROVE - BRAND NEW GCP INSTALLATION
#
# FIRST INSTALLATION ONLY.
#
# Creates:
#   - Required APIs
#   - Artifact Registry
#   - PostgreSQL Cloud SQL
#   - rove database
#   - API container
#   - rove-api Cloud Run service
#   - React build
#   - Web container
#   - rove-web Cloud Run service
#
# IMPORTANT:
#   Run cleanup-rove-gcp.sh before this script when performing
#   a completely fresh installation.
# ============================================================


# ============================================================
# CONFIGURATION
# ============================================================

PROJECT_DIR="/Users/srujan.kumar/Documents/git/rove"

PROJECT="tactical-innovation-f2b97b"
PROJECT_NUMBER="446065888450"

REGION="europe-west1"

REPOSITORY="cloud-run-source-deploy"

API_SERVICE="rove-api"
WEB_SERVICE="rove-web"

API_IMAGE="rove-api"
WEB_IMAGE="rove-web"

DB_INSTANCE="rove-postgres"
DB_NAME="rove"
DB_USER="postgres"

# Bank password policy:
# > 12 characters + upper/lower/numeric/special
DB_PASSWORD='RovePov#2026Secure!'

SERVICE_ACCOUNT="${PROJECT_NUMBER}-compute@developer.gserviceaccount.com"

DEPLOYMENT_ID="$(date +%Y%m%d-%H%M%S)"

API_DOCKERFILE="server/Dockerfile.gcp"

WEB_BUILD_DIR="${PROJECT_DIR}/.rove-gcp-web-build"


cd "$PROJECT_DIR"


# ============================================================
# CLEANUP TEMP FILES ON EXIT
# ============================================================

cleanup_temp_files() {

    rm -f "$API_DOCKERFILE"
    rm -rf "$WEB_BUILD_DIR"

}

trap cleanup_temp_files EXIT


# ============================================================
# HEADER
# ============================================================

echo
echo "============================================================"
echo " ROVE - BRAND NEW GCP INSTALLATION"
echo "============================================================"
echo
echo "Project       : $PROJECT"
echo "Region        : $REGION"
echo "Deployment ID : $DEPLOYMENT_ID"
echo
echo "============================================================"


# ============================================================
# PRE-CHECKS
# ============================================================

command -v gcloud >/dev/null 2>&1 || {
    echo "ERROR: gcloud is not installed."
    exit 1
}

command -v docker >/dev/null 2>&1 || {
    echo "ERROR: Docker is not installed."
    exit 1
}

command -v npm >/dev/null 2>&1 || {
    echo "ERROR: npm is not installed."
    exit 1
}

command -v python3 >/dev/null 2>&1 || {
    echo "ERROR: python3 is not installed."
    exit 1
}

command -v curl >/dev/null 2>&1 || {
    echo "ERROR: curl is not installed."
    exit 1
}


docker info >/dev/null 2>&1 || {
    echo "ERROR: Docker Desktop is not running."
    exit 1
}


# Remove stale local Node certificate reference if file is missing.

if [ -n "${NODE_EXTRA_CA_CERTS:-}" ] && \
   [ ! -f "${NODE_EXTRA_CA_CERTS}" ]
then

    unset NODE_EXTRA_CA_CERTS

fi


# ============================================================
# 1. AUTHENTICATION + REQUIRED APIs
# ============================================================

echo
echo "============================================================"
echo "[1/10] GCP AUTHENTICATION + APIs"
echo "============================================================"


if ! gcloud auth print-access-token >/dev/null 2>&1
then

    gcloud auth login

fi


gcloud config set project "$PROJECT" >/dev/null


echo
echo "Active account:"

gcloud auth list \
    --filter=status:ACTIVE \
    --format="value(account)"


echo
echo "Enabling required APIs..."


# IMPORTANT:
#
# sql-component.googleapis.com is explicitly included.
#
# Previous deployment failed because gcloud run deploy attempted
# to prompt for this API. --quiet converted that prompt into
# "Aborted by user".

gcloud services enable \
    run.googleapis.com \
    sqladmin.googleapis.com \
    sql-component.googleapis.com \
    artifactregistry.googleapis.com \
    iam.googleapis.com \
    serviceusage.googleapis.com \
    --project="$PROJECT" \
    --quiet


echo
echo "Required APIs enabled."


# ============================================================
# 2. CREATE ARTIFACT REGISTRY
# ============================================================

echo
echo "============================================================"
echo "[2/10] CREATE ARTIFACT REGISTRY"
echo "============================================================"


gcloud artifacts repositories create "$REPOSITORY" \
    --project="$PROJECT" \
    --location="$REGION" \
    --repository-format=docker \
    --description="ROVE Docker images" \
    --quiet


gcloud auth configure-docker \
    "${REGION}-docker.pkg.dev" \
    --quiet >/dev/null


echo "Artifact Registry created."


# ============================================================
# 3. CREATE POSTGRESQL
# ============================================================

echo
echo "============================================================"
echo "[3/10] CREATE POSTGRESQL"
echo "============================================================"


echo "Creating bank-compliant PostgreSQL..."


gcloud sql instances create "$DB_INSTANCE" \
    --project="$PROJECT" \
    --database-version=POSTGRES_16 \
    --edition=ENTERPRISE \
    --tier=db-f1-micro \
    --region="$REGION" \
    --storage-size=10GB \
    --storage-type=SSD \
    --availability-type=zonal \
    --ssl-mode=ENCRYPTED_ONLY \
    --database-flags=cloudsql.iam_authentication=on,cloudsql.enable_pgaudit=on,pgaudit.log=all \
    --enable-password-policy \
    --password-policy-min-length=12 \
    --password-policy-complexity=COMPLEXITY_DEFAULT \
    --password-policy-disallow-username-substring \
    --root-password="$DB_PASSWORD" \
    --no-deletion-protection \
    --quiet


echo
echo "PostgreSQL created."


# ============================================================
# 4. CREATE ROVE DATABASE
# ============================================================

echo
echo "============================================================"
echo "[4/10] CREATE ROVE DATABASE"
echo "============================================================"


gcloud sql databases create "$DB_NAME" \
    --instance="$DB_INSTANCE" \
    --project="$PROJECT" \
    --quiet


echo
echo "Synchronising postgres password..."


gcloud sql users set-password "$DB_USER" \
    --instance="$DB_INSTANCE" \
    --project="$PROJECT" \
    --password="$DB_PASSWORD" \
    --quiet


CONNECTION_NAME="$(gcloud sql instances describe "$DB_INSTANCE" \
    --project="$PROJECT" \
    --format='value(connectionName)')"


echo
echo "Cloud SQL connection:"
echo "$CONNECTION_NAME"


DB_STATE="$(gcloud sql instances describe "$DB_INSTANCE" \
    --project="$PROJECT" \
    --format='value(state)')"


if [ "$DB_STATE" != "RUNNABLE" ]
then

    echo "ERROR: PostgreSQL state is $DB_STATE"
    exit 1

fi


# ============================================================
# 5. CLOUD RUN SERVICE ACCOUNT
# ============================================================

echo
echo "============================================================"
echo "[5/10] CONFIGURE CLOUD SQL ACCESS"
echo "============================================================"


echo "Cloud Run service account:"
echo "$SERVICE_ACCOUNT"


echo
echo "Granting Cloud SQL Client..."


gcloud projects add-iam-policy-binding "$PROJECT" \
    --member="serviceAccount:${SERVICE_ACCOUNT}" \
    --role="roles/cloudsql.client" \
    --condition=None \
    --quiet


# ============================================================
# DATABASE URL
# ============================================================

DB_PASSWORD_URL="$(
    DB_PASSWORD="$DB_PASSWORD" python3 <<'PY'
import os
from urllib.parse import quote

print(quote(os.environ["DB_PASSWORD"], safe=""))
PY
)"


DATABASE_URL="postgresql+psycopg://${DB_USER}:${DB_PASSWORD_URL}@/${DB_NAME}?host=/cloudsql/${CONNECTION_NAME}"


# ============================================================
# 6. BUILD API
# ============================================================

echo
echo "============================================================"
echo "[6/10] BUILD API"
echo "============================================================"


cat > "$API_DOCKERFILE" <<'EOF'
FROM python:3.12-slim

WORKDIR /app

ENV PYTHONDONTWRITEBYTECODE=1
ENV PYTHONUNBUFFERED=1


COPY requirements.txt .


RUN pip install \
    --no-cache-dir \
    --trusted-host pypi.org \
    --trusted-host files.pythonhosted.org \
    --trusted-host pypi.python.org \
    -r requirements.txt


COPY app ./app
COPY scripts ./scripts
COPY alembic ./alembic
COPY alembic.ini ./alembic.ini


EXPOSE 8080


CMD ["sh", "-c", "alembic upgrade head && exec uvicorn app.main:app --host 0.0.0.0 --port ${PORT:-8080}"]
EOF


API_VERSION_IMAGE="${REGION}-docker.pkg.dev/${PROJECT}/${REPOSITORY}/${API_IMAGE}:${DEPLOYMENT_ID}"


echo
echo "Building:"
echo "$API_VERSION_IMAGE"


docker buildx build \
    --platform linux/amd64 \
    -f "$API_DOCKERFILE" \
    -t "$API_VERSION_IMAGE" \
    --push \
    server


echo
echo "API image built."


# ============================================================
# 7. DEPLOY API
# ============================================================

echo
echo "============================================================"
echo "[7/10] DEPLOY API"
echo "============================================================"


# sql-component.googleapis.com was already explicitly enabled
# at STEP 1. Therefore gcloud will NOT stop for the hidden
# Cloud SQL API enablement prompt that previously caused:
#
#   ERROR: (gcloud.run.deploy) Aborted by user


gcloud run deploy "$API_SERVICE" \
    --project="$PROJECT" \
    --region="$REGION" \
    --platform=managed \
    --execution-environment=gen2 \
    --image="$API_VERSION_IMAGE" \
    --port=8080 \
    --timeout=300 \
    --memory=512Mi \
    --cpu=1 \
    --min=0 \
    --max=2 \
    --add-cloudsql-instances="$CONNECTION_NAME" \
    --service-account="$SERVICE_ACCOUNT" \
    --set-env-vars="DATABASE_URL=${DATABASE_URL},APP_ENV=production,DEPLOYMENT_ID=${DEPLOYMENT_ID}" \
    --no-invoker-iam-check \
    --quiet


API_URL="$(gcloud run services describe "$API_SERVICE" \
    --project="$PROJECT" \
    --region="$REGION" \
    --format='value(status.url)')"


if [ -z "$API_URL" ]
then

    echo "ERROR: rove-api URL not returned."
    exit 1

fi


echo
echo "API URL:"
echo "$API_URL"


# ============================================================
# TEST API
# ============================================================

echo
echo "Testing API..."


sleep 5


API_STATUS="$(curl \
    --silent \
    --output /tmp/rove-api-response.html \
    --write-out "%{http_code}" \
    "$API_URL/docs" || true)"


echo "FastAPI /docs HTTP: $API_STATUS"


if [ "$API_STATUS" != "200" ]
then

    echo
    echo "ERROR: API did not start correctly."
    echo
    echo "Cloud Run logs:"
    echo


    gcloud logging read \
        "resource.type=cloud_run_revision AND resource.labels.service_name=${API_SERVICE}" \
        --project="$PROJECT" \
        --freshness=30m \
        --limit=100 \
        --format="value(timestamp,severity,textPayload)"


    exit 1

fi


echo
echo "Testing database health..."


curl \
    --silent \
    "$API_URL/api/health"


echo
echo
echo "API and PostgreSQL are working."


# ============================================================
# 8. BUILD REACT
# ============================================================

echo
echo "============================================================"
echo "[8/10] BUILD REACT"
echo "============================================================"


rm -rf dist


NODE_USE_SYSTEM_CA=1 npm ci

NODE_USE_SYSTEM_CA=1 npm run build


if [ ! -f "dist/index.html" ]
then

    echo "ERROR: React build failed."
    exit 1

fi


echo "React build successful."


# ============================================================
# 9. BUILD WEB IMAGE
# ============================================================

echo
echo "============================================================"
echo "[9/10] BUILD WEB IMAGE"
echo "============================================================"


rm -rf "$WEB_BUILD_DIR"

mkdir -p "$WEB_BUILD_DIR"


cp -R dist "$WEB_BUILD_DIR/dist"


API_HOST="$(echo "$API_URL" \
    | sed 's#^https://##' \
    | sed 's#/$##')"


cat > "$WEB_BUILD_DIR/nginx.conf" <<EOF
server {

    listen 8080;

    server_name _;

    root /usr/share/nginx/html;

    index index.html;


    location /api/ {

        proxy_ssl_server_name on;

        proxy_ssl_name ${API_HOST};

        proxy_set_header Host ${API_HOST};

        proxy_set_header X-Real-IP \$remote_addr;

        proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;

        proxy_set_header X-Forwarded-Proto https;

        proxy_connect_timeout 30s;

        proxy_read_timeout 120s;

        proxy_pass ${API_URL}/api/;

    }


    location / {

        try_files \$uri \$uri/ /index.html;

    }

}
EOF


cat > "$WEB_BUILD_DIR/Dockerfile" <<'EOF'
FROM nginx:alpine

COPY nginx.conf /etc/nginx/conf.d/default.conf

COPY dist /usr/share/nginx/html

EXPOSE 8080

CMD ["nginx", "-g", "daemon off;"]
EOF


WEB_VERSION_IMAGE="${REGION}-docker.pkg.dev/${PROJECT}/${REPOSITORY}/${WEB_IMAGE}:${DEPLOYMENT_ID}"


echo
echo "Building:"
echo "$WEB_VERSION_IMAGE"


docker buildx build \
    --platform linux/amd64 \
    -t "$WEB_VERSION_IMAGE" \
    --push \
    "$WEB_BUILD_DIR"


echo "Web image built."


# ============================================================
# 10. DEPLOY WEB
# ============================================================

echo
echo "============================================================"
echo "[10/10] DEPLOY WEB"
echo "============================================================"


gcloud run deploy "$WEB_SERVICE" \
    --project="$PROJECT" \
    --region="$REGION" \
    --platform=managed \
    --execution-environment=gen2 \
    --image="$WEB_VERSION_IMAGE" \
    --port=8080 \
    --memory=256Mi \
    --cpu=1 \
    --min=0 \
    --max=2 \
    --set-env-vars="DEPLOYMENT_ID=${DEPLOYMENT_ID}" \
    --no-invoker-iam-check \
    --quiet


WEB_URL="$(gcloud run services describe "$WEB_SERVICE" \
    --project="$PROJECT" \
    --region="$REGION" \
    --format='value(status.url)')"


if [ -z "$WEB_URL" ]
then

    echo "ERROR: rove-web URL not returned."
    exit 1

fi


# ============================================================
# FINAL TEST
# ============================================================

echo
echo "============================================================"
echo " FINAL TESTS"
echo "============================================================"


echo
echo "API:"
curl \
    --silent \
    --output /dev/null \
    --write-out "HTTP %{http_code}\n" \
    "$API_URL/docs"


echo
echo "Database:"
curl \
    --silent \
    "$API_URL/api/health"


echo
echo
echo "Website:"
curl \
    --silent \
    --output /dev/null \
    --write-out "HTTP %{http_code}\n" \
    "$WEB_URL/"


echo
echo "API through website:"
curl \
    --silent \
    "$WEB_URL/api/health"


echo
echo
echo "============================================================"
echo " ROVE FIRST INSTALLATION SUCCESSFUL"
echo "============================================================"
echo
echo "PostgreSQL:"
echo "$DB_INSTANCE"
echo
echo "Database:"
echo "$DB_NAME"
echo
echo "API:"
echo "$API_URL"
echo
echo "Website:"
echo "$WEB_URL"
echo
echo "API Image:"
echo "$API_VERSION_IMAGE"
echo
echo "Web Image:"
echo "$WEB_VERSION_IMAGE"
echo
echo "============================================================"


open "$WEB_URL" 2>/dev/null || true

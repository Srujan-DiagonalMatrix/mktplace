python3 <<'PY'
from pathlib import Path

path = Path("server/alembic/env.py")

if not path.exists():
    raise SystemExit("ERROR: server/alembic/env.py not found")

text = path.read_text()

old = 'config.set_main_option("sqlalchemy.url", database_url)'

new = '''config.set_main_option(
    "sqlalchemy.url",
    database_url.replace("%", "%%")
)'''

if old in text:
    path.write_text(text.replace(old, new))
    print("Alembic env.py updated successfully.")

elif 'database_url.replace("%", "%%")' in text:
    print("Alembic env.py is already fixed.")

else:
    print("ERROR: Expected Alembic configuration line was not found.")
    print()
    print("Current server/alembic/env.py:")
    print("------------------------------------------------------------")
    print(text)
    raise SystemExit(1)
PY

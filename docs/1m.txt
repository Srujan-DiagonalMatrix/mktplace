#!/usr/bin/env bash
set -Eeuo pipefail
set +x
umask 077

export GCP_PROJECT="${GCP_PROJECT:-tactical-innovation-f2b97b}"
REGION="${REGION:-europe-west1}"
REPOSITORY="${REPOSITORY:-cloud-run-source-deploy}"

WEB_SERVICE="${WEB_SERVICE:-rove}"
API_SERVICE="${API_SERVICE:-rove-api}"
BRANCH="${BRANCH:-main}"
REMOTE="${REMOTE:-origin}"

export CLOUD_SQL_CONNECTION="${CLOUD_SQL_CONNECTION:-tactical-innovation-f2b97b:europe-west1:rove-postgres}"

export DATABASE_URL_SECRET="${DATABASE_URL_SECRET:-rove-database-url}"
export ADMIN_USERNAME_SECRET="${ADMIN_USERNAME_SECRET:-rove-admin-username}"
export ADMIN_PASSWORD_SECRET="${ADMIN_PASSWORD_SECRET:-rove-admin-password}"

export DATABASE_URL_SECRET="${DATABASE_URL_SECRET%%:*}"
export ADMIN_USERNAME_SECRET="${ADMIN_USERNAME_SECRET%%:*}"
export ADMIN_PASSWORD_SECRET="${ADMIN_PASSWORD_SECRET%%:*}"

export ALLOW_DB_PASSWORD_ROTATION="${ALLOW_DB_PASSWORD_ROTATION:-0}"
export DB_NAME="${DB_NAME:-}"
export DB_USER="${DB_USER:-}"
export EXTRA_SECRETS="${EXTRA_SECRETS:-}"

# Optional:
# FORCE_API_BUILD=1
# API_ENV_FILE=/absolute/path/non-secret-settings.json
# EXTRA_SECRETS='SMTP_PASSWORD=existing-secret:latest'
# CORPORATE_CA_FILE=/absolute/path/company-ca.pem
# REPO_DIR=/absolute/path/rove
# ROVE_CREDENTIALS_DIR=/private/directory
#
# Frontend is always rebuilt from the selected remote branch.
# Existing API image is reused where available.
# Existing database is retained.
# Normal Alembic migrations run on API startup.

fail() {
  echo "ERROR: $*" >&2
  exit 1
}

gc() {
  gcloud --project="$GCP_PROJECT" "$@"
}

for tool in git node npm docker gcloud python3 curl tar; do
  command -v "$tool" >/dev/null || fail "Missing command: $tool"
done

docker info >/dev/null
docker buildx version >/dev/null

script_dir=$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)
root=$(git -C "${REPO_DIR:-$script_dir}" rev-parse --show-toplevel)

work=$(mktemp -d)
chmod 700 "$work"

trap 'rm -rf "$work"' EXIT
trap 'echo "Stopped at line $LINENO. Earlier cloud changes may remain. No database reset was performed." >&2' ERR

echo "=== Verify existing database ==="

gc sql instances describe "${CLOUD_SQL_CONNECTION##*:}" \
  --format=json > "$work/sql.json"

python3 - "$work/sql.json" <<'PY'
import json
import os
import sys

instance = json.load(open(sys.argv[1]))
connection = os.environ["CLOUD_SQL_CONNECTION"]

if (
    instance.get("connectionName") != connection
    or instance.get("state") != "RUNNABLE"
    or not instance.get("databaseVersion", "").startswith("POSTGRES")
    or connection.split(":")[0] != os.environ["GCP_PROJECT"]
):
    raise SystemExit(
        "Existing runnable PostgreSQL instance in GCP_PROJECT is required."
    )

if not any(
    address.get("type") == "PRIMARY"
    for address in instance.get("ipAddresses", [])
):
    raise SystemExit(
        "Private-IP-only SQL requires the original VPC configuration."
    )
PY

echo "=== Fetch latest remote source and build fresh UI ==="

git -C "$root" check-ref-format --branch "$BRANCH" >/dev/null
git -C "$root" fetch "$REMOTE" "refs/heads/$BRANCH"

commit=$(git -C "$root" rev-parse FETCH_HEAD)

mkdir "$work/source"

git -C "$root" archive "$commit" |
  tar -x -C "$work/source"

cd "$work/source"

echo "Frontend source: $BRANCH at $commit"

NODE_USE_SYSTEM_CA=1 npm ci --include=dev

mkdir -p "$work/web/dist"

NODE_USE_SYSTEM_CA=1 VITE_API_URL='' npm run build \
  -- --outDir "$work/web/dist" --emptyOutDir

test -s "$work/web/dist/index.html"

echo "=== Ensure deployment resources exist ==="

gc services enable \
  run.googleapis.com \
  artifactregistry.googleapis.com \
  secretmanager.googleapis.com \
  sqladmin.googleapis.com \
  iam.googleapis.com \
  --quiet

if ! gc artifacts repositories describe "$REPOSITORY" \
  --location="$REGION" >/dev/null 2>&1; then
  gc artifacts repositories create "$REPOSITORY" \
    --location="$REGION" \
    --repository-format=docker \
    --quiet
fi

[[ "$(gc artifacts repositories describe "$REPOSITORY" \
  --location="$REGION" --format='value(format)')" == DOCKER ]] ||
  fail "Repository must use Docker format."

for name in rove-api-runtime rove-web-runtime; do
  if ! gc iam service-accounts describe \
    "$name@$GCP_PROJECT.iam.gserviceaccount.com" >/dev/null 2>&1; then
    gc iam service-accounts create "$name" \
      --display-name="$name" \
      --quiet
  fi
done

api_account="rove-api-runtime@$GCP_PROJECT.iam.gserviceaccount.com"
web_account="rove-web-runtime@$GCP_PROJECT.iam.gserviceaccount.com"

gc projects add-iam-policy-binding "$GCP_PROJECT" \
  --member="serviceAccount:$api_account" \
  --role=roles/cloudsql.client \
  --condition=None \
  --quiet >/dev/null

echo "=== Configure credentials automatically ==="

# Generated credentials survive build-directory cleanup.
# Directory and files are accessible only to the current local user.
export ROVE_CREDENTIALS_DIR="${ROVE_CREDENTIALS_DIR:-$(mktemp -d "${TMPDIR:-/tmp}/rove-recovery-credentials.XXXXXXXX")}"

mkdir -p "$ROVE_CREDENTIALS_DIR"
chmod 700 "$ROVE_CREDENTIALS_DIR"

python3 - "$work" "${API_ENV_FILE:-}" <<'PY'
import json
import os
import re
import secrets
import subprocess
import sys
import time

from pathlib import Path
from urllib.parse import quote, urlencode
from urllib.request import Request, urlopen
from urllib.error import HTTPError, URLError

project = os.environ["GCP_PROJECT"]
connection = os.environ["CLOUD_SQL_CONNECTION"]
instance = connection.split(":")[2]

work = Path(sys.argv[1])
saved = Path(os.environ["ROVE_CREDENTIALS_DIR"])

def stop(message):
    raise SystemExit(message)

def cli(*arguments, optional=False):
    result = subprocess.run(
        ["gcloud", "--project=" + project, *arguments],
        capture_output=True,
        text=True,
    )
    if result.returncode and not optional:
        stop(
            "Google Cloud command failed: "
            + " ".join(arguments[:3])
            + ". Check account permissions and API access."
        )
    return result

def cloud_json(*arguments):
    return json.loads(cli(*arguments, "--format=json").stdout)

existing = {
    str(secret["name"]).split("/")[-1]
    for secret in cloud_json("secrets", "list")
}

def ready(name):
    if name not in existing:
        return False

    result = cli(
        "secrets", "versions", "describe", "latest",
        "--secret=" + name,
        "--format=value(state)",
        optional=True,
    )

    if result.returncode:
        stop(
            "Cannot inspect the latest version of existing secret "
            + name
            + ". Check its versions and permissions."
        )

    if result.stdout.strip() != "ENABLED":
        stop("Latest secret version is disabled: " + name)

    return True

def save_and_store(name, value):
    if not re.fullmatch(r"[A-Za-z0-9_-]+", name):
        stop("Invalid secret name.")

    path = saved / (name + ".txt")

    descriptor = os.open(
        path,
        os.O_WRONLY | os.O_CREAT | os.O_EXCL,
        0o600,
    )

    with os.fdopen(descriptor, "w") as file:
        file.write(value)

    if name in existing:
        cli(
            "secrets", "versions", "add", name,
            "--data-file=" + str(path),
            "--quiet",
        )
    else:
        cli(
            "secrets", "create", name,
            "--replication-policy=automatic",
            "--data-file=" + str(path),
            "--quiet",
        )
        existing.add(name)

database_secret = os.environ["DATABASE_URL_SECRET"]

if not ready(database_secret):
    if os.environ["ALLOW_DB_PASSWORD_ROTATION"] != "1":
        stop(
            "Cloud SQL cannot reveal an existing password. "
            "Re-run with ALLOW_DB_PASSWORD_ROTATION=1 to explicitly "
            "authorize rotating the existing application user's password. "
            "Other clients using that user will need the new password."
        )

    databases = {
        database["name"]
        for database in cloud_json(
            "sql", "databases", "list", "--instance=" + instance
        )
        if database["name"] not in (
            "postgres", "template0", "template1", "cloudsqladmin"
        )
    }

    users = {
        user["name"]
        for user in cloud_json(
            "sql", "users", "list", "--instance=" + instance
        )
        if user.get("type", "BUILT_IN") == "BUILT_IN"
    }

    database = os.environ["DB_NAME"]

    if not database:
        if "rove" in databases:
            database = "rove"
        elif len(databases) == 1:
            database = next(iter(databases))

    username = os.environ["DB_USER"]

    if not username and "rove" in users:
        username = "rove"

    if database not in databases:
        stop(
            "Cannot choose the application database automatically. "
            "Set DB_NAME to its existing name."
        )

    if username not in users:
        stop(
            "Existing application user rove was not found. "
            "Set DB_USER explicitly to the existing application user. "
            "No users were changed."
        )

    print(
        "Rotating password for existing SQL user "
        + username
        + " on "
        + instance
        + "; database "
        + database
        + " is retained."
    )

    password = secrets.token_urlsafe(36)

    database_url = (
        "postgresql+psycopg://"
        + quote(username, safe="")
        + ":"
        + quote(password, safe="")
        + "@/"
        + quote(database, safe="")
        + "?host=/cloudsql/"
        + connection
    )

    # Save credentials before requesting the password change.
    recovery = saved / "database-recovery.json"

    descriptor = os.open(
        recovery,
        os.O_WRONLY | os.O_CREAT | os.O_EXCL,
        0o600,
    )

    with os.fdopen(descriptor, "w") as file:
        json.dump(
            {
                "username": username,
                "database": database,
                "password": password,
                "DATABASE_URL": database_url,
            },
            file,
        )

    token = cli("auth", "print-access-token").stdout.strip()

    base = (
        "https://sqladmin.googleapis.com/sql/v1beta4/projects/"
        + quote(project, safe="")
    )

    def sql_api(method, path, data=None):
        payload = json.dumps(data).encode() if data is not None else None

        request = Request(
            base + path,
            data=payload,
            method=method,
            headers={
                "Authorization": "Bearer " + token,
                "Content-Type": "application/json",
            },
        )

        try:
            with urlopen(request, timeout=60) as response:
                return json.load(response)
        except HTTPError as error:
            stop(
                "Cloud SQL API returned HTTP "
                + str(error.code)
                + ". Recovery credentials remain in the protected directory."
            )
        except URLError:
            stop(
                "Cloud SQL API connection failed. "
                "Recovery credentials remain in the protected directory."
            )

    operation = sql_api(
        "PUT",
        "/instances/"
        + quote(instance, safe="")
        + "/users?"
        + urlencode({"name": username}),
        {"name": username, "password": password},
    )

    deadline = time.monotonic() + 180

    while operation.get("status") != "DONE":
        if time.monotonic() > deadline:
            stop(
                "Password update is still pending. Check Cloud SQL operations "
                "before retrying. Recovery credentials were saved."
            )

        time.sleep(2)

        operation = sql_api(
            "GET",
            "/operations/" + quote(operation["name"], safe=""),
        )

    if operation.get("error"):
        stop(
            "Cloud SQL password update failed. Inspect its operation. "
            "Recovery credentials were saved."
        )

    save_and_store(database_secret, database_url)

else:
    print("Reusing existing database secret; SQL password unchanged.")

admin_username_secret = os.environ["ADMIN_USERNAME_SECRET"]
admin_password_secret = os.environ["ADMIN_PASSWORD_SECRET"]

if not ready(admin_username_secret):
    save_and_store(admin_username_secret, "admin")

if not ready(admin_password_secret):
    save_and_store(admin_password_secret, secrets.token_urlsafe(32))

references = {
    key: os.environ[key + "_SECRET"] + ":latest"
    for key in ("DATABASE_URL", "ADMIN_USERNAME", "ADMIN_PASSWORD")
}

for item in filter(None, os.environ["EXTRA_SECRETS"].split(",")):
    key, separator, value = item.partition("=")

    if not separator or key in references:
        stop("Invalid or duplicate EXTRA_SECRETS entry.")

    references[key] = value

for key, value in references.items():
    if (
        not re.fullmatch(r"[A-Z][A-Z0-9_]*", key)
        or not re.fullmatch(r"[A-Za-z0-9_-]+:(latest|[0-9]+)", value)
    ):
        stop("Invalid Secret Manager reference.")

(work / "secret-args").write_text(
    ",".join(key + "=" + value for key, value in references.items())
)

(work / "secret-refs").write_text(
    "".join(value + "\n" for value in sorted(set(references.values())))
)

settings = json.load(open(sys.argv[2])) if sys.argv[2] else {}

if (
    not isinstance(settings, dict)
    or any(not isinstance(value, str) for value in settings.values())
    or set(settings).intersection(references)
):
    stop("API_ENV_FILE must contain JSON string settings without secret keys.")

settings.update(
    APP_ENV="production",
    AUTH_COOKIE_SECURE="true",
    SESSION_COOKIE_SECURE="true",
)

(work / "env.json").write_text(json.dumps(settings))

print("Generated credentials, if any, are saved privately in: " + str(saved))
PY

while IFS=: read -r secret version; do
  [[ "$(gc secrets versions describe "$version" \
    --secret="$secret" --format='value(state)')" == ENABLED ]] ||
    fail "Secret is not enabled: $secret:$version"

  gc secrets add-iam-policy-binding "$secret" \
    --member="serviceAccount:$api_account" \
    --role=roles/secretmanager.secretAccessor \
    --condition=None \
    --quiet >/dev/null
done < "$work/secret-refs"

gc auth configure-docker "$REGION-docker.pkg.dev" --quiet

registry="$REGION-docker.pkg.dev/$GCP_PROJECT/$REPOSITORY"
tag="demo-$(date -u +%Y%m%d%H%M%S)-${commit:0:12}"

api_image="${API_IMAGE:-}"

if [[ -z "$api_image" && "${FORCE_API_BUILD:-0}" != 1 ]]; then
  gc artifacts docker images list "$registry" \
    --format=json > "$work/images.json"

  api_image=$(python3 - "$work/images.json" "$registry/rove-api" <<'PY'
import json
import re
import sys

rows = json.load(open(sys.argv[1]))

rows.sort(
    key=lambda row: row.get("updateTime", row.get("createTime", "")),
    reverse=True,
)

for row in rows:
    image = (
        row.get("package")
        or row.get("image")
        or row.get("uri")
        or ""
    )

    if str(image).split("@")[0] not in (
        sys.argv[2],
        "https://" + sys.argv[2],
    ):
        continue

    digest = re.search(r"sha256:[0-9a-f]{64}", json.dumps(row))

    if digest:
        print(sys.argv[2] + "@" + digest.group())
        break
PY
)
fi

if [[ -z "$api_image" || "${FORCE_API_BUILD:-0}" == 1 ]]; then
  api_image="$registry/rove-api:$tag"

  if [[ -n "${CORPORATE_CA_FILE:-}" ]]; then
    test -s "$CORPORATE_CA_FILE"

    docker buildx build \
      --platform linux/amd64 --pull --push \
      --secret "id=corporate_ca,src=$CORPORATE_CA_FILE" \
      -t "$api_image" -f server/Dockerfile server
  else
    docker buildx build \
      --platform linux/amd64 --pull --push \
      -t "$api_image" -f server/Dockerfile server
  fi
fi

echo "=== Deploy API ==="
echo "API image: $api_image"

# Public API access supports the existing Nginx proxy.
# Application authentication remains enabled.
# Normal Alembic migrations run before API startup.
gc run deploy "$API_SERVICE" \
  --region="$REGION" \
  --image="$api_image" \
  --service-account="$api_account" \
  --port=8000 \
  --ingress=all \
  --allow-unauthenticated \
  --add-cloudsql-instances="$CLOUD_SQL_CONNECTION" \
  --env-vars-file="$work/env.json" \
  --update-secrets="$(cat "$work/secret-args")" \
  --max-instances=1 \
  --command=sh \
  --args='-c,alembic upgrade head && exec uvicorn app.main:app --host 0.0.0.0 --port 8000' \
  --quiet

gc run services update-traffic "$API_SERVICE" \
  --region="$REGION" --to-latest --quiet

api_url=$(gc run services describe "$API_SERVICE" \
  --region="$REGION" --format='value(status.url)')

curl --fail --silent --show-error \
  --retry 5 --retry-all-errors --max-time 60 \
  "$api_url/api/health" >/dev/null

echo "=== Build and deploy fresh frontend image ==="

cat > "$work/web/nginx.conf" <<EOF
server {
  listen 8080;
  server_name _;

  root /usr/share/nginx/html;
  index index.html;

  location /api/ {
    proxy_pass ${api_url};
    proxy_ssl_server_name on;
    proxy_set_header Host ${api_url#https://};
    proxy_set_header X-Forwarded-Proto https;
    proxy_http_version 1.1;
    proxy_read_timeout 120s;
  }

  location = /index.html {
    add_header Cache-Control "no-cache, no-store, must-revalidate" always;
  }

  location = /version.json {
    add_header Cache-Control "no-store" always;
  }

  location /assets/ {
    try_files \$uri =404;
    add_header Cache-Control "public, max-age=31536000, immutable";
  }

  location / {
    add_header Cache-Control "no-cache, no-store, must-revalidate" always;
    try_files \$uri \$uri/ /index.html;
  }
}
EOF

python3 - "$work/web/dist/version.json" "$commit" <<'PY'
import json
import sys

json.dump({"commit": sys.argv[2]}, open(sys.argv[1], "w"))
PY

cat > "$work/web/Dockerfile" <<'EOF'
FROM nginx:alpine
COPY nginx.conf /etc/nginx/conf.d/default.conf
COPY dist /usr/share/nginx/html
EXPOSE 8080
CMD ["nginx", "-g", "daemon off;"]
EOF

web_image="$registry/rove-web:$tag"

docker buildx build \
  --platform linux/amd64 \
  --pull --no-cache --push \
  -t "$web_image" "$work/web"

gc run deploy "$WEB_SERVICE" \
  --region="$REGION" \
  --image="$web_image" \
  --service-account="$web_account" \
  --port=8080 \
  --ingress=all \
  --allow-unauthenticated \
  --quiet

web_url=$(gc run services describe "$WEB_SERVICE" \
  --region="$REGION" --format='value(status.url)')

gc run services update "$API_SERVICE" \
  --region="$REGION" \
  --update-env-vars="CORS_ORIGIN=$web_url" \
  --quiet

gc run services update-traffic "$API_SERVICE" \
  --region="$REGION" --to-latest --quiet

gc run services update-traffic "$WEB_SERVICE" \
  --region="$REGION" --to-latest --quiet

echo "=== Verify live deployment ==="

curl --fail --silent --show-error \
  --retry 5 --retry-all-errors --max-time 60 \
  "$web_url/api/health" >/dev/null

curl --fail --silent --show-error --max-time 60 \
  "$web_url/version.json?deploy=$tag" > "$work/version.json"

python3 - "$work/version.json" "$commit" <<'PY'
import json
import sys

if json.load(open(sys.argv[1])).get("commit") != sys.argv[2]:
    raise SystemExit("Live frontend commit mismatch.")

print("PASS: live UI matches newly built source.")
PY

curl --fail --silent --show-error --max-time 60 \
  "$web_url/api/catalogue/vehicles" > "$work/catalogue.json"

python3 - "$work/catalogue.json" <<'PY'
import json
import sys

vehicles = json.load(open(sys.argv[1]))

if not isinstance(vehicles, list):
    raise SystemExit("Unexpected catalogue response.")

print("Public catalogue vehicles:", len(vehicles))

if not vehicles:
    print("WARNING: Inventory is empty; dropdowns will have no values.")
PY

echo
echo "DEMO URL: $web_url/?deploy=$tag"
echo "Frontend: $BRANCH at $commit"
echo "Protected generated credential files: $ROVE_CREDENTIALS_DIR"
echo "Open credential files locally; do not paste them into chat."
echo "Use a private browser window for the demo."

cd ~/Documents/repo/rove || exit 1
bash -n deploy/new-deploy.sh

ALLOW_DB_PASSWORD_ROTATION=1 BRANCH=main \
  bash deploy/new-deploy.sh

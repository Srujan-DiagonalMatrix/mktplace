bash <<'ROVE_UPDATE_AOS'
set +x
set -euo pipefail

ca_override=".rove-aos-ca.mYJ1fy/compose.ca.yaml"
if [ ! -f "$ca_override" ]; then
  printf 'Working CA override missing: %s\n' "$ca_override" >&2
  exit 1
fi

compose_file=""
for candidate in compose.yaml compose.yml docker-compose.yml docker-compose.yaml; do
  if [ -f "$candidate" ]; then
    compose_file="$candidate"
    break
  fi
done
if [ -z "$compose_file" ] || [ ! -f .env.docker ]; then
  printf 'Run this from the Rove repo with its Compose file and .env.docker.\n' >&2
  exit 1
fi

credential_override="$(mktemp "$PWD/.rove-aos-ca.mYJ1fy/credentials.XXXXXX")"
trap 'rm -f "$credential_override"' EXIT
cat >"$credential_override" <<'YAML'
services:
  api:
    environment:
      AOS_CLIENT_ID: bcarove
      AOS_CLIENT_SECRET: ${ROVE_TEST_AOS_SECRET:?Enter the AutosOnShow secret}
YAML

printf 'New AutosOnShow secret for bcarove: ' > /dev/tty
IFS= read -r -s ROVE_TEST_AOS_SECRET < /dev/tty
printf '\n' > /dev/tty
[ -n "$ROVE_TEST_AOS_SECRET" ] || { printf 'No secret entered.\n'; exit 1; }
export ROVE_TEST_AOS_SECRET

docker compose --env-file .env.docker \
  -f "$compose_file" -f "$ca_override" -f "$credential_override" \
  up -d --no-deps --force-recreate api </dev/null
unset ROVE_TEST_AOS_SECRET

docker compose --env-file .env.docker -f "$compose_file" -f "$ca_override" \
  exec -T api python -u - <<'PY'
import os
from datetime import datetime, timezone

import httpx

print("Runtime CA:", os.environ.get("SSL_CERT_FILE"))
print("UTC:", datetime.now(timezone.utc).isoformat(timespec="seconds"))
try:
    response = httpx.post(
        "https://br-api.aos.tv/token/",
        data={
            "grant_type": "client_credentials",
            "client_id": os.environ["AOS_CLIENT_ID"],
            "client_secret": os.environ["AOS_CLIENT_SECRET"],
        },
        timeout=15,
    )
except httpx.RequestError as exc:
    raise SystemExit(f"{type(exc).__name__}: {exc}") from None

print("Upstream HTTP:", response.status_code)
print("Content-Type:", response.headers.get("content-type", "(absent)"))
if response.is_success:
    try:
        payload = response.json()
    except ValueError:
        payload = {}
    token = payload.get("access_token") if isinstance(payload, dict) else None
    print("Valid access_token:", isinstance(token, str) and bool(token.strip()))
PY
ROVE_UPDATE_AOS

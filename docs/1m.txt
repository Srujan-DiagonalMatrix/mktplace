docker compose exec -T api python - <<'PY'
import asyncio, httpx

async def main():
    try:
        async with httpx.AsyncClient(timeout=10) as client:
            # No credentials: this checks only whether the container can connect.
            response = await client.post(
                "https://br-api.aos.tv/token/",
                data={"grant_type": "client_credentials"},
            )
            print("Container reached AutosOnShow; HTTP", response.status_code)
    except Exception as exc:
        cause = exc.__cause__
        print("Container connection failed:", type(exc).__name__)
        print("Underlying cause:", type(cause).__name__ if cause else "none")

asyncio.run(main())
PY

python3 - <<'PY'
import json, urllib.error, urllib.request

url = "http://localhost:9000/api/catalogue/aos-token"
try:
    response = urllib.request.urlopen(url, timeout=25)
    body = json.loads(response.read())
    print("Rove API HTTP:", response.status)
    print("Token received:", "yes (value hidden)" if body.get("token") else "no")
except urllib.error.HTTPError as exc:
    body = json.loads(exc.read())
    print("Rove API HTTP:", exc.code)
    print("API detail:", body.get("detail", "No detail returned"))
except Exception as exc:
    print("Could not call local Rove API:", type(exc).__name__, str(exc))
PY

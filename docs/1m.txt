PROJECT="tactical-innovation-f2b97b"
REGION="europe-west1"
SERVICE="rove-api"

echo "============================================================"
echo "1. CHECK WHETHER ROVE-API EXISTS"
echo "============================================================"

gcloud run services describe "$SERVICE" \
  --project="$PROJECT" \
  --region="$REGION" \
  --format="yaml(metadata.name,status)" \
  2>&1 || true


echo
echo "============================================================"
echo "2. CLOUD RUN ADMIN / AUDIT EVENTS - LAST 2 HOURS"
echo "============================================================"

gcloud logging read \
  'protoPayload.serviceName="run.googleapis.com"' \
  --project="$PROJECT" \
  --freshness=2h \
  --limit=100 \
  --format="table(
timestamp,
severity,
protoPayload.methodName,
protoPayload.status.code,
protoPayload.status.message
)"


echo
echo "============================================================"
echo "3. CLOUD RUN ERRORS ONLY"
echo "============================================================"

gcloud logging read \
  'protoPayload.serviceName="run.googleapis.com" AND severity>=ERROR' \
  --project="$PROJECT" \
  --freshness=2h \
  --limit=50 \
  --format="json"


echo
echo "============================================================"
echo "4. CHECK POSTGRESQL"
echo "============================================================"

gcloud sql instances describe rove-postgres \
  --project="$PROJECT" \
  --format="yaml(
name,
state,
connectionName
)" \
  2>&1 || true


echo
echo "============================================================"
echo "5. CHECK API IMAGE EXISTS"
echo "============================================================"

gcloud artifacts docker images list \
  "europe-west1-docker.pkg.dev/$PROJECT/cloud-run-source-deploy" \
  --project="$PROJECT" \
  --include-tags \
  --filter="package~rove-api" \
  --sort-by="~UPDATE_TIME" \
  --limit=5 \
  --format="table(package,version,updateTime)"


echo
echo "============================================================"
echo " ANALYSIS COMPLETE"
echo "============================================================"

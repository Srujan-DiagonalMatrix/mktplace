const submissionMonth = new Intl.DateTimeFormat('en-GB', { month: 'short', year: 'numeric', timeZone: 'UTC' });

function quotedAmount(value) {
  if (value == null || value === '') return null;
  const amount = Number(value);
  return Number.isFinite(amount) && amount >= 0 ? Math.round(amount * 100) : null;
}

function reservationPipeline(records, preferredCurrency) {
  const active = records.filter(record => record.has_active_reservation
    && ['submitted', 'processing'].includes(record.status));
  const quoted = active.map(record => {
    const quote = record.selections || {};
    return {
      record,
      quote,
      currency: quote.currency,
      total: quotedAmount(quote.bundle_total) ?? quotedAmount(quote.total_payable),
    };
  }).filter(item => item.total != null && /^[A-Z]{3}$/.test(item.currency || ''));
  const currencies = [...new Set(quoted.map(item => item.currency))].sort();
  const currency = currencies.includes(preferredCurrency) ? preferredCurrency : currencies[0] || 'GBP';
  const currencyFormat = new Intl.NumberFormat('en-GB', { style: 'currency', currency });
  const formatter = pennies => currencyFormat.format(pennies / 100);
  const visible = quoted.filter(item => item.currency === currency);
  const product = ({ quote }) => quote.finance_product_name;
  const chart = (title, group, amount = item => item.total) => {
    const totals = new Map();
    for (const item of visible) {
      const value = amount(item);
      if (value == null) continue;
      const label = group(item) || 'Not recorded';
      totals.set(label, (totals.get(label) || 0) + value);
    }
    return { title, formatter, slices: [...totals].map(([label, value]) => ({ label, value })) };
  };

  return {
    activeCount: active.length,
    quotedCount: visible.length,
    currency,
    currencies,
    charts: [
      chart('Potential contract value by product and status', ({ quote, record }) =>
        [quote.finance_product_name, record.status?.replaceAll('_', ' ')].filter(Boolean).join(' · ')),
      chart('Potential contract value by reservation and month', ({ record }) => {
        const reservation = ({
          '48_hours': '48 hours', '30_days': '30 days', buy_now: 'Buy now',
        })[record.reservation_type];
        const time = Date.parse(record.submitted_at);
        const month = Number.isFinite(time) ? submissionMonth.format(time) : null;
        return [reservation, month].filter(Boolean).join(' · ');
      }),
      chart('Potential contract value by term and fuel type', ({ quote }) =>
        [quote.term_months && `${quote.term_months} months`, quote.vehicle_fuel_type]
          .filter(Boolean).join(' · ')),
      (() => {
        const totals = new Map();
        for (const item of visible) {
          for (const [kind, raw] of [['Monthly', item.quote.monthly_payment], ['Upfront', item.quote.upfront_payment]]) {
            const value = quotedAmount(raw);
            if (value == null) continue;
            const label = `${product(item) || "Not recorded"} · ${kind}`;
            totals.set(label, (totals.get(label) || 0) + value);
          }
        }
        return { title: 'Potential monthly and upfront payments by product', formatter, slices: [...totals].map(([label, value]) => ({ label, value })) };
      })(),
    ],
  };
}

function snapshotDetails(value, label) {
  if (value == null) return null;
  return <details><summary>{label}</summary><pre className="admin-application-snapshot">{JSON.stringify(value, null, 2)}</pre></details>;
}

function applicationValue(record, key) {
  return Object.hasOwn(record, key) ? record[key] : record.selections?.[key];
}

function downloadApplicationDetails(records) {
  const headers = APPLICATION_COLUMNS.map(([, label]) => label);
  const rows = records.map(record => APPLICATION_COLUMNS.map(([key]) => {
    const value = applicationValue(record, key);
    const text = value != null && typeof value === 'object' ? JSON.stringify(value) : String(value ?? '');
    return /^[\t\r\n]|^\s*[=+@-]/.test(text) ? "'" + text : text;
  }));
  const csv = [headers, ...rows].map(row => row.map(csvCell).join(',')).join('\r\n') + '\r\n';
  const url = URL.createObjectURL(new Blob(['\uFEFF', csv], { type: 'text/csv;charset=utf-8' }));
  const link = document.createElement('a');
  link.href = url;
  link.download = 'rove-customer-application-details.csv';
  link.click();
  window.setTimeout(() => URL.revokeObjectURL(url), 1000);
}

function DetailFields({ fields }) {
  return <dl className="admin-application-detail-fields">
    {fields.map(([label, value]) => <div key={label}><dt>{label}</dt><dd>{value ?? '—'}</dd></div>)}
  </dl>;
}

function snapshotFields(value, prefix = '') {
  if (value == null || typeof value !== 'object') {
    return [[prefix, typeof value === 'boolean' ? (value ? 'Yes' : 'No') : value]];
  }
  const entries = Object.entries(value);
  if (!entries.length) return [[prefix || 'Details', '—']];
  return entries.flatMap(([key, child]) => {
    const label = Array.isArray(value) ? String(Number(key) + 1) : key.replaceAll('_', ' ');
    return snapshotFields(child, prefix ? `${prefix} / ${label}` : label);
  });
}

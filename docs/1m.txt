bash <<'BCA_CHECK'
set +x
set -euo pipefail
umask 077

work_dir=$(mktemp -d "${TMPDIR:-/tmp}/bca-check.XXXXXX")
trap 'rm -f "$work_dir/token.json"' EXIT

printf 'AutosOnShow client secret for bcarove: ' >/dev/tty
IFS= read -r -s bca_secret </dev/tty
printf '\n' >/dev/tty
[ -n "$bca_secret" ] || { echo 'No secret entered.'; exit 1; }

token_status=$(printf '%s' "$bca_secret" | curl -sS --max-time 20 \
  -o "$work_dir/token.json" -w '%{http_code}' \
  --data-urlencode 'grant_type=client_credentials' \
  --data-urlencode 'client_id=bcarove' \
  --data-urlencode 'client_secret@-' \
  'https://br-api.aos.tv/token/')
unset bca_secret

printf 'Token endpoint HTTP %s\n' "$token_status"
[ "$token_status" = 200 ] || exit 1

bca_token=$(python3 -c '
import json, sys
token = json.load(sys.stdin).get("access_token")
if not isinstance(token, str) or not token.strip():
    sys.exit("No access token returned.")
print(token)
' <"$work_dir/token.json")

printf 'Authorization: Bearer %s\n' "$bca_token" | curl -sS --max-time 20 \
  --header @- --header 'Accept: application/json' \
  -o "$work_dir/vehicle.json" -w 'Vehicle endpoint HTTP %{http_code}\n' \
  'https://br-api.aos.tv/vehicles/BF73ATX'
unset bca_token
printf 'Vehicle response saved to %s\n' "$work_dir/vehicle.json"
BCA_CHECK

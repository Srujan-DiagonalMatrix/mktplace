cd /Users/srujan.kumar/Documents/git/rove

PROJECT="tactical-innovation-f2b97b"
REGION="europe-west1"

SERVICE="rove-api"

DB_INSTANCE="rove-postgres"
DB_NAME="rove"
DB_USER="postgres"
DB_PASSWORD='RovePov#2026Secure!'

IMAGE="europe-west1-docker.pkg.dev/tactical-innovation-f2b97b/cloud-run-source-deploy/rove-api:20261029-142016"


echo "============================================================"
echo "1. GET CLOUD SQL CONNECTION"
echo "============================================================"

CONNECTION_NAME="$(gcloud sql instances describe "$DB_INSTANCE" \
  --project="$PROJECT" \
  --format='value(connectionName)')"

echo "$CONNECTION_NAME"


echo
echo "============================================================"
echo "2. URL-ENCODE PASSWORD"
echo "============================================================"

DB_PASSWORD_URL="$(
  DB_PASSWORD="$DB_PASSWORD" python3 <<'PY'
import os
from urllib.parse import quote
print(quote(os.environ["DB_PASSWORD"], safe=""))
PY
)"

DATABASE_URL="postgresql+psycopg://${DB_USER}:${DB_PASSWORD_URL}@/${DB_NAME}?host=/cloudsql/${CONNECTION_NAME}"


echo
echo "============================================================"
echo "3. GET SERVICE ACCOUNT"
echo "============================================================"

PROJECT_NUMBER="$(gcloud projects describe "$PROJECT" \
  --format='value(projectNumber)')"

SERVICE_ACCOUNT="${PROJECT_NUMBER}-compute@developer.gserviceaccount.com"

echo "$SERVICE_ACCOUNT"


echo
echo "============================================================"
echo "4. DEPLOY API WITH DATABASE"
echo "============================================================"

gcloud run deploy "$SERVICE" \
  --project="$PROJECT" \
  --region="$REGION" \
  --platform=managed \
  --image="$IMAGE" \
  --port=8080 \
  --add-cloudsql-instances="$CONNECTION_NAME" \
  --service-account="$SERVICE_ACCOUNT" \
  --set-env-vars="DATABASE_URL=${DATABASE_URL},APP_ENV=production" \
  --no-invoker-iam-check \
  --quiet \
  --verbosity=debug


echo
echo "============================================================"
echo "5. TEST API"
echo "============================================================"

API_URL="$(gcloud run services describe "$SERVICE" \
  --project="$PROJECT" \
  --region="$REGION" \
  --format='value(status.url)')"

echo "API URL:"
echo "$API_URL"

echo
echo "Docs:"
curl -s -o /dev/null \
  -w "HTTP %{http_code}\n" \
  "$API_URL/docs"

echo
echo "Database health:"
curl -s "$API_URL/api/health"

echo

bash <<'ROVE_READ_BUILD'
diag_dir=$(ls -td "${TMPDIR:-/tmp}"/rove-build-diag.* 2>/dev/null | head -n 1)

if [ -z "$diag_dir" ] || [ ! -f "$diag_dir/build-private.log" ]; then
  echo "Diagnostic build log not found."
  exit 1
fi

log_file="$diag_dir/build-private.log"

redact() {
  sed -E \
    -e 's#(https?://)[^/@[:space:]]+:[^/@[:space:]]+@#\1[REDACTED]@#g' \
    -e 's#([Aa]uthorization:).*#\1 [REDACTED]#g'
}

echo "--- Log size and last modification ---"
stat -f 'Size: %z bytes; Last updated: %Sm' \
  -t '%Y-%m-%d %H:%M:%S' "$log_file"

echo
echo "--- Latest 100 build lines ---"
tail -n 100 "$log_file" | redact

echo
echo "--- Local build command processes ---"
ps -axo pid,etime,comm |
  awk 'NR == 1 || /docker|buildx|buildkit/'
ROVE_READ_BUILD

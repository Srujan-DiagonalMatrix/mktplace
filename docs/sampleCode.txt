
(
  cd /Users/srujan.kumar/Documents/git/rove || exit 1

  echo "===== ROVE CORPORATE SSL FIX ====="

  echo "Checking Docker..."
  docker info >/dev/null 2>&1 || {
    echo "ERROR: Docker engine unavailable"
    exit 1
  }

  echo "Checking corporate certificates..."

  if [ ! -s company-ca.pem ] || ! grep -q 'BEGIN CERTIFICATE' company-ca.pem; then
    echo "No valid existing company-ca.pem found."
    echo "Exporting macOS trusted certificates using system binary..."

    /usr/bin/security find-certificate -a -p \
      /Library/Keychains/System.keychain \
      > company-ca.pem || {
        echo "ERROR: Certificate export failed."
        echo "Ask bank IT for the approved corporate CA PEM bundle."
        exit 1
      }
  fi

  if ! grep -q 'BEGIN CERTIFICATE' company-ca.pem; then
    echo "ERROR: Certificate bundle is empty or invalid."
    exit 1
  fi

  echo "Corporate certificate file available."

  cat > docker-compose.corporate.yml <<'YAML'
services:
  api:
    build:
      secrets:
        - corporate_ca

secrets:
  corporate_ca:
    file: ./company-ca.pem
YAML

  grep -qxF 'docker-compose.corporate.yml' .git/info/exclude 2>/dev/null ||
    echo 'docker-compose.corporate.yml' >> .git/info/exclude

  export DOCKER_BUILDKIT=1

  echo "Validating Docker configuration..."

  docker compose \
    -f docker-compose.yml \
    -f docker-compose.corporate.yml \
    config --quiet || exit 1

  echo "Building API and Web..."

  docker compose \
    -f docker-compose.yml \
    -f docker-compose.corporate.yml \
    build api web || exit 1

  echo "===== SUCCESS: DOCKER IMAGES BUILT ====="
)

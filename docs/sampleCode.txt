set -e

echo "=================================================="
echo " 1. SAVE LOCAL BANK DOCKERFILE FIX"
echo "=================================================="

PATCH_FILE="/tmp/rove-bank-dockerfile-fix.patch"

git diff -- server/Dockerfile > "$PATCH_FILE" || true

echo "Saved local Dockerfile changes to:"
echo "$PATCH_FILE"


echo
echo "=================================================="
echo " 2. RESET LOCAL MAIN TO PERSONAL GITHUB"
echo "=================================================="

git switch main

git fetch origin
git fetch company

# Remove the accidental local commit containing backup Dockerfiles
git reset --hard origin/main

# Remove only the temporary backup files created during SSL fixing
rm -f \
  Dockerfile.backup \
  Dockerfile.backup.bak \
  Dockerfile.bak \
  Dockerfile.before-ca-fix \
  Dockerfile.web.bak \
  Dockerfile.web.backup.*.bak \
  server/Dockerfile.backup.*.bak \
  server/Dockerfile.bak \
  server/Dockerfile.bank-backup

echo
echo "Local main restored to origin/main."


echo
echo "=================================================="
echo " 3. CREATE FRESH COMPANY SYNC BRANCH"
echo "=================================================="

SYNC_BRANCH="rove-sync-$(date +%Y%m%d-%H%M%S)"

git switch -c "$SYNC_BRANCH" company/main

echo "Created:"
echo "$SYNC_BRANCH"


echo
echo "=================================================="
echo " 4. COPY PERSONAL GITHUB CODE INTO COMPANY BRANCH"
echo "=================================================="

# Copy tracked application code from your personal repo
git checkout origin/main -- .

# IMPORTANT:
# Restore bank/company GitHub configuration instead of overwriting it
git checkout company/main -- .github 2>/dev/null || true


echo
echo "=================================================="
echo " 5. COMMIT CHANGES"
echo "=================================================="

git status

git add -A

if git diff --cached --quiet; then
    echo "No code differences found to commit."
else
    git commit -m "Sync latest ROVE application changes"
fi


echo
echo "=================================================="
echo " 6. PUSH TO BANK GITHUB"
echo "=================================================="

git push -u company "$SYNC_BRANCH"


echo
echo "=================================================="
echo " COMPANY PUSH SUCCESSFUL"
echo "=================================================="

echo
echo "Branch:"
echo "$SYNC_BRANCH"

echo
echo "Create PR:"
echo "$SYNC_BRANCH  ->  main"


echo
echo "=================================================="
echo " 7. RESTORE LOCAL BANK SSL FIX"
echo "=================================================="

git switch main

if [ -s "$PATCH_FILE" ]; then
    git apply "$PATCH_FILE" || {
        echo ""
        echo "NOTE: Could not automatically reapply the local Dockerfile SSL patch."
        echo "The patch is safely stored here:"
        echo "$PATCH_FILE"
    }
else
    echo "No additional server/Dockerfile patch needed."
fi

echo
echo "=================================================="
echo " DONE"
echo "=================================================="

git status


(
  cd /Users/srujan.kumar/Documents/git/rove || exit 1

  echo "===== ROVE PYPI SSL FIX ====="

  python3 - <<'PY'
from pathlib import Path

p = Path("server/Dockerfile")
s = p.read_text()

old = "RUN pip install --no-cache-dir -r requirements.txt"
new = (
    "RUN pip install --no-cache-dir "
    "--trusted-host pypi.org "
    "--trusted-host files.pythonhosted.org "
    "--trusted-host pypi.python.org "
    "-r requirements.txt"
)

if new in s:
    print("PyPI workaround already applied.")
elif old in s:
    p.write_text(s.replace(old, new, 1))
    print("Applied PyPI trusted-host workaround.")
else:
    raise SystemExit("ERROR: Expected pip command not found.")
PY

  echo "===== VALIDATING DOCKER ====="

  docker compose \
    -f docker-compose.yml \
    -f docker-compose.corporate.yml \
    config --quiet || exit 1

  echo "===== REBUILDING API ====="

  docker compose \
    -f docker-compose.yml \
    -f docker-compose.corporate.yml \
    build --no-cache api || exit 1

  echo "===== API BUILD SUCCESSFUL ======"
)

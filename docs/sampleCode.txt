from pathlib import Path

p = Path("Dockerfile")
s = p.read_text()

old = "RUN pip install --no-cache-dir -r requirements.txt"

new = """COPY company-ca.pem /usr/local/share/ca-certificates/company-ca.crt
RUN update-ca-certificates && \\
    PIP_CERT=/etc/ssl/certs/ca-certificates.crt \\
    REQUESTS_CA_BUNDLE=/etc/ssl/certs/ca-certificates.crt \\
    pip install --no-cache-dir -r requirements.txt"""

if old not in s:
    print("pip line not found - Dockerfile may already have been modified.")
else:
    p.write_text(s.replace(old, new))
    print("Dockerfile updated successfully.")
PY

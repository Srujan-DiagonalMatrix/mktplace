python3 <<'PY'
import json
import subprocess
from pathlib import Path
import re
import sys

print("\n=== ROVE BANK SSL FIX ===\n")

# ---------------------------------------------------------
# 1. Ask Docker Compose for the EXACT API build configuration
# ---------------------------------------------------------
try:
    raw = subprocess.check_output(
        ["docker", "compose", "config", "--format", "json"],
        text=True
    )
    cfg = json.loads(raw)
except Exception as e:
    print("ERROR: Unable to read docker compose configuration:")
    print(e)
    sys.exit(1)

services = cfg.get("services", {})

if "api" not in services:
    print("ERROR: Could not find service 'api' in docker compose.")
    print("Available services:", ", ".join(services.keys()))
    sys.exit(1)

build = services["api"].get("build")

if isinstance(build, str):
    context = Path(build)
    dockerfile_name = "Dockerfile"
elif isinstance(build, dict):
    context = Path(build.get("context", "."))
    dockerfile_name = build.get("dockerfile", "Dockerfile")
else:
    print("ERROR: API service has no Docker build configuration.")
    sys.exit(1)

dockerfile = context / dockerfile_name
dockerfile = dockerfile.resolve()

print("Compose API context :", context)
print("REAL API Dockerfile :", dockerfile)

if not dockerfile.exists():
    print("ERROR: Dockerfile does not exist:", dockerfile)
    sys.exit(1)

# ---------------------------------------------------------
# 2. Back it up
# ---------------------------------------------------------
backup = dockerfile.with_suffix(dockerfile.suffix + ".bank-backup")

if not backup.exists():
    backup.write_text(dockerfile.read_text())
    print("Backup created       :", backup)

text = dockerfile.read_text()

# ---------------------------------------------------------
# 3. Remove any previous experimental pip SSL modifications
#    and replace the ACTUAL pip install command
# ---------------------------------------------------------

# Restore various forms of our previous changes back to a known form
text = re.sub(
    r'RUN\s+(?:python\s+-m\s+)?pip\s+install\s+'
    r'(?:--trusted-host\s+\S+\s+)*'
    r'--no-cache-dir\s+(?:--trusted-host\s+\S+\s+)*'
    r'-r\s+requirements\.txt',
    'RUN pip install --no-cache-dir -r requirements.txt',
    text,
    flags=re.I | re.S
)

old = "RUN pip install --no-cache-dir -r requirements.txt"

new = """# BANK-PYPI-SSL-FIX
ENV PIP_TRUSTED_HOST="pypi.org files.pythonhosted.org pypi.python.org"
ENV PIP_DISABLE_PIP_VERSION_CHECK=1

RUN python -m pip install \\
    --no-cache-dir \\
    --trusted-host pypi.org \\
    --trusted-host files.pythonhosted.org \\
    --trusted-host pypi.python.org \\
    --index-url https://pypi.org/simple \\
    -r requirements.txt"""

if old not in text:
    print("\nERROR: Could not locate the failing pip command.")
    print("\nCurrent pip-related Dockerfile lines:\n")

    for i, line in enumerate(text.splitlines(), 1):
        if "pip" in line.lower():
            print(f"{i}: {line}")

    sys.exit(1)

text = text.replace(old, new, 1)
dockerfile.write_text(text)

print("\nDockerfile patched successfully.\n")

print("=== VERIFYING PATCH ===")

for i, line in enumerate(dockerfile.read_text().splitlines(), 1):
    if (
        "PIP_" in line
        or "pip install" in line
        or "--trusted-host" in line
        or "--index-url" in line
    ):
        print(f"{i}: {line}")

print("\n=== PATCH COMPLETE ===\n")
PY

STATUS=$?

if [ "$STATUS" -ne 0 ]; then
    echo "Patch failed. Docker build NOT started."
    exit "$STATUS"
fi

echo "=================================================="
echo " REMOVING OLD API IMAGE AND BUILD CACHE"
echo "=================================================="

docker compose stop api 2>/dev/null || true
docker compose rm -f api 2>/dev/null || true
docker image rm rove-api 2>/dev/null || true
docker builder prune -af

echo
echo "=================================================="
echo " BUILDING API FROM THE PATCHED DOCKERFILE"
echo "=================================================="

docker compose build \
  --no-cache \
  --pull \
  --progress=plain \
  api

BUILD_STATUS=$?

if [ "$BUILD_STATUS" -ne 0 ]; then
    echo
    echo "=================================================="
    echo " API BUILD FAILED - APPLICATION NOT STARTED"
    echo "=================================================="
    exit "$BUILD_STATUS"
fi

echo
echo "=================================================="
echo " API BUILD SUCCESSFUL"
echo "=================================================="

docker compose up -d

echo
echo "=================================================="
echo " ROVE STATUS"
echo "=================================================="

docker compose ps

echo
echo "=================================================="
echo " SUCCESS"
echo "=================================================="

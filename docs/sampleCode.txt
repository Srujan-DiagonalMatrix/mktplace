set -e

echo "=================================================="
echo " FIXING BANK CORPORATE SSL FOR DOCKER + PIP"
echo "=================================================="

# --------------------------------------------------
# 1. Export certificates trusted by this Mac
# --------------------------------------------------

CERT_FILE="company-ca-bundle.pem"

rm -f "$CERT_FILE"

echo "Exporting macOS System certificates..."

security find-certificate -a -p \
  /Library/Keychains/System.keychain \
  >> "$CERT_FILE" 2>/dev/null || true

security find-certificate -a -p \
  /System/Library/Keychains/SystemRootCertificates.keychain \
  >> "$CERT_FILE" 2>/dev/null || true

security find-certificate -a -p \
  "$HOME/Library/Keychains/login.keychain-db" \
  >> "$CERT_FILE" 2>/dev/null || true

echo "Certificates exported:"
grep -c "BEGIN CERTIFICATE" "$CERT_FILE" || true

# --------------------------------------------------
# 2. Backup Dockerfile
# --------------------------------------------------

if [ ! -f Dockerfile.bank-backup ]; then
    cp Dockerfile Dockerfile.bank-backup
fi

# --------------------------------------------------
# 3. Patch Python Dockerfile so pip uses Mac/bank CA
# --------------------------------------------------

python3 <<'PY'
from pathlib import Path
import re

p = Path("Dockerfile")
s = p.read_text()

# Remove previous version of this fix if already present
s = re.sub(
    r'\n?# BANK-CORPORATE-CA-FIX-START.*?# BANK-CORPORATE-CA-FIX-END\n?',
    '\n',
    s,
    flags=re.S
)

old_patterns = [
    r'RUN pip install --no-cache-dir\s+--trusted-host\s+pypi\.org\s+--trusted-host\s+files\.pythonhosted\.org\s+--trusted-host\s+pypi\.python\.org\s+-r requirements\.txt',
    r'RUN pip config set global\.trusted-host.*?requirements\.txt',
    r'RUN pip install --no-cache-dir -r requirements\.txt'
]

for pattern in old_patterns:
    if re.search(pattern, s, flags=re.S):
        s = re.sub(
            pattern,
            '''# BANK-CORPORATE-CA-FIX-START
COPY company-ca-bundle.pem /tmp/company-ca-bundle.pem

ENV SSL_CERT_FILE=/tmp/company-ca-bundle.pem
ENV REQUESTS_CA_BUNDLE=/tmp/company-ca-bundle.pem
ENV PIP_CERT=/tmp/company-ca-bundle.pem

RUN python -m pip install --no-cache-dir -r requirements.txt
# BANK-CORPORATE-CA-FIX-END''',
            s,
            count=1,
            flags=re.S
        )
        break
else:
    raise SystemExit("ERROR: Could not find the pip install line in Dockerfile.")

p.write_text(s)

print("Dockerfile successfully patched.")
PY

# --------------------------------------------------
# 4. Make sure Docker can COPY certificate
# --------------------------------------------------

if [ -f .dockerignore ]; then
    sed -i.bak '/company-ca-bundle\.pem/d' .dockerignore 2>/dev/null || true
fi

# Keep corporate certificates out of Git
touch .gitignore

grep -qxF "company-ca-bundle.pem" .gitignore || \
    echo "company-ca-bundle.pem" >> .gitignore

# --------------------------------------------------
# 5. Completely clear old API build cache
# --------------------------------------------------

echo
echo "Stopping old containers..."
docker compose down --remove-orphans || true

echo
echo "Clearing Docker build cache..."
docker builder prune -af

# --------------------------------------------------
# 6. Rebuild from scratch
# --------------------------------------------------

echo
echo "=================================================="
echo " REBUILDING ROVE"
echo "=================================================="

docker compose build --no-cache --pull --progress=plain

# --------------------------------------------------
# 7. Start application
# --------------------------------------------------

echo
echo "=================================================="
echo " STARTING ROVE"
echo "=================================================="

docker compose up -d

echo
echo "=================================================="
echo " CONTAINER STATUS"
echo "=================================================="

docker compose ps

echo
echo "=================================================="
echo " SSL FIX COMPLETE"
echo "=================================================="

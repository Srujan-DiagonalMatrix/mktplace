python3 <<'PY'
from pathlib import Path

p = Path(".dockerignore")

if not p.exists():
    p.write_text("")
    print("Created .dockerignore")
else:
    lines = p.read_text().splitlines()

    cleaned = []

    for line in lines:
        value = line.strip().rstrip("/")

        if value in {
            "dist",
            "./dist",
            "**/dist",
            "*/dist"
        }:
            print("Removing:", line)
            continue

        cleaned.append(line)

    p.write_text("\n".join(cleaned) + "\n")

print("Current .dockerignore:")
print("----------------------")
print(p.read_text())
PY

cp Dockerfile Dockerfile.backup 2>/dev/null || true

find . -name "Dockerfile*" -type f -print0 | while IFS= read -r -d '' f; do
  echo "Updating: $f"
  cp "$f" "$f.bak"
  sed -i 's|RUN pip install --no-cache-dir -r requirements.txt|RUN pip install --no-cache-dir --trusted-host pypi.org --trusted-host files.pythonhosted.org --trusted-host pypi.python.org -r requirements.txt|g' "$f"
done

docker compose down --remove-orphans
docker builder prune -f
docker compose build --no-cache
docker compose up

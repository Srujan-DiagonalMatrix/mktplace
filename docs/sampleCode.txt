
set -e

echo "===== ROVE CORPORATE DOCKER SSL FIX ====="

cd "$(git rev-parse --show-toplevel)"

echo "Branch: $(git branch --show-current)"

# 1. Export trusted certificates from macOS keychains
echo "Exporting corporate SSL certificates..."

CA_FILE="$PWD/company-ca.pem"
: > "$CA_FILE"

for KEYCHAIN in \
  /Library/Keychains/System.keychain \
  "$HOME/Library/Keychains/login.keychain-db" \
  /System/Library/Keychains/SystemRootCertificates.keychain
do
  if [ -f "$KEYCHAIN" ]; then
    security find-certificate -a -p "$KEYCHAIN" \
      >> "$CA_FILE" 2>/dev/null || true
  fi
done

if ! grep -q "BEGIN CERTIFICATE" "$CA_FILE"; then
  echo "ERROR: No certificates exported"
  exit 1
fi

echo "Certificate bundle created."

# 2. Create local Docker Compose corporate override
cat > docker-compose.corporate.yml <<'YAML'
services:
  api:
    build:
      secrets:
        - corporate_ca

secrets:
  corporate_ca:
    file: ./company-ca.pem
YAML

# 3. Exclude local override from Git
grep -qxF 'docker-compose.corporate.yml' .git/info/exclude \
  || echo 'docker-compose.corporate.yml' >> .git/info/exclude

# 4. Enable BuildKit
export DOCKER_BUILDKIT=1
export COMPOSE_DOCKER_CLI_BUILD=1

# 5. Validate configuration
docker compose \
  -f docker-compose.yml \
  -f docker-compose.corporate.yml \
  config --quiet

# 6. Build using corporate certificates
docker compose \
  -f docker-compose.yml \
  -f docker-compose.corporate.yml \
  build api web

echo "===== BUILD COMPLETE ====="
echo "Corporate SSL configuration applied."

######
docker compose \
  -f docker-compose.yml \
  -f docker-compose.corporate.yml \
  up -d

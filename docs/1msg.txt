#!/usr/bin/env bash
set -euo pipefail

PROJECT_DIR="/Users/srujan.kumar/Documents/git/rove"

GCP_PROJECT="tactical-innovation-f2b97b"
REGION="europe-west1"
REPOSITORY="cloud-run-source-deploy"

WEB_SERVICE="rove-web"
API_SERVICE="rove-api"

WEB_IMAGE="rove-web"
API_IMAGE="rove-api"

cd "$PROJECT_DIR"

echo "=================================================="
echo " ROVE - FULL GCP DEPLOYMENT"
echo " FRONTEND + API"
echo "=================================================="

gcloud config set project "$GCP_PROJECT" >/dev/null

if ! gcloud auth print-access-token >/dev/null 2>&1; then
    gcloud auth login
fi

gcloud auth configure-docker \
    "${REGION}-docker.pkg.dev" \
    --quiet >/dev/null


TIMESTAMP="$(date +%Y%m%d-%H%M%S)"

API_VERSION_IMAGE="${REGION}-docker.pkg.dev/${GCP_PROJECT}/${REPOSITORY}/${API_IMAGE}:${TIMESTAMP}"

WEB_VERSION_IMAGE="${REGION}-docker.pkg.dev/${GCP_PROJECT}/${REPOSITORY}/${WEB_IMAGE}:${TIMESTAMP}"


echo
echo "=================================================="
echo "[1/8] BUILD API"
echo "=================================================="

if [ ! -f "server/Dockerfile" ]; then
    echo "ERROR: server/Dockerfile not found."
    exit 1
fi

docker buildx build \
    --platform linux/amd64 \
    -f server/Dockerfile \
    -t "$API_VERSION_IMAGE" \
    --push \
    server


echo
echo "=================================================="
echo "[2/8] DEPLOY API TO CLOUD RUN"
echo "=================================================="

gcloud run deploy "$API_SERVICE" \
    --image "$API_VERSION_IMAGE" \
    --region "$REGION" \
    --platform managed \
    --port 8000 \
    --allow-unauthenticated \
    --quiet


echo
echo "=================================================="
echo "[3/8] GET API URL"
echo "=================================================="

API_URL="$(gcloud run services describe "$API_SERVICE" \
    --region "$REGION" \
    --format='value(status.url)')"

echo "API URL: $API_URL"


echo
echo "Testing API..."

curl -i "$API_URL/" | head -20 || true

echo
echo "Testing health endpoints..."

curl -i "$API_URL/health" | head -20 || true


echo
echo "=================================================="
echo "[4/8] BUILD REACT"
echo "=================================================="

rm -rf dist

NODE_USE_SYSTEM_CA=1 npm ci
NODE_USE_SYSTEM_CA=1 npm run build

if [ ! -d "dist" ]; then
    echo "ERROR: React dist directory not created."
    exit 1
fi


echo
echo "=================================================="
echo "[5/8] FIX DOCKERIGNORE"
echo "=================================================="

if [ -f ".dockerignore" ]; then

python3 <<'PY'
from pathlib import Path

p = Path(".dockerignore")

lines = p.read_text().splitlines()

ignore = {
    "dist",
    "./dist",
    "**/dist",
    "*/dist"
}

clean = []

for line in lines:

    value = line.strip().rstrip("/")

    if value in ignore:
        print("Removing:", line)
        continue

    clean.append(line)

p.write_text("\n".join(clean) + "\n")

PY

fi


echo
echo "=================================================="
echo "[6/8] CREATE NGINX API PROXY"
echo "=================================================="

API_HOST="$(echo "$API_URL" | sed 's#https://##' | sed 's#/$##')"

cat > nginx.gcp.conf <<EOF
server {

    listen 8080;
    server_name _;

    root /usr/share/nginx/html;
    index index.html;


    # -------------------------------
    # ROVE BACKEND API
    # -------------------------------

    location /api/ {

        proxy_ssl_server_name on;

        proxy_set_header Host ${API_HOST};

        proxy_set_header X-Real-IP \$remote_addr;
        proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto \$scheme;

        proxy_pass ${API_URL}/api/;
    }


    # -------------------------------
    # API HEALTH
    # -------------------------------

    location = /health {

        proxy_ssl_server_name on;

        proxy_set_header Host ${API_HOST};

        proxy_pass ${API_URL}/health;
    }


    # -------------------------------
    # REACT SPA
    # -------------------------------

    location / {

        try_files \$uri \$uri/ /index.html;

    }

}
EOF


cat > Dockerfile.gcp <<'EOF'
FROM nginx:alpine

COPY nginx.gcp.conf /etc/nginx/conf.d/default.conf

COPY dist /usr/share/nginx/html

EXPOSE 8080

CMD ["nginx", "-g", "daemon off;"]
EOF


echo
echo "Nginx configured:"
echo
cat nginx.gcp.conf


echo
echo "=================================================="
echo "[7/8] BUILD + PUSH WEB"
echo "=================================================="

docker buildx build \
    --platform linux/amd64 \
    -f Dockerfile.gcp \
    -t "$WEB_VERSION_IMAGE" \
    --push \
    .


echo
echo "=================================================="
echo "[8/8] DEPLOY WEB"
echo "=================================================="

gcloud run deploy "$WEB_SERVICE" \
    --image "$WEB_VERSION_IMAGE" \
    --region "$REGION" \
    --platform managed \
    --port 8080 \
    --allow-unauthenticated \
    --quiet


WEB_URL="$(gcloud run services describe "$WEB_SERVICE" \
    --region "$REGION" \
    --format='value(status.url)')"


echo
echo "=================================================="
echo " ROVE DEPLOYMENT COMPLETE"
echo "=================================================="

echo
echo "API:"
echo "$API_URL"

echo
echo "Website:"
echo "$WEB_URL"

echo
echo "Testing API through website:"
echo

curl -i "$WEB_URL/health" | head -20 || true

echo
echo "=================================================="

open "$WEB_URL" 2>/dev/null || true
